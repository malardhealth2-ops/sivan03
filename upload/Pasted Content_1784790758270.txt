```typescript
interface PassengerRegistration {
  // مرحله ۱: شماره موبایل + OTP
  phone: string; // 09XXXXXXXXX format
  otp_code: string;
  
  // مرحله ۲: اطلاعات شخصی
  full_name: string;
  email?: string;
  national_id?: string; // کد ملی (اختیاری ولی توصیه شده)
  gender: 'male' | 'female';
  birth_date?: Date;
  
  // مرحله ۳: تنظیم رمز عبور
  password: string; // min 8 chars, 1 uppercase, 1 number
  confirm_password: string;
  
  // مرحله ۴: تأیید شماره موبایل (SMS verification)
}

interface PassengerProfile {
  id: UUID;
  phone: string;
  full_name: string;
  email: string;
  avatar_url?: string;
  rating: number; // 0-5
  total_trips: number;
  wallet_balance: number; // موجودی کیف پول
  saved_addresses: Address[];
  payment_methods: PaymentMethod[];
  created_at: Date;
  status: 'active' | 'suspended' | 'banned';
}
```

```typescript
interface DriverRegistration {
  // مرحله ۱: شماره موبایل + OTP
  phone: string;
  
  // مرحله ۲: اطلاعات شخصی
  full_name: string;
  father_name: string;
  national_id: string; // اجباری
  birth_date: Date;
  gender: 'male' | 'female';
  marital_status: 'single' | 'married';
  
  // مرحله ۳: مدارک شناسایی (Upload)
  national_card_image: File; // تصویر کارت ملی (جلو و پشت)
  license_image: File; // گواهینامه رانندگی
  health_certificate?: File; // گواهی سلامت
  criminal_record?: File; // عدم سوءپیشینه
  
  // مرحله ۴: اطلاعات خودرو
  car_type: 'sedan' | 'suv' | 'van';
  car_brand: string; // مثلاً Hyundai
  car_model: string; // مثلاً Sonata
  car_year: number; // سال ساخت
  car_color: string;
  car_plate_number: string; // پلاک
  car_insurance_expiry: Date;
  technical_diagnosis_expiry: Date;
  car_images: File[]; // عکس‌های خودرو (۴ زاویه)
  
  // مرحله ۵: حساب بانکی
  sheba_number: string; // شبا (IRXXXXXXXXXXXXXXX)
  bank_name: string;
  account_owner: string;
  
  // مرحله ۶: تنظیم رمز و تأیید نهایی
  password: string;
  accept_terms: boolean;
}

interface DriverProfile {
  id: UUID;
  user_id: UUID;
  phone: string;
  full_name: string;
  avatar_url?: string;
  rating: number; // 0-5
  total_trips: number;
  acceptance_rate: number; // درصد پذیرش سفر
  completion_rate: number; // درصد تکمیل سفر
  online_status: 'online' | 'offline' | 'busy';
  current_location: GeoJSON.Point;
  vehicle: VehicleInfo;
  documents: DocumentStatus; // approved/pending/rejected
  earnings: {
    today: number;
    this_week: number;
    this_month: number;
    total: number;
  };
  joined_at: Date;
  verification_status: 
    | 'pending_review'
    | 'approved'
    | 'rejected'
    | 'suspended';
}
```

## 🗄️ **طراحی دیتابیس (Database Schema)**

### **جداول اصلی (Core Tables):**

```sql
-- جدول کاربران (Users)
CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    phone VARCHAR(11) UNIQUE NOT NULL,
    email VARCHAR(255) UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    full_name VARCHAR(100) NOT NULL,
    role ENUM('passenger', 'driver', 'admin') NOT NULL,
    avatar_url TEXT,
    is_active BOOLEAN DEFAULT true,
    is_verified BOOLEAN DEFAULT false,
    last_login_at TIMESTAMP WITH TIME ZONE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- جدول مسافران (Passengers)
CREATE TABLE passengers (
    id UUID PRIMARY KEY REFERENCES users(id),
    national_id VARCHAR(10) UNIQUE,
    gender ENUM('male', 'female'),
    birth_date DATE,
    wallet_balance DECIMAL(12, 2) DEFAULT 0,
    rating DECIMAL(2, 1) DEFAULT 5.0,
    total_trips INTEGER DEFAULT 0,
    referral_code VARCHAR(10) UNIQUE,
    referred_by UUID REFERENCES passengers(id)
);

-- جدول رانندگان (Drivers)
CREATE TABLE drivers (
    id UUID PRIMARY KEY REFERENCES users(id),
    national_id VARCHAR(10) UNIQUE NOT NULL,
    father_name VARCHAR(50),
    license_number VARCHAR(20),
    license_expiry DATE,
    health_cert_expiry DATE,
    verification_status ENUM('pending', 'approved', 'rejected', 'suspended') DEFAULT 'pending',
    rating DECIMAL(2, 1) DEFAULT 5.0,
    total_trips INTEGER DEFAULT 0,
    acceptance_rate DECIMAL(5, 2) DEFAULT 100.00,
    completion_rate DECIMAL(5, 2) DEFAULT 100.00,
    current_latitude DECIMAL(10, 8),
    current_longitude DECIMAL(11, 8),
    online_status ENUM('online', 'offline', 'busy') DEFAULT 'offline',
    balance DECIMAL(12, 2) DEFAULT 0, -- موجودی قابل تسویه
    sheba_number VARCHAR(26), -- IRXXXXXXXXXXXXXXX
    bank_name VARCHAR(50),
    account_owner VARCHAR(100),
    joined_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- جدول خودروها (Vehicles)
CREATE TABLE vehicles (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    driver_id UUID NOT NULL REFERENCES drivers(id),
    type ENUM('sedan', 'suv', 'van', 'electric') NOT NULL,
    brand VARCHAR(50) NOT NULL,
    model VARCHAR(50) NOT NULL,
    year INTEGER NOT NULL,
    color VARCHAR(30) NOT NULL,
    plate_number VARCHAR(10) UNIQUE NOT NULL,
    insurance_expiry DATE,
    tech_diagnosis_expiry DATE,
    status ENUM('active', 'inactive', 'under_maintenance') DEFAULT 'active',
    images JSONB, -- array of image URLs
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- جدول سفرها (Trips)
CREATE TABLE trips (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    passenger_id UUID NOT NULL REFERENCES passengers(id),
    driver_id UUID REFERENCES drivers(id),
    
    -- اطلاعات مسیر
    origin_address TEXT NOT NULL,
    origin_lat DECIMAL(10, 8) NOT NULL,
    origin_lng DECIMAL(11, 8) NOT NULL,
    destination_address TEXT NOT NULL,
    destination_lat DECIMAL(10, 8) NOT NULL,
    destination_lng DECIMAL(11, 8) NOT NULL,
    
    -- جزئیات سفر
    trip_type ENUM('economy', 'vip', 'luxury', 'van', 'electric') NOT NULL,
    status ENUM(
        'pending',        -- منتظر پذیرش راننده
        'accepted',       -- پذیرفته شده توسط راننده
        'driver_arrived', -- راننده رسید
        'in_progress',    -- در حال حرکت
        'completed',      -- تکمیل شده
        'cancelled',      -- لغو شده
        'no_show'         -- مسافر حاضر نشد
    ) DEFAULT 'pending',
    
    -- زمان‌بندی
    requested_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    accepted_at TIMESTAMP WITH TIME ZONE,
    picked_up_at TIMESTAMP WITH TIME ZONE,
    completed_at TIMESTAMP WITH TIME ZONE,
    cancelled_at TIMESTAMP WITH TIME ZONE,
    scheduled_for TIMESTAMP WITH TIME ZONE, -- برای رزرو آینده
    
    -- مالی
    base_fare DECIMAL(10, 2),
    distance_fare DECIMAL(10, 2),
    time_fare DECIMAL(10, 2),
    discount_amount DECIMAL(10, 2) DEFAULT 0,
    total_fare DECIMAL(10, 2) NOT NULL,
    commission_amount DECIMAL(10, 2), -- سهم سیستم
    driver_earnings DECIMAL(10, 2),   -- سهم راننده
    payment_method ENUM('cash', 'online', 'wallet') DEFAULT 'cash',
    payment_status ENUM('pending', 'paid', 'refunded') DEFAULT 'pending',
    
    -- اطلاعات اضافی
    distance_km DECIMAL(8, 2),
    duration_minutes INTEGER,
    passenger_count INTEGER DEFAULT 1,
    notes TEXT,
    cancel_reason TEXT,
    cancelled_by UUID REFERENCES users(id),
    
    -- امتیازدهی
    passenger_rating DECIMAL(2, 1),
    driver_rating DECIMAL(2, 1),
    passenger_review TEXT,
    driver_review TEXT,
    
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- جدول تراکنش‌ها (Transactions)
CREATE TABLE transactions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(id),
    trip_id UUID REFERENCES trips(id),
    type ENUM('credit', 'debit', 'commission', 'refund', 'topup', 'withdrawal') NOT NULL,
    amount DECIMAL(12, 2) NOT NULL,
    description TEXT,
    reference_id VARCHAR(100), -- Reference from payment gateway
    status ENUM('pending', 'completed', 'failed') DEFAULT 'pending',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- جدول کیف پول (Wallet)
CREATE TABLE wallets (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID UNIQUE NOT NULL REFERENCES users(id),
    balance DECIMAL(12, 2) DEFAULT 0,
    frozen_balance DECIMAL(12, 2) DEFAULT 0, -- برای تراکنش‌های در حال پردازش
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- جدول بلاگ (Blog Posts)
CREATE TABLE blog_posts (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    title VARCHAR(255) NOT NULL,
    slug VARCHAR(255) UNIQUE NOT NULL,
    excerpt TEXT,
    content TEXTB NOT NULL, -- Markdown or HTML
    featured_image_url TEXT,
    author_id UUID REFERENCES users(id),
    category_id UUID REFERENCES blog_categories(id),
    tags TEXT[],
    status ENUM('draft', 'published', 'archived') DEFAULT 'draft',
    published_at TIMESTAMP WITH TIME ZONE,
    view_count INTEGER DEFAULT 0,
    like_count INTEGER DEFAULT 0,
    comment_count INTEGER DEFAULT 0,
    meta_title VARCHAR(255),
    meta_description TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- جدول نظرات بلاگ (Comments)
CREATE TABLE blog_comments (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    post_id UUID NOT NULL REFERENCES blog_posts(id),
    user_id UUID REFERENCES users(id), -- can be null for guest
    parent_id UUID REFERENCES blog_comments(id), -- for replies
    content TEXT NOT NULL,
    is_approved BOOLEAN DEFAULT false,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- جدول تیکت‌های پشتیبانی (Support Tickets)
CREATE TABLE support_tickets (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(id),
    subject VARCHAR(255) NOT NULL,
    description TEXT NOT NULL,
    priority ENUM('low', 'medium', 'high', 'urgent') DEFAULT 'medium',
    status ENUM('open', 'in_progress', 'waiting', 'resolved', 'closed') DEFAULT 'open',
    assigned_to UUID REFERENCES users(id), -- admin/support agent
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    closed_at TIMESTAMP WITH TIME ZONE
);

-- جدول پیام‌های تیکت (Ticket Messages)
CREATE TABLE ticket_messages (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    ticket_id UUID NOT NULL REFERENCES support_tickets(id),
    sender_id UUID NOT NULL REFERENCES users(id),
    message TEXT NOT NULL,
    attachments JSONB, -- array of file URLs
    is_read BOOLEAN DEFAULT false,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- جدول کدهای OTP (One Time Passwords)
CREATE TABLE otp_codes (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    phone VARCHAR(11) NOT NULL,
    code VARCHAR(6) NOT NULL,
    purpose ENUM('registration', 'login', 'password_reset', 'transaction') NOT NULL,
    expires_at TIMESTAMP WITH TIME ZONE NOT NULL,
    used_at TIMESTAMP WITH TIME ZONE,
    attempts INTEGER DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- جدول نوتیفیکیشن‌ها (Notifications)
CREATE TABLE notifications (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(id),
    title VARCHAR(255) NOT NULL,
    message TEXT NOT NULL,
    type ENUM('info', 'success', 'warning', 'error') DEFAULT 'info',
    channel ENUM('sms', 'email', 'push', 'in_app') DEFAULT 'in_app',
    data JSONB, -- additional metadata
    is_read BOOLEAN DEFAULT false,
    sent_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- جدول تخفیف‌ها (Coupons/Discounts)
CREATE TABLE coupons (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    code VARCHAR(20) UNIQUE NOT NULL,
    description TEXT,
    discount_type ENUM('percentage', 'fixed') NOT NULL,
    discount_value DECIMAL(10, 2) NOT NULL,
    min_order_amount DECIMAL(10, 2) DEFAULT 0,
    max_discount_amount DECIMAL(10, 2),
    usage_limit INTEGER,
    usage_count INTEGER DEFAULT 0,
    valid_from TIMESTAMP WITH TIME ZONE NOT NULL,
    valid_until TIMESTAMP WITH TIME ZONE NOT NULL,
    is_active BOOLEAN DEFAULT true,
    applicable_to ENUM('all', 'new_users', 'specific_trip_type'),
    created_by UUID REFERENCES users(id),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- جدول لاگ فعالیت‌ها (Activity Logs)
CREATE TABLE activity_logs (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(id),
    action VARCHAR(100) NOT NULL,
    entity_type VARCHAR(50), -- e.g., 'trip', 'user', 'post'
    entity_id UUID,
    old_values JSONB,
    new_values JSONB,
    ip_address INET,
    user_agent TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);
```

### **الف) Authentication & Authorization:**

```javascript
// JWT Configuration
const jwtConfig = {
  accessToken: {
    expiresIn: '15m',           // Access token: 15 دقیقه
    algorithm: 'HS256'
  },
  refreshToken: {
    expiresIn: '7d',            // Refresh token: 7 روز
    algorithm: 'HS256'
  },
  secret: process.env.JWT_SECRET
};

// RBAC Roles
const roles = {
  PASSENGER: ['booking:read', 'booking:create', 'profile:read', 'profile:update'],
  DRIVER: [
    'trip:accept', 'trip:reject', 'trip:start', 'trip:complete',
    'location:update', 'earning:view', 'withdraw:create',
    'profile:read', 'profile:update'
  ],
  ADMIN: ['*'], // Full access
  SUPER_ADMIN: ['*'] + ['admin:user:delete', 'settings:manage']
};

// Middleware Example
const authorize = (requiredPermission) => async (req, res, next) => {
  const user = req.user;
  if (!user) return res.status(401).json({ error: 'Unauthorized' });
  
  const userRoles = await getUserRoles(user.id);
  const hasPermission = userRoles.some(role => 
    roles[role].includes(requiredPermission) || roles[role].includes('*')
  );
  
  if (!hasPermission) return res.status(403).json({ error: 'Forbidden' });
  next();
};
```
