   444→function parseSectionData(raw: { title?: string; subtitle?: string; body?: string } | undefined): Record<string, string> {
   445→  const result: Record<string, string> = {};
   446→  if (!raw) return result;
   447→  result.title = raw.title || '';
   448→  result.subtitle = raw.subtitle || '';
   449→  if (raw.body) {
   450→    try {
   451→      const parsed = JSON.parse(raw.body);
   452→      if (parsed && typeof parsed === 'object') {
   453→        for (const [k, v] of Object.entries(parsed)) {
   454→          if (typeof v === 'string') result[k] = v;
   455→          else if (Array.isArray(v)) result[k] = (v as string[]).join('\n');
   456→        }
   457→      } else {
   458→        result.body = raw.body;
   459→      }
   460→    } catch {
   461→      result.body = raw.body;
   462→    }
   463→  }
   464→  return result;
   465→}
   466→
   467→function serializeSectionData(data: Record<string, string>): { title: string; subtitle: string; body: string } {
   468→  const title = data.title || '';
   469→  const subtitle = data.subtitle || '';
   470→  const extra: Record<string, string | string[]> = {};
   471→  for (const [k, v] of Object.entries(data)) {
   472→    if (k === 'title' || k === 'subtitle') continue;
   473→    if (k === 'items' || k === 'body') {
   474→      extra[k] = v;
   475→    } else {
   476→      extra[k] = v;
   477→    }
   478→  }
   479→  const hasExtra = Object.keys(extra).length > 0;
   480→  return { title, subtitle, body: hasExtra ? JSON.stringify(extra) : '' };
   481→}

  1653→  const [loadingDevices, setLoadingDevices] = useState(true);

  1681→      // 1. Get VAPID public key

  1693→      if (permission !== 'granted') {
  1694→        toast.error('دسترسی نوتیفیکیشن رد شد. از تنظیمات مرورگر اجازه دهید.');
  1695→        setEnabling(false);

  1699→      const reg = await navigator.serviceWorker.ready;
  1700→      let sub = await reg.pushManager.getSubscription();

  1712→        method: 'POST',
  1713→        headers: { 'Content-Type': 'application/json' },

  1715→          endpoint: subJson.endpoint,

  1904→function urlBase64ToUint8Array(base64String: string): Uint8Array {
  1905→  const padding = '='.repeat((4 - (base64String.length % 4)) % 4);
  1906→  const base64 = (base64String + padding).replace(/-/g, '+').replace(/_/g, '/');
  1907→  const rawData = atob(base64);
  1908→  const output = new Uint8Array(rawData.length);
  1909→  for (let i = 0; i < rawData.length; ++i) output[i] = rawData.charCodeAt(i);
  1910→  return output;
  1911→}

  1913→/** Plays a short two-tone "ding" using the Web Audio API (no audio file needed). */
  1914→function playNotifSound() {
  1915→  try {
  1916→    const AudioCtx = window.AudioContext || (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;
  1917→    if (!AudioCtx) return;
  1918→    const ctx = new AudioCtx();
  1919→    const now = ctx.currentTime;
  1920→
  1921→    const playTone = (freq: number, start: number, duration: number) => {
  1922→      const osc = ctx.createOscillator();
  1923→      const gain = ctx.createGain();
  1924→      osc.type = 'sine';
  1925→      osc.frequency.setValueAtTime(freq, now + start);
  1926→      gain.gain.setValueAtTime(0, now + start);
  1927→      gain.gain.linearRampToValueAtTime(0.25, now + start + 0.02);
  1928→      gain.gain.exponentialRampToValueAtTime(0.001, now + start + duration);
  1929→      osc.connect(gain);
  1930→      gain.connect(ctx.destination);
  1931→      osc.start(now + start);
  1932→      osc.stop(now + start + duration + 0.05);
  1933→    };
  1934→
  1935→    // Pleasant two-tone ascending chime (gold/bell-like)
  1936→    playTone(880, 0, 0.18);     // A5
  1937→    playTone(1318.5, 0.12, 0.3); // E6
  1938→
  1939→    setTimeout(() => { try { ctx.close(); } catch { /* ignore */ } }, 600);
  1940→  } catch { /* audio not available — silent */ }
  1941→}
