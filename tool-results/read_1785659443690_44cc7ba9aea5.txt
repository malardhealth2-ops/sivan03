   170→   170→    const parsed = JSON.parse(state.categoryTopicIndex || '{}');
   171→   171→    for (const cat of Object.keys(categoryTopicIndex) as TopicCategory[]) {
   172→   172→      if (typeof parsed[cat] === 'number') {
   173→   173→        categoryTopicIndex[cat] = parsed[cat];

   175→   175→    }
   176→   176→    if (state.lastAutoGenAt) {
   177→   177→      lastGeneratedAt = state.lastAutoGenAt.toISOString();
   178→   178→    }

   209→   209→/**
   210→   210→ * Check if a topic is too similar to any existing published post.
   211→   211→ * Compares the topic's keyword words against post titles and tags.
   212→   212→ * Returns true if a duplicate is detected (>50% word overlap with an existing post).
   213→   213→ */
   214→   214→async function isDuplicateTopic(topic: Topic): Promise<boolean> {
   215→   215→  try {
   216→   216→    // Get all published post titles and tags
   217→   217→    const existingPosts = await db.blogPost.findMany({
   218→   218→      where: { status: 'published' },
   219→   219→      select: { title: true, tags: true },
   220→   220→    });
   221→   221→    if (existingPosts.length === 0) return false;
   222→   222→
   223→   223→    // Normalize the topic keyword into a set of meaningful words
   224→   224→    const topicWords = new Set(
   225→   225→      topic.keyword
   226→   226→        .replace(/[«»"'!?.,:;()\[\]{}،؛؟]/g, ' ')
   227→   227→        .split(/\s+/)
   228→   228→        .filter((w) => w.length > 2)
   229→   229→        .map((w) => w.trim())
   230→   230→    );
   231→   231→    // Also add key words from the title
   232→   232→    const titleWords = new Set(
   233→   233→      topic.title
   234→   234→        .replace(/[«»"'!?.,:;()\[\]{}،؛؟]/g, ' ')
   235→   235→        .split(/\s+/)
   236→   236→        .filter((w) => w.length > 2)
   237→   237→        .map((w) => w.trim())
   238→   238→    );
   239→   239→    // Merge both sets for comparison
   240→   240→    const allTopicWords = new Set([...topicWords, ...titleWords]);
   241→   241→    if (allTopicWords.size === 0) return false;
   242→   242→
   243→   243→    for (const post of existingPosts) {
   244→   244→      // Normalize post title and tags into a set of words
   245→   245→      const postText = `${post.title} ${post.tags || ''}`;
   246→   246→      const postWords = new Set(
   247→   247→        postText
   248→   248→          .replace(/[«»"'!?.,:;()\[\]{}،؛؟\[\]]/g, ' ')
   249→   249→          .split(/\s+/)
   250→   250→          .filter((w) => w.length > 2)
   251→   251→          .map((w) => w.trim())
   252→   252→      );
   253→   253→
   254→   254→      // Count overlapping words
   255→   255→      let overlap = 0;
   256→   256→      for (const w of allTopicWords) {
   257→   257→        if (postWords.has(w)) overlap++;
   258→   258→      }
   259→   259→      const similarity = overlap / allTopicWords.size;
   260→   260→      // If more than 50% of the topic's key words match, consider it duplicate
   261→   261→      if (similarity >= 0.5 && overlap >= 2) {
   262→   262→        console.log(`[blog-generator] duplicate detected: topic "${topic.title}" overlaps ${Math.round(similarity * 100)}% with existing post "${post.title}" (shared words: ${overlap}/${allTopicWords.size})`);
   263→   263→        return true;
   264→   264→      }
   265→   265→    }
   266→   266→    return false;
   267→   267→  } catch (err) {
   268→   268→    console.error('[blog-generator] duplicate check failed:', err);
   269→   269→    return false; // on error, allow generation
   270→   270→  }
   271→   271→}

   273→   273→/**
   274→   274→ * Pick a non-duplicate topic. Tries up to MAX_PICK_ATTEMPTS times to find
   275→   275→ * a topic that doesn't overlap with existing posts. If all attempts fail,
   276→   276→ * returns the last picked topic anyway (so generation doesn't block forever).
   277→   277→ */
   278→   278→const MAX_PICK_ATTEMPTS = 20;
   279→   279→
   280→   280→async function pickNonDuplicateTopic(): Promise<Topic> {
   281→   281→  for (let attempt = 0; attempt < MAX_PICK_ATTEMPTS; attempt++) {
   282→   282→    const topic = pickTopic();
   283→   283→    const dup = await isDuplicateTopic(topic);
   284→   284→    if (!dup) return topic;
   285→   285→    console.log(`[blog-generator] attempt ${attempt + 1}: topic "${topic.title}" is duplicate, trying next...`);
   286→   286→  }
   287→   287→  // All attempts exhausted — return last picked topic to avoid infinite loop
   288→   288→  console.log('[blog-generator] all pick attempts exhausted, using last topic');
   289→   289→  return pickTopic();
   290→   290→}
   291→   291→
   292→   292→function pickTopic(): Topic {
   293→   293→  const category = CATEGORY_ORDER[cycleIndex % CATEGORY_ORDER.length];
   294→   294→  cycleIndex++;
   295→   295→  const pool = TOPICS.filter((t) => t.category === category);
   296→   296→  const idx = categoryTopicIndex[category] % pool.length;
   297→   297→  categoryTopicIndex[category]++;
   298→   298→  return pool[idx];
   299→   299→}

   301→   301→function makeSlug(title: string): string {
   302→   302→  const base = title
   303→   303→    .replace(/[«»"'!?.,:;()\[\]{}]/g, '')
   304→   304→    .replace(/\s+/g, '-')
   305→   305→    .replace(/-+/g, '-')
   306→   306→    .trim()
   307→   307→    .slice(0, 60);
   308→   308→  const suffix = Date.now().toString(36).slice(-6);
   309→   309→  return `${base}-${suffix}`;
   310→   310→}

   312→   312→function parseArticle(raw: string): {
   313→   313→  title: string;
   314→   314→  excerpt: string;
   315→   315→  html: string;
   316→   316→  tags: string[];
   317→   317→  metaDescription: string;
   318→   318→} | null {
   319→   319→  if (!raw) return null;
   320→   320→  const lines = raw.split('\n');
   321→   321→
   322→   322→  let title = '';
   323→   323→  let excerpt = '';
   324→   324→  let metaDescription = '';
   325→   325→  let tags: string[] = [];
   326→   326→  const htmlLines: string[] = [];
   327→   327→
   328→   328→  for (const line of lines) {
   329→   329→    const trimmed = line.trim();
   330→   330→    if (!trimmed) continue;
   331→   331→    if (trimmed.startsWith('عنوان:')) {
   332→   332→      title = trimmed.slice(6).trim().replace(/^["'«]|["'»]$/g, '');
   333→   333→    } else if (trimmed.startsWith('خلاصه:')) {
   334→   334→      excerpt = trimmed.slice(6).trim().replace(/^["'«]|["'»]$/g, '');
   335→   335→    } else if (trimmed.startsWith('متا:')) {
   336→   336→      metaDescription = trimmed.slice(4).trim().replace(/^["'«]|["'»]$/g, '');
   337→   337→    } else if (trimmed.startsWith('برچسب‌ها:')) {
   338→   338→      const rest = trimmed.replace(/^برچسب‌ها:\s*/, '');
   339→   339→      tags = rest
   340→   340→        .split(',')
   341→   341→        .map((t) => t.trim().replace(/^["'«]|["'»]$/g, ''))
   342→   342→        .filter(Boolean)
   343→   343→        .slice(0, 6);
   344→   344→    } else {
   345→   345→      htmlLines.push(trimmed);
   346→   346→    }
   347→   347→  }
   348→   348→
   349→   349→  let html = htmlLines.join('\n').trim();
   350→   350→  if (!html) return null;
   351→   351→
   352→   352→  if (!/<[a-z][\s\S]*>/i.test(html)) {
   353→   353→    html = html
   354→   354→      .split(/\n{2,}/)
   355→   355→      .map((p) => `<p>${p.replace(/\n/g, '<br/>')}</p>`)
   356→   356→      .join('\n');
   357→   357→  }
   358→   358→
   359→   359→  if (!title) {
   360→   360→    const h = html.match(/<h[1-3][^>]*>([\s\S]*?)<\/h[1-3]>/i);
   361→   361→    title = h ? h[1].replace(/<[^>]+>/g, '').trim() : html.slice(0, 60).replace(/<[^>]+>/g, '');
   362→   362→  }
   363→   363→  if (!excerpt) {
   364→   364→    const firstP = html.match(/<p[^>]*>([\s\S]*?)<\/p>/i);
   365→   365→    excerpt = firstP ? firstP[1].replace(/<[^>]+>/g, '').trim().slice(0, 180) : title;
   366→   366→  }
   367→   367→  if (!metaDescription) {
   368→   368→    metaDescription = excerpt.slice(0, 155);
   369→   369→  }
   370→   370→  if (tags.length === 0) {
   371→   371→    tags = ['سفر', 'تاکسی VIP', 'سیوان'];
   372→   372→  }
   373→   373→
   374→   374→  return { title, excerpt, html, tags, metaDescription };
   375→   375→}

   600→   600→قالب خروجی (دقیقاً همین ترتیب، بدون علامت‌گذاری اضافه مثل markdown یا \`\`\`):
   601→   601→عنوان: <عنوان جذاب مقاله — حاوی کلمه کلیدی>
   602→   602→خلاصه: <خلاصه ۱-۲ جمله‌ای مقاله>

   604→   604→<بدنه HTML مقاله اینجا — فقط تگ‌های h2, h3, p, ul, li, blockquote, strong>
   605→   605→برچسب‌ها: <tag1>, <tag2>, <tag3>, <tag4>
   606→   606→
   607→   607→مقاله را بنویس:`;
   608→   608→
   609→   609→  let raw = '';
   610→   610→  try {
   611→   611→    const completion = await zai.chat.completions.create({
   612→   612→      messages: [
   613→   613→        { role: 'assistant', content: systemPrompt },
   614→   614→        { role: 'user', content: userPrompt },
   615→   615→      ],
   616→   616→      thinking: { type: 'disabled' },
   617→   617→    });
   618→   618→    raw = completion?.choices?.[0]?.message?.content || '';
   619→   619→  } catch (err) {
   620→   620→    console.error('[blog-generator] LLM call failed, retrying once:', err);

   622→   622→      const retry = await zai.chat.completions.create({
   623→   623→        messages: [
   624→   624→          { role: 'assistant', content: systemPrompt },
   625→   625→          { role: 'user', content: userPrompt },
   626→   626→        ],
   627→   627→        thinking: { type: 'disabled' },
   628→   628→      });
   629→   629→      raw = retry?.choices?.[0]?.message?.content || '';
   630→   630→    } catch (err2) {
   631→   631→      console.error('[blog-generator] LLM retry failed:', err2);
   632→   632→      lastError = 'LLM generation failed';
   633→   633→      isGenerating = false;
   634→   634→      return { ok: false, error: 'LLM generation failed' };
   635→   635→    }
   636→   636→  }
   637→   637→
   638→   638→  const parsed = parseArticle(raw);
   639→   639→  if (!parsed || !parsed.title) {
   640→   640→    console.error('[blog-generator] failed to parse article from:\n', raw.slice(0, 500));
   641→   641→    lastError = 'Could not parse article';
   642→   642→    isGenerating = false;
   643→   643→    return { ok: false, error: 'Could not parse article' };
   644→   644→  }
   645→   645→
   646→   646→  // Double-check: compare the generated title against existing posts
   647→   647→  const titleDup = await isDuplicateTopic({ title: parsed.title, keyword: topic.keyword, category: topic.category });
   648→   648→  if (titleDup) {
   649→   649→    console.log(`[blog-generator] generated title "${parsed.title}" is duplicate, discarding`);
   650→   650→    lastError = 'Generated article title is duplicate';
   651→   651→    isGenerating = false;
   652→   652→    return { ok: false, error: 'Generated article is too similar to an existing post' };
   653→   653→  }
   654→   654→
   655→   655→  const slug = makeSlug(parsed.title);
   656→   656→  const featuredImageUrl = await generateCoverImage(parsed.title, slug, topic);
   657→   657→

   659→   659→    const post = await db.blogPost.create({
   660→   660→      data: {
   661→   661→        title: parsed.title,
   662→   662→        slug,
   663→   663→        excerpt: parsed.excerpt,
   664→   664→        content: parsed.html,
   665→   665→        featuredImageUrl,
   666→   666→        status: 'published',
   667→   667→        publishedAt: new Date(),
   668→   668→        tags: JSON.stringify(parsed.tags),
   669→   669→        authorId: null,
   670→   670→        categoryId: null,
   671→   671→      },
   672→   672→    });
   673→   673→    lastGeneratedAt = new Date().toISOString();
   674→   674→    console.log(`[blog-generator] ✓ published${isCustom ? ' (CUSTOM)' : ''}: "${post.title}" (slug: ${post.slug}, image: ${featuredImageUrl || 'none'})`);
   675→   675→
   676→   676→    // Persist state for auto-generations only (not custom ones, so the cycle
   677→   677→    // tracking stays accurate for the scheduled rotation).
   678→   678→    if (!isCustom) {
   679→   679→      await saveState();
   680→   680→      // Schedule the next generation

   683→   683→    }
   684→   684→

   686→   686→    return { ok: true, title: post.title, slug: post.slug, custom: isCustom };

   689→   689→    lastError = 'DB save failed';
   690→   690→    isGenerating = false;
   691→   691→    return { ok: false, error: 'DB save failed' };
   692→   692→  }
   693→   693→}

   695→   695→/**
   696→   696→ * On startup, decide when the next generation should happen.
   697→   697→ *
   698→   698→ * Strategy:
   699→   699→ * 1. Restore persisted state (cycleIndex, categoryTopicIndex, lastAutoGenAt)
   700→   700→ * 2. If fewer than MIN_POSTS_FOR_IMMEDIATE_SKIP published posts exist → generate now (after 5s delay)
   701→   701→ * 3. If lastAutoGenAt is recorded in DB and it's been >= 6h since then → generate now
   702→   702→ * 4. If lastAutoGenAt is recorded but it's been < 6h → schedule the remaining time
   703→   703→ * 5. If no lastAutoGenAt but enough posts exist → start a fresh 6h interval
   704→   704→ */
   705→   705→async function scheduleGenerationOnStartup() {
   706→   706→  try {
   707→   707→    const count = await db.blogPost.count({ where: { status: 'published' } });
   708→   708→    console.log(`[blog-generator] current published posts: ${count}`);
   709→   709→
   710→   710→    if (count < MIN_POSTS_FOR_IMMEDIATE_SKIP) {
   711→   711→      console.log(`[blog-generator] only ${count} published posts; generating one now (5s delay)...`);
   712→   712→      setTimeout(() => generateArticle().catch((e) => console.error(e)), 5000);
   713→   713→      return;
   714→   714→    }
   715→   715→
   716→   716→    // Check persisted lastAutoGenAt from DB (restored into lastGeneratedAt)
   717→   717→    if (lastGeneratedAt) {
   718→   718→      const elapsed = Date.now() - new Date(lastGeneratedAt).getTime();
   719→   719→      console.log(`[blog-generator] last auto-gen was ${Math.round(elapsed / 60000)}min ago`);
   720→   720→      if (elapsed >= GENERATION_INTERVAL_MS) {
   721→   721→        // Overdue — generate now
   722→   722→        console.log('[blog-generator] overdue — generating now (5s delay)...');
   723→   723→        setTimeout(() => generateArticle().catch((e) => console.error(e)), 5000);
   724→   724→        return;
   725→   725→      }
   726→   726→      // Not yet due — wait the remaining time
   727→   727→      const remaining = GENERATION_INTERVAL_MS - elapsed;
   728→   728→      console.log(`[blog-generator] next generation in ${Math.round(remaining / 60000)}min (remaining from last auto-gen)`);
   729→   729→      nextGenAt = new Date(Date.now() + remaining);
   730→   730→      setTimeout(() => {
   731→   731→        generateArticle().catch((e) => console.error('[blog-generator] scheduled error:', e));
   732→   732→        // After this first (delayed) generation, set up the recurring interval
   733→   733→        setInterval(() => {
   734→   734→          generateArticle().catch((e) => console.error('[blog-generator] scheduled error:', e));
   735→   735→        }, GENERATION_INTERVAL_MS);
   736→   736→      }, remaining);
   737→   737→      return;
   738→   738→    }
   739→   739→
   740→   740→    // No lastAutoGenAt recorded but we have enough posts (edge case: posts
   741→   741→    // were seeded or created manually, or DB state was lost). Start fresh.
   742→   742→    console.log(`[blog-generator] no previous auto-gen recorded; starting fresh ${GENERATION_INTERVAL_MS / 3600000}h interval`);
   743→   743→    nextGenAt = new Date(Date.now() + GENERATION_INTERVAL_MS);
   744→   744→    setInterval(() => {
   745→   745→      generateArticle().catch((e) => console.error('[blog-generator] scheduled error:', e));
   746→   746→    }, GENERATION_INTERVAL_MS);
   747→   747→  } catch (err) {
   748→   748→    console.error('[blog-generator] startup scheduling failed:', err);
   749→   749→    // Fallback: just start a fresh interval
   750→   750→    setInterval(() => {
   751→   751→      generateArticle().catch((e) => console.error('[blog-generator] fallback scheduled error:', e));
   752→   752→    }, GENERATION_INTERVAL_MS);
   753→   753→  }
   754→   754→}

   765→   765→  if (!nextGenAt) return GENERATION_INTERVAL_MS;

   835→   835→      return Response.json({ started: true, custom: true, topic }, { headers: corsHeaders });
   836→   836→    }
