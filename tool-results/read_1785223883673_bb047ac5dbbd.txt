   132→   132→// ─── Invalidate Map Size Handler ─────────────────────────────────────────────
   133→   133→
   134→   134→function MapInvalidator() {
   135→   135→  const map = useMap();
   136→   136→
   137→   137→  useEffect(() => {
   138→   138→    // Force Leaflet to recalculate its size after the container is fully rendered
   139→   139→    const timer = setTimeout(() => {
   140→   140→      map.invalidateSize();
   141→   141→    }, 200);
   142→   142→    return () => clearTimeout(timer);
   143→   143→  }, [map]);
   144→   144→
   145→   145→  // Also listen for scroll/resize events that might change the container
   146→   146→  useEffect(() => {
   147→   147→    let frameId: number;
   148→   148→    const handleUpdate = () => {
   149→   149→      if (frameId) cancelAnimationFrame(frameId);
   150→   150→      frameId = requestAnimationFrame(() => {
   151→   151→        map.invalidateSize({ animate: false });
   152→   152→      });
   153→   153→    };
   154→   154→
   155→   155→    window.addEventListener('resize', handleUpdate);
   156→   156→    const observer = new IntersectionObserver((entries) => {
   157→   157→      for (const entry of entries) {
   158→   158→        if (entry.isIntersecting) {
   159→   159→          setTimeout(() => map.invalidateSize({ animate: false }), 100);
   160→   160→        }
   161→   161→      }
   162→   162→    }, { threshold: 0.1 });
   163→   163→
   164→   164→    const container = map.getContainer();
   165→   165→    if (container) observer.observe(container);
   166→   166→
   167→   167→    return () => {
   168→   168→      window.removeEventListener('resize', handleUpdate);
   169→   169→      observer.disconnect();
   170→   170→      if (frameId) cancelAnimationFrame(frameId);
   171→   171→    };
   172→   172→  }, [map]);
   173→   173→
   174→   174→  return null;
   175→   175→}

   179→   179→function FitBoundsHandler({
   180→   180→  points,
   181→   181→  routeData,
   182→   182→}: {
   183→   183→  points: { origin: MapPoint | null; destination: MapPoint | null };
   184→   184→  routeData: RouteResponse | null;
   185→   185→}) {
   186→   186→  const map = useMap();
   187→   187→
   188→   188→  useEffect(() => {
   189→   189→    if (routeData && routeData.routes.length > 0 && routeData.routes[0].path.length > 0) {
   190→   190→      const polyline = L.polyline(routeData.routes[0].path);
   191→   191→      const bounds = polyline.getBounds().pad(0.15);
   192→   192→      map.flyToBounds(bounds, { duration: 1.2 });
   193→   193→    } else if (points.origin && points.destination) {
   194→   194→      const bounds = L.latLngBounds(
   195→   195→        [points.origin.lat, points.origin.lng],
   196→   196→        [points.destination.lat, points.destination.lng]
   197→   197→      ).pad(0.3);
   198→   198→      map.flyToBounds(bounds, { duration: 1 });
   199→   199→    } else if (points.origin) {
   200→   200→      map.flyTo([points.origin.lat, points.origin.lng], 12, { duration: 1 });
   201→   201→    }
   202→   202→  }, [points, routeData, map]);
   203→   203→
   204→   204→  return null;
   205→   205→}

   329→   329→  const [origin, setOrigin] = useState<MapPoint | null>(null);

   336→   336→
   337→   337→  // Search state

   340→   340→  const [originResults, setOriginResults] = useState<
   341→   341→    { display_name: string; lat: string; lon: string }[]
   342→   342→  >([]);
   343→   343→  const [destResults, setDestResults] = useState<
   344→   344→    { display_name: string; lat: string; lon: string }[]
   345→   345→  >([]);

   349→   349→
   350→   350→  // Identify place name using VLM when clicking on map
   351→   351→  const identifyPlace = useCallback(async (lat: number, lng: number, type: 'origin' | 'dest') => {
   352→   352→    setIdentifyingPlace(type);

   355→   355→      if (res.ok) {

   358→   358→        if (name) {

   360→   360→            setOriginSearch(name);
   361→   361→            setOrigin(prev => prev ? { ...prev, name } : null);
   362→   362→          } else {
   363→   363→            setDestSearch(name);
   364→   364→            setDestination(prev => prev ? { ...prev, name } : null);
   365→   365→          }
   366→   366→        }
   367→   367→      }
   368→   368→    } catch {
   369→   369→      // Silently fail - the route API will also try to get place names

   371→   371→      setIdentifyingPlace(null);
   372→   372→    }
   373→   373→  }, []);

   375→   375→  // Handle map click

   381→   381→        setActiveRoute(0);
   382→   382→        setSelectionStep('destination');

   384→   384→      } else if (selectionStep === 'destination') {
   385→   385→        if (origin && Math.abs(origin.lat - lat) < 0.001 && Math.abs(origin.lng - lng) < 0.001) {
   386→   386→          return;
   387→   387→        }
   388→   388→        setDestination({ lat, lng });
   389→   389→        setSelectionStep('ready');
   390→   390→        identifyPlace(lat, lng, 'destination');
   391→   391→      } else {
   392→   392→        clearSelection();
   393→   393→      }
   394→   394→    },
   395→   395→    [selectionStep, origin, identifyPlace]
   396→   396→  );
   397→   397→
   398→   398→  // Search function
   399→   399→  const handleSearch = useCallback(
   400→   400→    (query: string, type: 'origin' | 'dest') => {
   401→   401→      if (searchTimer.current) clearTimeout(searchTimer.current);
   402→   402→
   403→   403→      if (!query || query.length < 2) {
   404→   404→        if (type === 'origin') setOriginResults([]);
   405→   405→        else setDestResults([]);

   407→   407→      }
   408→   408→
   409→   409→      setSearchLoading(type);
   410→   410→
   411→   411→      searchTimer.current = setTimeout(async () => {

   414→   414→          const url = `https://nominatim.openstreetmap.org/search?q=${encoded}&format=json&limit=5&accept-language=fa&addressdetails=1`;
   415→   415→

   417→   417→            headers: {
   418→   418→              'User-Agent': 'SivanVIPTaxi/1.0',
   419→   419→            },
   420→   420→          });
   421→   421→
   422→   422→          if (!res.ok) return;
   423→   423→
   424→   424→          const data = await res.json();
   425→   425→          if (type === 'origin') {
   426→   426→            setOriginResults(data || []);
   427→   427→          } else {
   428→   428→            setDestResults(data || []);
   429→   429→          }
   430→   430→        } catch {
   431→   431→          // ignore search errors
   432→   432→        } finally {
   433→   433→          setSearchLoading(null);
   434→   434→        }
   435→   435→      }, 800);
   436→   436→    },
   437→   437→    []
   438→   438→  );
   439→   439→
   440→   440→  // Handle search select
   441→   441→  const handleSearchSelect = useCallback(
   442→   442→    (lat: number, lng: number, name: string, type: 'origin' | 'dest') => {
   443→   443→      if (type === 'origin') {
   444→   444→        setOrigin({ lat, lng, name });
   445→   445→        setOriginSearch(name);
   446→   446→        setOriginResults([]);
   447→   447→        if (!destination) {
   448→   448→          setSelectionStep('destination');
   449→   449→        } else {
   450→   450→          setSelectionStep('ready');
   451→   451→        }
   452→   452→      } else {
   453→   453→        setDestination({ lat, lng, name });
   454→   454→        setDestSearch(name);
   455→   455→        setDestResults([]);
   456→   456→        if (origin) {
   457→   457→          setSelectionStep('ready');
   458→   458→        } else {
   459→   459→          setSelectionStep('origin');
   460→   460→        }
   461→   461→      }
   462→   462→    },
   463→   463→    [origin, destination]
   464→   464→  );
   465→   465→
   466→   466→  // Fetch route when both points selected
   467→   467→  useEffect(() => {
   468→   468→    if (origin && destination) {
   469→   469→      setRouteLoading(true);
   470→   470→      setRouteError('');
   471→   471→
   472→   472→      const timer = setTimeout(async () => {
   473→   473→        try {
   474→   474→          const res = await fetch(
   475→   475→            `/api/map/route?originLat=${origin.lat}&originLng=${origin.lng}&destLat=${destination.lat}&destLng=${destination.lng}`
   476→   476→          );
   477→   477→
   478→   478→          if (!res.ok) {
   479→   479→            const err = await res.json().catch(() => ({ error: 'خطا در محاسبه مسیر' }));
   480→   480→            throw new Error(err.error || 'خطا');
   481→   481→          }
   482→   482→
   483→   483→          const data = await res.json();
