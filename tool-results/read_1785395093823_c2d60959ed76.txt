   136→// ─── Invalidate Map Size Handler ─────────────────────────────────────────────

   138→function MapInvalidator() {
   139→  const map = useMap();
   140→
   141→  useEffect(() => {
   142→    // Force Leaflet to recalculate its size after the container is fully rendered
   143→    const timer = setTimeout(() => {
   144→      map.invalidateSize();
   145→    }, 200);
   146→    return () => clearTimeout(timer);
   147→  }, [map]);
   148→
   149→  // Also listen for scroll/resize events that might change the container
   150→  useEffect(() => {
   151→    let frameId: number;
   152→    const handleUpdate = () => {
   153→      if (frameId) cancelAnimationFrame(frameId);
   154→      frameId = requestAnimationFrame(() => {
   155→        map.invalidateSize({ animate: false });
   156→      });
   157→    };
   158→
   159→    window.addEventListener('resize', handleUpdate);
   160→    const observer = new IntersectionObserver((entries) => {
   161→      for (const entry of entries) {
   162→        if (entry.isIntersecting) {
   163→          setTimeout(() => map.invalidateSize({ animate: false }), 100);
   164→        }
   165→      }
   166→    }, { threshold: 0.1 });
   167→
   168→    const container = map.getContainer();
   169→    if (container) observer.observe(container);
   170→
   171→    return () => {
   172→      window.removeEventListener('resize', handleUpdate);
   173→      observer.disconnect();
   174→      if (frameId) cancelAnimationFrame(frameId);
   175→    };
   176→  }, [map]);
   177→
   178→  return null;
   179→}

   181→// ─── Fit to Bounds Handler ──────────────────────────────────────────────────
   182→
   183→function FitBoundsHandler({
   184→  points,
   185→  routeData,
   186→}: {
   187→  points: { origin: MapPoint | null; destination: MapPoint | null };
   188→  routeData: RouteResponse | null;
   189→}) {
   190→  const map = useMap();
   191→
   192→  useEffect(() => {
   193→    if (routeData && routeData.routes.length > 0 && routeData.routes[0].path.length > 0) {
   194→      const polyline = L.polyline(routeData.routes[0].path);
   195→      const bounds = polyline.getBounds().pad(0.15);
   196→      map.flyToBounds(bounds, { duration: 1.2 });
   197→    } else if (points.origin && points.destination) {
   198→      const bounds = L.latLngBounds(
   199→        [points.origin.lat, points.origin.lng],
   200→        [points.destination.lat, points.destination.lng]
   201→      ).pad(0.3);
   202→      map.flyToBounds(bounds, { duration: 1 });
   203→    } else if (points.origin) {
   204→      map.flyTo([points.origin.lat, points.origin.lng], 12, { duration: 1 });
   205→    }
   206→  }, [points, routeData, map]);
   207→
   208→  return null;
   209→}

   427→export default function InteractiveMapInner() {
   428→  const [origin, setOrigin] = useState<MapPoint | null>(null);
   429→  const [destination, setDestination] = useState<MapPoint | null>(null);
   430→  const [selectionStep, setSelectionStep] = useState<SelectionStep>('origin');
   431→  const [routeData, setRouteData] = useState<RouteResponse | null>(null);
   432→  const [routeLoading, setRouteLoading] = useState(false);
   433→  const [routeError, setRouteError] = useState('');
   434→  const [activeRoute, setActiveRoute] = useState(0);
   435→
   436→  // Search state
   437→  const [originSearch, setOriginSearch] = useState('');
   438→  const [destSearch, setDestSearch] = useState('');
   439→  const [originResults, setOriginResults] = useState<
   440→    { display_name: string; lat: string; lon: string }[]
   441→  >([]);
   442→  const [destResults, setDestResults] = useState<
   443→    { display_name: string; lat: string; lon: string }[]
   444→  >([]);
   445→  const [searchLoading, setSearchLoading] = useState<'origin' | 'dest' | null>(null);
   446→  const searchTimer = useRef<ReturnType<typeof setTimeout>>();
   447→  const [identifyingPlace, setIdentifyingPlace] = useState<'origin' | 'dest' | null>(null);
   448→
   449→  // Identify place name using VLM when clicking on map
   450→  const identifyPlace = useCallback(async (lat: number, lng: number, type: 'origin' | 'dest') => {
   451→    setIdentifyingPlace(type);
   452→    try {
   453→      const res = await fetch(`/api/map/identify-place?lat=${lat}&lng=${lng}`);
   454→      if (res.ok) {
   455→        const data = await res.json();
   456→        const name = data.name || '';
   457→        if (name) {
   458→          if (type === 'origin') {
   459→            setOriginSearch(name);
   460→            setOrigin(prev => prev ? { ...prev, name } : null);
   461→          } else {
   462→            setDestSearch(name);
   463→            setDestination(prev => prev ? { ...prev, name } : null);
   464→          }
   465→        }
   466→      }
   467→    } catch {
   468→      // Silently fail - the route API will also try to get place names
   469→    } finally {
   470→      setIdentifyingPlace(null);
   471→    }
   472→  }, []);
   473→
   474→  // Handle map click
   475→  const handleSelectPoint = useCallback(
   476→    (lat: number, lng: number) => {
   477→      if (selectionStep === 'origin') {
   478→        setOrigin({ lat, lng });
   479→        setRouteData(null);
   480→        setActiveRoute(0);
   481→        setSelectionStep('destination');
   482→        identifyPlace(lat, lng, 'origin');
   483→      } else if (selectionStep === 'destination') {
   484→        if (origin && Math.abs(origin.lat - lat) < 0.001 && Math.abs(origin.lng - lng) < 0.001) {
   485→          return;
   486→        }
   487→        setDestination({ lat, lng });
   488→        setSelectionStep('ready');
   489→        identifyPlace(lat, lng, 'destination');
   490→      } else {
   491→        clearSelection();
   492→      }
   493→    },
   494→    [selectionStep, origin, identifyPlace]
   495→  );
   496→
   497→  // Search function
   498→  const handleSearch = useCallback(
   499→    (query: string, type: 'origin' | 'dest') => {
   500→      if (searchTimer.current) clearTimeout(searchTimer.current);
   501→
   502→      if (!query || query.length < 2) {
   503→        if (type === 'origin') setOriginResults([]);
   504→        else setDestResults([]);
   505→        return;
   506→      }
   507→
   508→      setSearchLoading(type);
   509→
   510→      searchTimer.current = setTimeout(async () => {
   511→        try {
   512→          const encoded = encodeURIComponent(query + ', ایران');
   513→          const url = `https://nominatim.openstreetmap.org/search?q=${encoded}&format=json&limit=5&accept-language=fa&addressdetails=1`;
   514→
   515→          const res = await fetch(url, {
   516→            headers: {
   517→              'User-Agent': 'SivanVIPTaxi/1.0',
   518→            },
   519→          });
   520→
   521→          if (!res.ok) return;
   522→
   523→          const data = await res.json();
   524→          if (type === 'origin') {
   525→            setOriginResults(data || []);
   526→          } else {
   527→            setDestResults(data || []);
   528→          }
   529→        } catch {
   530→          // ignore search errors
   531→        } finally {
   532→          setSearchLoading(null);
   533→        }
   534→      }, 800);
   535→    },
   536→    []
   537→  );
   538→
   539→  // Handle search select
   540→  const handleSearchSelect = useCallback(
   541→    (lat: number, lng: number, name: string, type: 'origin' | 'dest') => {
   542→      if (type === 'origin') {
   543→        setOrigin({ lat, lng, name });
   544→        setOriginSearch(name);
   545→        setOriginResults([]);
   546→        if (!destination) {
   547→          setSelectionStep('destination');
   548→        } else {
   549→          setSelectionStep('ready');
   550→        }
   551→      } else {
   552→        setDestination({ lat, lng, name });
   553→        setDestSearch(name);
   554→        setDestResults([]);
   555→        if (origin) {
   556→          setSelectionStep('ready');
   557→        } else {
   558→          setSelectionStep('origin');
   559→        }
   560→      }
   561→    },
   562→    [origin, destination]
   563→  );
   564→
   565→  // Fetch route when both points selected
   566→  useEffect(() => {
   567→    if (origin && destination) {
   568→      setRouteLoading(true);
   569→      setRouteError('');
   570→
   571→      const timer = setTimeout(async () => {
   572→        try {
   573→          const res = await fetch(
   574→            `/api/map/route?originLat=${origin.lat}&originLng=${origin.lng}&destLat=${destination.lat}&destLng=${destination.lng}`
   575→          );
   576→
   577→          if (!res.ok) {
   578→            const err = await res.json().catch(() => ({ error: 'خطا در محاسبه مسیر' }));
   579→            throw new Error(err.error || 'خطا');
