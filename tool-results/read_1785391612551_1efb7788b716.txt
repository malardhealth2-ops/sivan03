   136→   136→// ─── Invalidate Map Size Handler ─────────────────────────────────────────────
   137→   137→
   138→   138→function MapInvalidator() {
   139→   139→  const map = useMap();
   140→   140→
   141→   141→  useEffect(() => {
   142→   142→    // Force Leaflet to recalculate its size after the container is fully rendered
   143→   143→    const timer = setTimeout(() => {
   144→   144→      map.invalidateSize();
   145→   145→    }, 200);
   146→   146→    return () => clearTimeout(timer);
   147→   147→  }, [map]);
   148→   148→
   149→   149→  // Also listen for scroll/resize events that might change the container
   150→   150→  useEffect(() => {
   151→   151→    let frameId: number;
   152→   152→    const handleUpdate = () => {
   153→   153→      if (frameId) cancelAnimationFrame(frameId);
   154→   154→      frameId = requestAnimationFrame(() => {
   155→   155→        map.invalidateSize({ animate: false });
   156→   156→      });
   157→   157→    };
   158→   158→
   159→   159→    window.addEventListener('resize', handleUpdate);
   160→   160→    const observer = new IntersectionObserver((entries) => {
   161→   161→      for (const entry of entries) {
   162→   162→        if (entry.isIntersecting) {
   163→   163→          setTimeout(() => map.invalidateSize({ animate: false }), 100);
   164→   164→        }
   165→   165→      }
   166→   166→    }, { threshold: 0.1 });
   167→   167→
   168→   168→    const container = map.getContainer();
   169→   169→    if (container) observer.observe(container);
   170→   170→
   171→   171→    return () => {
   172→   172→      window.removeEventListener('resize', handleUpdate);
   173→   173→      observer.disconnect();
   174→   174→      if (frameId) cancelAnimationFrame(frameId);
   175→   175→    };
   176→   176→  }, [map]);
   177→   177→
   178→   178→  return null;
   179→   179→}

   183→   183→function FitBoundsHandler({
   184→   184→  points,
   185→   185→  routeData,
   186→   186→}: {
   187→   187→  points: { origin: MapPoint | null; destination: MapPoint | null };
   188→   188→  routeData: RouteResponse | null;
   189→   189→}) {
   190→   190→  const map = useMap();
   191→   191→
   192→   192→  useEffect(() => {
   193→   193→    if (routeData && routeData.routes.length > 0 && routeData.routes[0].path.length > 0) {
   194→   194→      const polyline = L.polyline(routeData.routes[0].path);
   195→   195→      const bounds = polyline.getBounds().pad(0.15);
   196→   196→      map.flyToBounds(bounds, { duration: 1.2 });
   197→   197→    } else if (points.origin && points.destination) {
   198→   198→      const bounds = L.latLngBounds(
   199→   199→        [points.origin.lat, points.origin.lng],
   200→   200→        [points.destination.lat, points.destination.lng]
   201→   201→      ).pad(0.3);
   202→   202→      map.flyToBounds(bounds, { duration: 1 });
   203→   203→    } else if (points.origin) {
   204→   204→      map.flyTo([points.origin.lat, points.origin.lng], 12, { duration: 1 });
   205→   205→    }
   206→   206→  }, [points, routeData, map]);
   207→   207→
   208→   208→  return null;
   209→   209→}
