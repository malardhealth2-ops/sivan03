    77→function getCityDisplayName(city: { province: string; city: string; district?: string; neighborhood?: string }): string {
    78→  if (city.neighborhood) {
    79→    return `${city.neighborhood}، ${city.district}، ${city.province}`;
    80→  }
    81→  if (city.district) {
    82→    return `${city.district}، ${city.province}`;
    83→  }
    84→  if (city.city) {
    85→    return `${city.city}، ${city.province}`;
    86→  }
    87→  return city.province;
    88→}

    94→export function BookingModal() {
    95→  const {
    96→    booking,
    97→    setBookingStep,
    98→    updateBookingForm,
    99→    setEstimatedPrice,
   100→    setBookingCode,

   103→  } = useAppStore();
   104→
   105→  const [open, setOpen] = useState(false);
   106→  const [distanceLoading, setDistanceLoading] = useState(false);
   107→  const [distanceError, setDistanceError] = useState('');
   108→

   121→      setBookingStep(booking.currentStep + 1);
   122→    }
   123→  };

   127→      setBookingStep(booking.currentStep - 1);
   128→    }
   129→  };

   131→  // Fetch distance when both cities are selected
   132→  const fetchDistance = useCallback(async () => {
   133→    const { originCity, destCity } = booking.formData;
   134→    if (!originCity.city || !destCity.city || originCity.city === destCity.city) {
   135→      updateBookingForm({ distanceKm: null, durationMin: null });
   136→      setDistanceError('');
   137→      return;

   140→    setDistanceLoading(true);
   141→    setDistanceError('');

   144→      const originQuery = originCity.neighborhood || originCity.district || originCity.city;

   146→
   147→      const res = await fetch(
   148→        `/api/distance?origin=${encodeURIComponent(originQuery)}&destination=${encodeURIComponent(destQuery)}`
   149→      );
   150→
   151→      if (!res.ok) {
   152→        const err = await res.json();
   153→        throw new Error(err.error || 'خطا در محاسبه فاصله');
   154→      }

   156→      const data = await res.json();

   158→        distanceKm: data.distanceKm,
   159→        durationMin: data.durationMin,
   160→      });

   162→      setDistanceError(err instanceof Error ? err.message : 'خطا در محاسبه فاصله');

   165→      setDistanceLoading(false);

   171→    if (originCity.city && destCity.city && originCity.city !== destCity.city) {
   172→      const timer = setTimeout(fetchDistance, 500);
   173→      return () => clearTimeout(timer);
   174→    }
   175→  }, [booking.formData.originCity.city, booking.formData.destCity.city, fetchDistance]);
   176→
   177→  const handleSubmit = async () => {
   178→    setBookingSubmitting(true);
   179→    try {
   180→      const { originCity, destCity, distanceKm, tripType, roundTrip } = booking.formData;
   181→      const originDisplay = getCityDisplayName(originCity);
   182→      const destDisplay = getCityDisplayName(destCity);
   183→
   184→      // Fetch price
   185→      let price = 500000; // fallback base
   186→      let duration = '';
   187→      if (distanceKm) {
   188→        try {
   189→          const res = await fetch(
   190→            `/api/pricing?tripType=${tripType}&distanceKm=${distanceKm}`
   191→          );
   192→          if (res.ok) {
   193→            const data = await res.json();
   194→            price = roundTrip ? data.price * 2 : data.price;
   195→            duration = data.duration;
   196→          }
   197→        } catch {
   198→          // use fallback
   199→        }
   200→      }
   201→
   202→      // Call booking API
   203→      const bookingRes = await fetch('/api/booking', {
   204→        method: 'POST',
   205→        headers: { 'Content-Type': 'application/json' },
   206→        body: JSON.stringify({
   207→          originAddress: originDisplay,
   208→          destAddress: destDisplay,
   209→          distanceKm: distanceKm || 0,
   210→          tripType,
   211→          roundTrip,
   212→          passengerCount: booking.formData.passengerCount,
   213→          date: booking.formData.date,
   214→          time: booking.formData.time,
   215→          fullName: booking.formData.fullName,
   216→          phone: booking.formData.phone,
   217→          notes: booking.formData.notes,
   218→          paymentMethod: booking.formData.paymentMethod,
   219→          totalAmount: price,
   220→        }),
   221→      });

   223→      if (!bookingRes.ok) {
   224→        const errData = await bookingRes.json().catch(() => ({ error: 'خطا در ثبت رزرو' }));
   225→        throw new Error(errData.error || 'خطا در ثبت رزرو');

   231→      setEstimatedPrice(price, duration || `~${booking.formData.durationMin || 60} دقیقه`);

   254→          duration: 3000,
   255→          className: 'toast-success-custom',
   256→        }

   259→      toast.error('خطا در ثبت رزرو. لطفاً دوباره تلاش کنید.', { duration: 3000 });

   261→      setBookingSubmitting(false);
   262→    }
   263→  };

   271→      case 2:

   289→        {/* Steps indicator */}

   307→                        ? 'bg-[#D4AF37]/20 text-[#D4AF37]'
   308→                        : 'bg-[#2d2d2d] text-[#a1a1aa]'

   311→                  {idx < booking.currentStep ? (
   312→                    <Check className="h-3.5 w-3.5" />

   314→                    <step.icon className="h-3.5 w-3.5" />
   315→                  )}
   316→                  <span className="hidden sm:inline">{step.title}</span>
   317→                </div>
   318→                {idx < steps.length - 1 && (
   319→                  <div
   320→                    className={`w-6 h-0.5 rounded-full ${
   321→                      idx < booking.currentStep ? 'bg-[#D4AF37]' : 'bg-[#333]'
   322→                    }`}
   323→                  />
   324→                )}
   325→              </div>
   326→            ))}
   327→          </div>
   328→        </div>
   329→
   330→        <div className="p-6">
   331→          <AnimatePresence mode="wait">
   332→            {/* Step 0: Route */}
   333→            {booking.currentStep === 0 && (
   334→              <motion.div
   335→                key="step-0"
   336→                initial={{ opacity: 0, x: -20 }}
   337→                animate={{ opacity: 1, x: 0 }}
   338→                exit={{ opacity: 0, x: 20 }}
   339→                transition={{ duration: 0.3 }}
   340→                className="space-y-5"
   341→              >
   342→                <CitySelector

   345→                  value={booking.formData.originCity}

   347→                    updateBookingForm({
   348→                      originCity: val,
   349→                      origin: val.neighborhood || val.district || val.city,
   350→                    });
   351→                  }}
   352→                  placeholder="استان و شهر مبدأ را انتخاب کنید"
   353→                />
   354→
   355→                <CitySelector
   356→                  label="شهر مقصد"
   357→                  iconColor="#E5C76B"
   358→                  value={booking.formData.destCity}
   359→                  onChange={(val) => {
   360→                    updateBookingForm({
   361→                      destCity: val,
   362→                      destination: val.neighborhood || val.district || val.city,

   365→                  placeholder="استان و شهر مقصد را انتخاب کنید"

   368→                {/* Distance info */}
   369→                {(distanceLoading || booking.formData.distanceKm || distanceError) && (
   370→                  <div className="p-4 bg-[#0a0a0a] rounded-xl border border-[#333]">

   372→                      <div className="flex items-center gap-3 text-[#D4AF37]">
   373→                        <Loader2 className="h-5 w-5 animate-spin" />

   375→                      </div>

   377→                    {booking.formData.distanceKm && !distanceLoading && (
   378→                      <div className="space-y-2">
   379→                        <div className="flex items-center gap-2 text-[#fafafa]">
   380→                          <Route className="h-5 w-5 text-[#D4AF37]" />
   381→                          <span className="font-bold">
