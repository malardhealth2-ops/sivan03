   136→// ─── Invalidate Map Size Handler ─────────────────────────────────────────────

   138→function MapInvalidator() {
   139→  const map = useMap();
   140→
   141→  useEffect(() => {
   142→    // Force Leaflet to recalculate its size after the container is fully rendered
   143→    const timer = setTimeout(() => {
   144→      map.invalidateSize();
   145→    }, 200);
   146→    return () => clearTimeout(timer);
   147→  }, [map]);
   148→
   149→  // Also listen for scroll/resize events that might change the container
   150→  useEffect(() => {
   151→    let frameId: number;
   152→    const handleUpdate = () => {
   153→      if (frameId) cancelAnimationFrame(frameId);
   154→      frameId = requestAnimationFrame(() => {
   155→        map.invalidateSize({ animate: false });
   156→      });
   157→    };
   158→
   159→    window.addEventListener('resize', handleUpdate);
   160→    const observer = new IntersectionObserver((entries) => {
   161→      for (const entry of entries) {
   162→        if (entry.isIntersecting) {
   163→          setTimeout(() => map.invalidateSize({ animate: false }), 100);
   164→        }
   165→      }
   166→    }, { threshold: 0.1 });
   167→
   168→    const container = map.getContainer();
   169→    if (container) observer.observe(container);
   170→
   171→    return () => {
   172→      window.removeEventListener('resize', handleUpdate);
   173→      observer.disconnect();
   174→      if (frameId) cancelAnimationFrame(frameId);
   175→    };
   176→  }, [map]);
   177→
   178→  return null;
   179→}

   181→// ─── Fit to Bounds Handler ──────────────────────────────────────────────────
   182→
   183→function FitBoundsHandler({
   184→  points,
   185→  routeData,
   186→}: {
   187→  points: { origin: MapPoint | null; destination: MapPoint | null };
   188→  routeData: RouteResponse | null;
   189→}) {
   190→  const map = useMap();
   191→
   192→  useEffect(() => {
   193→    if (routeData && routeData.routes.length > 0 && routeData.routes[0].path.length > 0) {
   194→      const polyline = L.polyline(routeData.routes[0].path);
   195→      const bounds = polyline.getBounds().pad(0.15);
   196→      map.flyToBounds(bounds, { duration: 1.2 });
   197→    } else if (points.origin && points.destination) {
   198→      const bounds = L.latLngBounds(
   199→        [points.origin.lat, points.origin.lng],
   200→        [points.destination.lat, points.destination.lng]
   201→      ).pad(0.3);
   202→      map.flyToBounds(bounds, { duration: 1 });
   203→    } else if (points.origin) {
   204→      map.flyTo([points.origin.lat, points.origin.lng], 12, { duration: 1 });
   205→    }
   206→  }, [points, routeData, map]);
   207→
   208→  return null;
   209→}

   431→export default function InteractiveMapInner() {
   432→  const [origin, setOrigin] = useState<MapPoint | null>(null);
   433→  const [destination, setDestination] = useState<MapPoint | null>(null);
   434→  const [selectionStep, setSelectionStep] = useState<SelectionStep>('origin');
   435→  const [routeData, setRouteData] = useState<RouteResponse | null>(null);
   436→  const [routeLoading, setRouteLoading] = useState(false);
   437→  const [routeError, setRouteError] = useState('');
   438→  const [activeRoute, setActiveRoute] = useState(0);
   439→
   440→  // Subscribe to hero route selection from the quick booking form
   441→  const heroRouteSelection = useAppStore((s) => s.heroRouteSelection);
   442→  const lastHeroTimestamp = useRef<number>(0);
   443→
   444→  // Search state
   445→  const [originSearch, setOriginSearch] = useState('');
   446→  const [destSearch, setDestSearch] = useState('');
   447→  const [originResults, setOriginResults] = useState<
   448→    { display_name: string; lat: string; lon: string }[]
   449→  >([]);
   450→  const [destResults, setDestResults] = useState<
   451→    { display_name: string; lat: string; lon: string }[]
   452→  >([]);
   453→  const [searchLoading, setSearchLoading] = useState<'origin' | 'dest' | null>(null);
   454→  const searchTimer = useRef<ReturnType<typeof setTimeout>>();
   455→  const [identifyingPlace, setIdentifyingPlace] = useState<'origin' | 'dest' | null>(null);
   456→
   457→  // Auto-populate map when hero form fills in origin & destination
   458→  useEffect(() => {
   459→    if (
   460→      heroRouteSelection.timestamp > 0 &&
   461→      heroRouteSelection.timestamp !== lastHeroTimestamp.current &&
   462→      heroRouteSelection.origin &&
   463→      heroRouteSelection.destination
   464→    ) {
   465→      lastHeroTimestamp.current = heroRouteSelection.timestamp;
   466→      const { origin: hOrigin, destination: hDest } = heroRouteSelection;
   467→
   468→      // Set origin and destination on the map
   469→      setOrigin({ lat: hOrigin.lat, lng: hOrigin.lng, name: hOrigin.name });
   470→      setOriginSearch(hOrigin.name);
   471→      setDestination({ lat: hDest.lat, lng: hDest.lng, name: hDest.name });
   472→      setDestSearch(hDest.name);
   473→      setSelectionStep('ready');
   474→      setRouteData(null);
   475→      handleSetActiveRoute(0);
   476→    }
   477→  }, [heroRouteSelection, handleSetActiveRoute]);
   478→
   479→  // Wrapper to sync active route with store (so hero pricing stays in sync)
   480→  const handleSetActiveRoute = useCallback((index: number) => {
   481→    setActiveRoute(index);
   482→    useAppStore.getState().setHeroRouteSelection({ activeRouteIndex: index });
   483→  }, []);
   484→
   485→  // Identify place name using VLM when clicking on map
   486→  const identifyPlace = useCallback(async (lat: number, lng: number, type: 'origin' | 'dest') => {
   487→    setIdentifyingPlace(type);
   488→    try {
   489→      const res = await fetch(`/api/map/identify-place?lat=${lat}&lng=${lng}`);
   490→      if (res.ok) {
   491→        const data = await res.json();
   492→        const name = data.name || '';
   493→        if (name) {
   494→          if (type === 'origin') {
   495→            setOriginSearch(name);
   496→            setOrigin(prev => prev ? { ...prev, name } : null);
   497→          } else {
   498→            setDestSearch(name);
   499→            setDestination(prev => prev ? { ...prev, name } : null);
   500→          }
   501→        }
   502→      }
   503→    } catch {
   504→      // Silently fail - the route API will also try to get place names
   505→    } finally {
   506→      setIdentifyingPlace(null);
   507→    }
   508→  }, []);
   509→
   510→  // Handle map click
   511→  const handleSelectPoint = useCallback(
   512→    (lat: number, lng: number) => {
   513→      if (selectionStep === 'origin') {
   514→        setOrigin({ lat, lng });
   515→        setRouteData(null);
   516→        setActiveRoute(0);
   517→        setSelectionStep('destination');
   518→        identifyPlace(lat, lng, 'origin');
   519→      } else if (selectionStep === 'destination') {
   520→        if (origin && Math.abs(origin.lat - lat) < 0.001 && Math.abs(origin.lng - lng) < 0.001) {
   521→          return;
   522→        }
   523→        setDestination({ lat, lng });
   524→        setSelectionStep('ready');
   525→        identifyPlace(lat, lng, 'dest');
   526→      } else {
   527→        clearSelection();
   528→      }
   529→    },
   530→    [selectionStep, origin, identifyPlace]
   531→  );
   532→
   533→  // Search function
   534→  const handleSearch = useCallback(
   535→    (query: string, type: 'origin' | 'dest') => {
   536→      if (searchTimer.current) clearTimeout(searchTimer.current);
   537→
   538→      if (!query || query.length < 2) {
   539→        if (type === 'origin') setOriginResults([]);
   540→        else setDestResults([]);
   541→        return;
   542→      }
   543→
   544→      setSearchLoading(type);
   545→
   546→      searchTimer.current = setTimeout(async () => {
   547→        try {
   548→          const encoded = encodeURIComponent(query + ', ایران');
   549→          const url = `https://nominatim.openstreetmap.org/search?q=${encoded}&format=json&limit=5&accept-language=fa&addressdetails=1`;
   550→
   551→          const res = await fetch(url, {
   552→            headers: {
   553→              'User-Agent': 'SivanVIPTaxi/1.0',
   554→            },
   555→          });
   556→
   557→          if (!res.ok) return;
   558→
   559→          const data = await res.json();
   560→          if (type === 'origin') {
   561→            setOriginResults(data || []);
   562→          } else {
   563→            setDestResults(data || []);
   564→          }
   565→        } catch {
   566→          // ignore search errors
   567→        } finally {
   568→          setSearchLoading(null);
   569→        }
   570→      }, 800);
   571→    },
   572→    []
   573→  );
