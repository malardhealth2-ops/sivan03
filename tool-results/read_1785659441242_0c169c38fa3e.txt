   177→      lastGeneratedAt = state.lastAutoGenAt.toISOString();
   178→    }

   181→    console.error('[blog-generator] failed to restore state from DB, starting fresh:', err);
   182→  }

   209→/**
   210→ * Check if a topic is too similar to any existing published post.
   211→ * Compares the topic's keyword words against post titles and tags.
   212→ * Returns true if a duplicate is detected (>50% word overlap with an existing post).
   213→ */
   214→async function isDuplicateTopic(topic: Topic): Promise<boolean> {
   215→  try {
   216→    // Get all published post titles and tags
   217→    const existingPosts = await db.blogPost.findMany({
   218→      where: { status: 'published' },
   219→      select: { title: true, tags: true },
   220→    });
   221→    if (existingPosts.length === 0) return false;
   222→
   223→    // Normalize the topic keyword into a set of meaningful words
   224→    const topicWords = new Set(
   225→      topic.keyword
   226→        .replace(/[«»"'!?.,:;()\[\]{}،؛؟]/g, ' ')
   227→        .split(/\s+/)
   228→        .filter((w) => w.length > 2)
   229→        .map((w) => w.trim())
   230→    );
   231→    // Also add key words from the title
   232→    const titleWords = new Set(
   233→      topic.title
   234→        .replace(/[«»"'!?.,:;()\[\]{}،؛؟]/g, ' ')
   235→        .split(/\s+/)
   236→        .filter((w) => w.length > 2)
   237→        .map((w) => w.trim())
   238→    );
   239→    // Merge both sets for comparison
   240→    const allTopicWords = new Set([...topicWords, ...titleWords]);
   241→    if (allTopicWords.size === 0) return false;
   242→
   243→    for (const post of existingPosts) {
   244→      // Normalize post title and tags into a set of words
   245→      const postText = `${post.title} ${post.tags || ''}`;
   246→      const postWords = new Set(
   247→        postText
   248→          .replace(/[«»"'!?.,:;()\[\]{}،؛؟\[\]]/g, ' ')
   249→          .split(/\s+/)
   250→          .filter((w) => w.length > 2)
   251→          .map((w) => w.trim())
   252→      );
   253→
   254→      // Count overlapping words
   255→      let overlap = 0;
   256→      for (const w of allTopicWords) {
   257→        if (postWords.has(w)) overlap++;
   258→      }
   259→      const similarity = overlap / allTopicWords.size;
   260→      // If more than 50% of the topic's key words match, consider it duplicate
   261→      if (similarity >= 0.5 && overlap >= 2) {
   262→        console.log(`[blog-generator] duplicate detected: topic "${topic.title}" overlaps ${Math.round(similarity * 100)}% with existing post "${post.title}" (shared words: ${overlap}/${allTopicWords.size})`);
   263→        return true;
   264→      }
   265→    }
   266→    return false;
   267→  } catch (err) {
   268→    console.error('[blog-generator] duplicate check failed:', err);
   269→    return false; // on error, allow generation
   270→  }
   271→}

   273→/**
   274→ * Pick a non-duplicate topic. Tries up to MAX_PICK_ATTEMPTS times to find
   275→ * a topic that doesn't overlap with existing posts. If all attempts fail,
   276→ * returns the last picked topic anyway (so generation doesn't block forever).
   277→ */
   278→const MAX_PICK_ATTEMPTS = 20;
   279→
   280→async function pickNonDuplicateTopic(): Promise<Topic> {
   281→  for (let attempt = 0; attempt < MAX_PICK_ATTEMPTS; attempt++) {
   282→    const topic = pickTopic();
   283→    const dup = await isDuplicateTopic(topic);
   284→    if (!dup) return topic;
   285→    console.log(`[blog-generator] attempt ${attempt + 1}: topic "${topic.title}" is duplicate, trying next...`);
   286→  }
   287→  // All attempts exhausted — return last picked topic to avoid infinite loop
   288→  console.log('[blog-generator] all pick attempts exhausted, using last topic');
   289→  return pickTopic();
   290→}
   291→
   292→function pickTopic(): Topic {
   293→  const category = CATEGORY_ORDER[cycleIndex % CATEGORY_ORDER.length];
   294→  cycleIndex++;
   295→  const pool = TOPICS.filter((t) => t.category === category);
   296→  const idx = categoryTopicIndex[category] % pool.length;
   297→  categoryTopicIndex[category]++;
   298→  return pool[idx];
   299→}

   301→function makeSlug(title: string): string {
   302→  const base = title
   303→    .replace(/[«»"'!?.,:;()\[\]{}]/g, '')
   304→    .replace(/\s+/g, '-')
   305→    .replace(/-+/g, '-')
   306→    .trim()
   307→    .slice(0, 60);
   308→  const suffix = Date.now().toString(36).slice(-6);
   309→  return `${base}-${suffix}`;
   310→}

   312→function parseArticle(raw: string): {
   313→  title: string;
   314→  excerpt: string;
   315→  html: string;
   316→  tags: string[];
   317→  metaDescription: string;
   318→} | null {
   319→  if (!raw) return null;
   320→  const lines = raw.split('\n');
   321→
   322→  let title = '';
   323→  let excerpt = '';
   324→  let metaDescription = '';
   325→  let tags: string[] = [];
   326→  const htmlLines: string[] = [];
   327→
   328→  for (const line of lines) {
   329→    const trimmed = line.trim();
   330→    if (!trimmed) continue;
   331→    if (trimmed.startsWith('عنوان:')) {
   332→      title = trimmed.slice(6).trim().replace(/^["'«]|["'»]$/g, '');
   333→    } else if (trimmed.startsWith('خلاصه:')) {
   334→      excerpt = trimmed.slice(6).trim().replace(/^["'«]|["'»]$/g, '');
   335→    } else if (trimmed.startsWith('متا:')) {
   336→      metaDescription = trimmed.slice(4).trim().replace(/^["'«]|["'»]$/g, '');
   337→    } else if (trimmed.startsWith('برچسب‌ها:')) {
   338→      const rest = trimmed.replace(/^برچسب‌ها:\s*/, '');
   339→      tags = rest
   340→        .split(',')
   341→        .map((t) => t.trim().replace(/^["'«]|["'»]$/g, ''))
   342→        .filter(Boolean)
   343→        .slice(0, 6);
   344→    } else {
   345→      htmlLines.push(trimmed);
   346→    }
   347→  }
   348→
   349→  let html = htmlLines.join('\n').trim();
   350→  if (!html) return null;
   351→
   352→  if (!/<[a-z][\s\S]*>/i.test(html)) {
   353→    html = html
   354→      .split(/\n{2,}/)
   355→      .map((p) => `<p>${p.replace(/\n/g, '<br/>')}</p>`)
   356→      .join('\n');
   357→  }
   358→
   359→  if (!title) {
   360→    const h = html.match(/<h[1-3][^>]*>([\s\S]*?)<\/h[1-3]>/i);
   361→    title = h ? h[1].replace(/<[^>]+>/g, '').trim() : html.slice(0, 60).replace(/<[^>]+>/g, '');
   362→  }
   363→  if (!excerpt) {
   364→    const firstP = html.match(/<p[^>]*>([\s\S]*?)<\/p>/i);
   365→    excerpt = firstP ? firstP[1].replace(/<[^>]+>/g, '').trim().slice(0, 180) : title;
   366→  }
   367→  if (!metaDescription) {

   418→async function generateImagePrompt(title: string, topic: Topic): Promise<string | null> {
   419→  if (!zai) return null;
   420→  const styleHint =
   421→    topic.category === 'luxury-cars' || topic.category === 'luxury-vs-economy'
   422→      ? 'professional automotive photography, moody cinematic lighting'
   423→      : topic.category === 'tourism'
   424→        ? 'professional travel photography, golden-hour cinematic lighting'
   425→        : topic.category === 'sivan-brand'
   426→          ? 'professional automotive photography, premium elegant cinematic lighting'
   427→          : 'professional travel photography, cinematic lighting';
   428→  try {
   429→    const completion = await zai.chat.completions.create({
   430→      messages: [
   431→        {
   432→          role: 'user',
   433→          content: `You are a professional art director for a luxury Persian travel & taxi blog. I will give you the title of a Persian article. Produce ONE single English sentence describing a striking, topic-specific cover photo that visually matches the article's ACTUAL subject.
   434→
   435→Rules (very important):
   436→- The image MUST visually depict the specific subject of the article (the exact place, exact car model, or exact concept) — NOT a generic scene.
   437→- Translate any Persian place/car name into its English equivalent so the image model understands it (e.g. "Hormuz Island", "Mercedes-Benz E-Class", "Tehran to Mashhad highway").
   438→- Photorealistic scene description only. No abstract art, no illustrations.
   439→- Absolutely NO text, words, letters, logos or watermark in the image.
   440→- Do NOT show clearly visible human faces.
   441→- Do NOT wrap the answer in quotes or markdown. Do NOT add any explanation or label. Output ONLY the one sentence.
   442→- Keep the sentence between 20 and 80 words.
   443→
   444→Article title (Persian): ${title}
   445→Article focus keyword (Persian): ${topic.keyword}
   446→
   447→End your sentence with exactly: ", ${styleHint}, high quality, no text, no watermark".
   448→
   449→Write the one-sentence English image prompt now:`,
   450→        },
   451→      ],
   452→      thinking: { type: 'disabled' },
   453→    });
   454→    const raw = (completion?.choices?.[0]?.message?.content || '').trim();
   455→    if (!raw) return null;
   456→    const cleaned = raw
   457→      .replace(/^```[a-z]*\n?/i, '')
   458→      .replace(/\n?```$/i, '')
   459→      .replace(/^["'«]|["'»]$/g, '')
   460→      .replace(/^(prompt|image prompt|description|cover photo)\s*:\s*/i, '')
   461→      .replace(/\s+/g, ' ')
   462→      .trim();
   463→    if (cleaned.length < 15 || cleaned.length > 600) return null;
   464→    if (!/no text/i.test(cleaned)) {
   465→      return `${cleaned}, ${styleHint}, high quality, no text, no watermark`;
   466→    }
   467→    return cleaned;
   468→  } catch (err) {
   469→    console.error('[blog-generator] image-prompt LLM call failed:', err);
   470→    return null;
   471→  }

   506→function deriveKeyword(topic: string): string {
   507→  const stopwords = new Set([
   508→    'در', 'از', 'به', 'با', 'و', 'یا', 'را', 'است', 'نیست', 'برای', 'تا', 'که',

   514→  const words = topic
   515→    .replace(/[«»"'!?.,:;()\[\]{}،؛؟]/g, ' ')

   695→/**
   696→ * On startup, decide when the next generation should happen.
   697→ *
   698→ * Strategy:
   699→ * 1. Restore persisted state (cycleIndex, categoryTopicIndex, lastAutoGenAt)
   700→ * 2. If fewer than MIN_POSTS_FOR_IMMEDIATE_SKIP published posts exist → generate now (after 5s delay)
   701→ * 3. If lastAutoGenAt is recorded in DB and it's been >= 6h since then → generate now
   702→ * 4. If lastAutoGenAt is recorded but it's been < 6h → schedule the remaining time
   703→ * 5. If no lastAutoGenAt but enough posts exist → start a fresh 6h interval
   704→ */
   705→async function scheduleGenerationOnStartup() {
   706→  try {
   707→    const count = await db.blogPost.count({ where: { status: 'published' } });
   708→    console.log(`[blog-generator] current published posts: ${count}`);
   709→
   710→    if (count < MIN_POSTS_FOR_IMMEDIATE_SKIP) {
   711→      console.log(`[blog-generator] only ${count} published posts; generating one now (5s delay)...`);
   712→      setTimeout(() => generateArticle().catch((e) => console.error(e)), 5000);
   713→      return;
   714→    }
   715→
   716→    // Check persisted lastAutoGenAt from DB (restored into lastGeneratedAt)
   717→    if (lastGeneratedAt) {
   718→      const elapsed = Date.now() - new Date(lastGeneratedAt).getTime();
   719→      console.log(`[blog-generator] last auto-gen was ${Math.round(elapsed / 60000)}min ago`);
   720→      if (elapsed >= GENERATION_INTERVAL_MS) {
   721→        // Overdue — generate now
   722→        console.log('[blog-generator] overdue — generating now (5s delay)...');
   723→        setTimeout(() => generateArticle().catch((e) => console.error(e)), 5000);
   724→        return;
   725→      }
   726→      // Not yet due — wait the remaining time
   727→      const remaining = GENERATION_INTERVAL_MS - elapsed;
   728→      console.log(`[blog-generator] next generation in ${Math.round(remaining / 60000)}min (remaining from last auto-gen)`);
   729→      nextGenAt = new Date(Date.now() + remaining);
   730→      setTimeout(() => {
   731→        generateArticle().catch((e) => console.error('[blog-generator] scheduled error:', e));
   732→        // After this first (delayed) generation, set up the recurring interval
   733→        setInterval(() => {
   734→          generateArticle().catch((e) => console.error('[blog-generator] scheduled error:', e));
   735→        }, GENERATION_INTERVAL_MS);
   736→      }, remaining);
   737→      return;
   738→    }
   739→
   740→    // No lastAutoGenAt recorded but we have enough posts (edge case: posts
   741→    // were seeded or created manually, or DB state was lost). Start fresh.
   742→    console.log(`[blog-generator] no previous auto-gen recorded; starting fresh ${GENERATION_INTERVAL_MS / 3600000}h interval`);
   743→    nextGenAt = new Date(Date.now() + GENERATION_INTERVAL_MS);
   744→    setInterval(() => {
   745→      generateArticle().catch((e) => console.error('[blog-generator] scheduled error:', e));
   746→    }, GENERATION_INTERVAL_MS);
   747→  } catch (err) {
   748→    console.error('[blog-generator] startup scheduling failed:', err);
   749→    // Fallback: just start a fresh interval
   750→    setInterval(() => {
   751→      generateArticle().catch((e) => console.error('[blog-generator] fallback scheduled error:', e));
   752→    }, GENERATION_INTERVAL_MS);
   753→  }
   754→}

   764→function getNextGenInMs(): number {
   765→  if (!nextGenAt) return GENERATION_INTERVAL_MS;
   766→  const remaining = nextGenAt.getTime() - Date.now();
   767→  return remaining > 0 ? remaining : 0;

   770→// ---- HTTP server (Bun.serve) ----
   771→const server = Bun.serve({
   772→  port: PORT,
   773→  async fetch(req) {
   774→    const url = new URL(req.url);
   775→
   776→    if (req.method === 'OPTIONS') {
   777→      return new Response(null, {
   778→        status: 204,
   779→        headers: {
   780→          'Access-Control-Allow-Origin': '*',
   781→          'Access-Control-Allow-Methods': 'GET, POST, OPTIONS',
   782→          'Access-Control-Allow-Headers': 'Content-Type',
   783→        },
   784→      });
   785→    }
   786→
   787→    const corsHeaders = {
   788→      'Access-Control-Allow-Origin': '*',
   789→      'Access-Control-Allow-Methods': 'GET, POST, OPTIONS',
   790→      'Access-Control-Allow-Headers': 'Content-Type',
   791→    };
   792→
   793→    if (req.method === 'GET' && (url.pathname === '/health' || url.pathname === '/status')) {
   794→      const totalPosts = await getTotalPosts();
   795→      return Response.json(
   796→        {
   797→          ok: true,
   798→          service: 'blog-generator',
   799→          port: PORT,
   800→          running: isGenerating,
   801→          lastGeneratedAt,
   802→          lastError,
   803→          totalPosts,
   804→          nextGenInMs: getNextGenInMs(),
   805→        },
   806→        { headers: corsHeaders }
   807→      );
   808→    }
   809→    if (req.method === 'POST' && url.pathname === '/generate') {
   810→      if (isGenerating) {
   811→        return Response.json({ started: false, message: 'Generation already in progress' }, { headers: corsHeaders });
   812→      }
   813→      generateArticle().catch((e) => console.error('[blog-generator] on-demand error:', e));
   814→      return Response.json({ started: true }, { headers: corsHeaders });
   815→    }
   816→    if (req.method === 'POST' && url.pathname === '/generate-custom') {
   817→      if (isGenerating) {
   818→        return Response.json({ started: false, message: 'تولید دیگری در حال انجام است؛ لطفاً صبر کنید' }, { headers: corsHeaders });
   819→      }
   820→      let body: any = null;
   821→      try {
   822→        body = await req.json();
   823→      } catch {
   824→        return Response.json({ started: false, message: 'بدنه درخواست نامعتبر است' }, { status: 400, headers: corsHeaders });
   825→      }
   826→      const topic = typeof body?.topic === 'string' ? body.topic.trim() : '';
   827→      if (!topic || topic.length < 3) {
   828→        return Response.json({ started: false, message: 'موضوع مقاله باید حداقل ۳ کاراکتر باشد' }, { status: 400, headers: corsHeaders });
   829→      }
   830→      if (topic.length > 200) {
   831→        return Response.json({ started: false, message: 'موضوع مقاله نباید بیشتر از ۲۰۰ کاراکتر باشد' }, { status: 400, headers: corsHeaders });
   832→      }
   833→      console.log(`[blog-generator] admin requested custom topic: "${topic}"`);
   834→      generateArticle(topic).catch((e) => console.error('[blog-generator] custom on-demand error:', e));
   835→      return Response.json({ started: true, custom: true, topic }, { headers: corsHeaders });
   836→    }
   837→    return new Response('Not Found', { status: 404, headers: corsHeaders });
   838→  },
   839→});
