     2→
     3→import { useState, useEffect, useRef } from 'react';
     4→import { motion, AnimatePresence } from 'framer-motion';
     5→import {
     6→  User,
     7→  Loader2,
     8→  Check,
     9→  AlertTriangle,
    10→  UserPlus,

    24→} from 'lucide-react';
    25→import {
    26→  Dialog,
    27→  DialogContent,
    28→  DialogHeader,
    29→  DialogTitle,
    30→  DialogDescription,
    31→} from '@/components/ui/dialog';
    32→import { Button } from '@/components/ui/button';
    33→import { Input } from '@/components/ui/input';
    34→import { InputOTP, InputOTPGroup, InputOTPSlot, InputOTPSeparator } from '@/components/ui/input-otp';
    35→import { Label } from '@/components/ui/label';

    37→import { toast } from 'sonner';
    38→
    39→// ─── Step tracking ──────────────────────────────────────────────
    40→
    41→type OTPStep = 'phone' | 'code' | 'register' | 'success';
    42→type RegisterMode = 'passenger' | 'driver';
    43→type DriverSubStep = 'select-type' | 'shahkar' | 'info' | 'vehicle';
    44→
    45→export function AuthModal() {
    46→  const {
    47→    auth,
    48→    closeAuth,
    49→    setAuthUsername,
    50→    setAuthFullName,
    51→    setAuthPhone,
    52→    setAuthUser,
    53→    setAuthVerified,
    54→    adminLogin,
    55→    setAdminOpen,
    56→    openAuth,
    57→    setUserPanelOpen,
    58→  } = useAppStore();
    59→
    60→  const [step, setStep] = useState<OTPStep>('phone');
    61→  const [loading, setLoading] = useState(false);
    62→  const [localError, setLocalError] = useState('');
    63→  const [otpSent, setOtpSent] = useState(false);
    64→  const [isDemo, setIsDemo] = useState(false);
    65→  const [demoOtp, setDemoOtp] = useState('');
    66→  const [countdown, setCountdown] = useState(0);
    67→  const [isExistingUser, setIsExistingUser] = useState(false);
    68→  const [existingUser, setExistingUser] = useState<{ id: string; fullName: string; username: string; role: string } | null>(null);
    69→
    70→  // Legacy login
    71→  const [legacyMode, setLegacyMode] = useState(false);
    72→  const [showPass, setShowPass] = useState(false);
    73→
    74→  // Driver registration state
    75→  const [registerMode, setRegisterMode] = useState<RegisterMode>('passenger');
    76→  const [driverSubStep, setDriverSubStep] = useState<DriverSubStep>('select-type');
    77→  const [nationalId, setNationalId] = useState('');
    78→  const [birthDate, setBirthDate] = useState('');
    79→  const [fatherName, setFatherName] = useState('');
    80→  const [licenseNumber, setLicenseNumber] = useState('');
    81→  const [shebaNumber, setShebaNumber] = useState('');
    82→  const [vehicleType, setVehicleType] = useState('sedan');
    83→  const [vehicleBrand, setVehicleBrand] = useState('');
    84→  const [vehicleModel, setVehicleModel] = useState('');
    85→  const [vehicleYear, setVehicleYear] = useState('');
    86→  const [vehicleColor, setVehicleColor] = useState('');
    87→  const [plateNumber, setPlateNumber] = useState('');
    88→
    89→  const countdownRef = useRef<ReturnType<typeof setInterval>>();
    90→  const inputRef = useRef<HTMLInputElement>(null);
    91→
    92→  // Reset when modal opens
    93→  useEffect(() => {
    94→    if (auth.isOpen) {
    95→      setStep('phone');
    96→      setLocalError('');
    97→      setOtpSent(false);
    98→      setIsDemo(false);
    99→      setDemoOtp('');
   100→      setCountdown(0);
   101→      setIsExistingUser(false);
   102→      setExistingUser(null);
   103→      setLegacyMode(false);
   104→      setShowPass(false);
   105→      setRegisterMode('passenger');
   106→      setDriverSubStep('select-type');
   107→      setNationalId('');
   108→      setBirthDate('');
   109→      setFatherName('');
   110→      setLicenseNumber('');
   111→      setShebaNumber('');
   112→      setVehicleType('sedan');
   113→      setVehicleBrand('');
   114→      setVehicleModel('');
   115→      setVehicleYear('');
   116→      setVehicleColor('');
   117→      setPlateNumber('');
   118→      setTimeout(() => inputRef.current?.focus(), 200);
   119→    }
   120→  }, [auth.isOpen]);
   121→
   122→  // Countdown timer
   123→  useEffect(() => {
   124→    if (countdown > 0) {
   125→      countdownRef.current = setInterval(() => {
   126→        setCountdown((c) => {
   127→          if (c <= 1) { clearInterval(countdownRef.current); return 0; }
   128→          return c - 1;
   129→        });
   130→      }, 1000);
   131→      return () => clearInterval(countdownRef.current);
   132→    }
   133→  }, [countdown]);
   134→
   135→  // ─── Step 1: Send OTP ──────────────────────────────────────
   136→
   137→  const handleSendOTP = async () => {
   138→    const phone = auth.phone.trim();
   139→    if (!phone || !/^09[0-9]{9}$/.test(phone)) {
   140→      setLocalError('شماره موبایل معتبر وارد کنید (مثال: ۰۹۱۲۳۴۵۶۷۸۹)');
   141→      return;
   142→    }
   143→    setLocalError('');
   144→    setLoading(true);
   145→
   146→    try {
   147→      const res = await fetch('/api/auth/send-otp', {
   148→        method: 'POST',
   149→        headers: { 'Content-Type': 'application/json' },
   150→        body: JSON.stringify({ phone }),
   151→      });
   152→
   153→      const data = await res.json();
   154→
   155→      if (!res.ok) {
   156→        setLocalError(data.error || 'خطا در ارسال کد تأیید');
   157→        setLoading(false);
   158→        return;
   159→      }
   160→
   161→      setOtpSent(true);
   162→      setIsDemo(!!data.isDemo);
   163→      if (data.isDemo && data.otp) {
   164→        setDemoOtp(data.otp);
   165→      }
   166→      setCountdown(120);
   167→      setStep('code');
   168→      toast.success('کد تأیید ارسال شد');
   169→    } catch {
   170→      setLocalError('خطا در ارتباط با سرور');
   171→    } finally {
   172→      setLoading(false);
   173→    }
   174→  };
   175→
   176→  // ─── Step 2: Verify OTP ─────────────────────────────────────
   177→
   178→  const handleVerifyOTP = async (code: string) => {
   179→    if (!code || code.length !== 6) return;
   180→    setLocalError('');
   181→    setLoading(true);
   182→
   183→    try {
   184→      const res = await fetch('/api/auth/verify-otp', {
   185→        method: 'POST',
   186→        headers: { 'Content-Type': 'application/json' },
   187→        body: JSON.stringify({ phone: auth.phone, code }),
   188→      });
   189→
   190→      const data = await res.json();
   191→
   192→      if (!res.ok) {
   193→        setLocalError(data.error || 'کد تأیید نامعتبر');
   194→        setLoading(false);
   195→        return;
   196→      }
   197→
   198→      if (data.isNewUser) {
   199→        setStep('register');
   200→        setDriverSubStep('select-type');
   201→        setLoading(false);
   202→      } else {
   203→        setIsExistingUser(true);
   204→        setExistingUser(data.user);
   205→        setAuthUser({
   206→          id: data.user.id,
   207→          username: data.user.username || data.user.phone,
   208→          fullName: data.user.fullName,
   209→          role: data.user.role,
   210→        });
   211→        setAuthVerified(true);
   212→        setStep('success');
   213→
   214→        if (data.user?.role === 'admin') {
   215→          adminLogin(data.user.username, '');
   216→          setTimeout(() => { setAdminOpen(true); closeAuth(); }, 1200);
   217→        } else {
   218→          setTimeout(() => { closeAuth(); setUserPanelOpen(true); }, 1200);
   219→        }
   220→
   221→        toast.success(`خوش آمدید، ${data.user?.fullName || 'کاربر'} عزیز`);
   222→      }
   223→    } catch {
   224→      setLocalError('خطا در ارتباط با سرور');
   225→    } finally {
   226→      setLoading(false);
   227→    }
   228→  };
   229→
   230→  // ─── Step 3a: Select Registration Type ──────────────────────
   231→
   232→  const handleSelectRegisterMode = (mode: RegisterMode) => {
   233→    setRegisterMode(mode);
   234→    setLocalError('');
   235→    if (mode === 'passenger') {
   236→      // Just show name input — no sub-step change needed
   237→    } else {
   238→      setDriverSubStep('shahkar');
   239→    }
   240→  };
   241→
   242→  // ─── Step 3b: Shahkar Verification (Driver) ─────────────────
   243→
   244→  const handleShahkarVerify = async () => {
   245→    if (!nationalId || !/^[0-9]{10}$/.test(nationalId)) {
   246→      setLocalError('کد ملی باید ۱۰ رقم باشد');
   247→      return;
   248→    }
   249→    setLocalError('');
   250→    setLoading(true);
   251→
   252→    try {
   253→      const res = await fetch('/api/auth/shahkar-verify', {
   254→        method: 'POST',
   255→        headers: { 'Content-Type': 'application/json' },
   256→        body: JSON.stringify({ nationalId, phone: auth.phone, birthDate: birthDate || undefined }),
   257→      });
   258→
   259→      const data = await res.json();
   260→      if (!res.ok) {
   261→        setLocalError(data.error || 'خطا در تأیید شاهکار');
   262→        setLoading(false);
   263→        return;
   264→      }
   265→      if (!data.verified) {
   266→        setLocalError('تطبیق شاهکار انجام نشد. شماره موبایل با کد ملی مطابقت ندارد.');
   267→        setLoading(false);
   268→        return;
   269→      }
   270→      if (data.personInfo) {
   271→        if (data.personInfo.firstName || data.personInfo.lastName) {
   272→          setAuthFullName(`${data.personInfo.firstName || ''} ${data.personInfo.lastName || ''}`.trim());
   273→        }
   274→        if (data.personInfo.fatherName) {
   275→          setFatherName(data.personInfo.fatherName);
   276→        }
   277→      }
   278→      setDriverSubStep('info');
   279→      toast.success(data.isDemo ? 'تأیید شاهکار موفق (آزمایشی)' : 'تأیید شاهکار موفق');
   280→    } catch {
   281→      setLocalError('خطا در ارتباط');
   282→    } finally {
   283→      setLoading(false);
   284→    }
   285→  };
   286→
   287→  // ─── Step 3c: Driver Personal Info ──────────────────────────
   288→
   289→  const handleDriverInfoSubmit = () => {
   290→    const fullName = auth.fullName.trim();
   291→    if (!fullName || fullName.length < 2) {
   292→      setLocalError('نام و نام خانوادگی را کامل وارد کنید');
   293→      return;
   294→    }
   295→    setLocalError('');
   296→    setDriverSubStep('vehicle');
   297→  };
   298→
   299→  // ─── Step 3d: Vehicle Info + Final Submit ───────────────────
   300→
   301→  const handleDriverSubmit = async () => {
   302→    if (!vehicleBrand || !vehicleModel || !vehicleYear || !plateNumber) {
   303→      setLocalError('اطلاعات خودرو را کامل وارد کنید');
   304→      return;
   305→    }
   306→    setLocalError('');
   307→    setLoading(true);
   308→
   309→    try {
   310→      // First register as user
   311→      const regRes = await fetch('/api/auth/register', {
   312→        method: 'POST',
   313→        headers: { 'Content-Type': 'application/json' },
   314→        body: JSON.stringify({
   315→          username: `driver_${auth.phone}`,
   316→          fullName: auth.fullName.trim(),
   317→          password: `pass_${Date.now()}`,
   318→          phone: auth.phone,
   319→        }),
   320→      });
   321→
   322→      const regData = await regRes.json();
   323→
   324→      // Then submit driver application
   325→      const driverRes = await fetch('/api/auth/register-driver', {
   326→        method: 'POST',
   327→        headers: { 'Content-Type': 'application/json' },
   328→        body: JSON.stringify({
   329→          phone: auth.phone,
   330→          nationalId,
   331→          fullName: auth.fullName.trim(),
   332→          fatherName: fatherName.trim(),
   333→          birthDate: birthDate || undefined,
   334→          licenseNumber: licenseNumber || undefined,
   335→          shebaNumber: shebaNumber || undefined,
   336→          vehicle: {
   337→            type: vehicleType,
   338→            brand: vehicleBrand,
   339→            model: vehicleModel,
   340→            year: parseInt(vehicleYear) || 1402,
   341→            color: vehicleColor,
   342→            plateNumber,
   343→          },
   344→        }),
   345→      });
   346→
   347→      const driverData = await driverRes.json();
   348→
   349→      if (!driverRes.ok) {
   350→        setLocalError(driverData.error || 'خطا در ثبت‌نام راننده');
   351→        setLoading(false);
   352→        return;
   353→      }
   354→
   355→      // Set authenticated user
   356→      if (regRes.ok && regData.user) {
   357→        setAuthUser({
   358→          id: regData.user.id,
   359→          username: regData.user.username,
   360→          fullName: regData.user.fullName,
   361→          role: regData.user.role,
   362→        });
   363→        setAuthVerified(true);
   364→      }
   365→
   366→      setStep('success');
   367→      toast.success('درخواست ثبت‌نام راننده با موفقیت ثبت شد');
   368→      setTimeout(() => {
   369→        closeAuth();
   370→        if (regRes.ok) setUserPanelOpen(true);
   371→      }, 1200);
   372→    } catch {
   373→      setLocalError('خطا در ارتباط');
   374→    } finally {
   375→      setLoading(false);
   376→    }
   377→  };
   378→
   379→  // ─── Step 3e: Complete Passenger Registration ──────────────
   380→
   381→  const handleCompleteRegistration = async () => {
   382→    const fullName = auth.fullName.trim();
   383→    if (!fullName || fullName.length < 2) {
   384→      setLocalError('نام و نام خانوادگی را کامل وارد کنید');
   385→      return;
   386→    }
   387→    setLocalError('');
   388→    setLoading(true);
   389→
   390→    try {
   391→      const res = await fetch('/api/auth/register', {
   392→        method: 'POST',
   393→        headers: { 'Content-Type': 'application/json' },
   394→        body: JSON.stringify({
   395→          username: `user_${auth.phone}`,
   396→          fullName,
   397→          password: `pass_${Date.now()}`,
   398→          phone: auth.phone,
   399→        }),
   400→      });
   401→
   402→      const data = await res.json();
   403→
   404→      if (!res.ok) {
   405→        setLocalError(data.error || 'خطا در ثبت‌نام');
   406→        setLoading(false);
   407→        return;
   408→      }
   409→
   410→      setAuthUser({
   411→        id: data.user.id,
   412→        username: data.user.username,
   413→        fullName: data.user.fullName,
   414→        role: data.user.role,
   415→      });
   416→      setAuthVerified(true);
   417→      setStep('success');
   418→
   419→      toast.success(`ثبت‌نام موفق! خوش آمدید، ${data.user.fullName} عزیز`);
   420→      setTimeout(() => { closeAuth(); setUserPanelOpen(true); }, 1200);
   421→    } catch {
   422→      setLocalError('خطا در ارتباط با سرور');
   423→    } finally {
   424→      setLoading(false);
   425→    }
   426→  };
   427→
   428→  // ─── Legacy login via username/password ─────────────────────
   429→
   430→  const handleLegacyLogin = async () => {
   431→    if (!auth.username || !auth.password) {
   432→      setLocalError('نام کاربری و رمز عبور را وارد کنید');
   433→      return;
   434→    }
   435→    setLocalError('');
   436→    setLoading(true);
   437→
   438→    try {
   439→      const res = await fetch('/api/auth/login', {
   440→        method: 'POST',
   441→        headers: { 'Content-Type': 'application/json' },
   442→        body: JSON.stringify({ username: auth.username, password: auth.password }),
   443→      });
   444→
   445→      const data = await res.json();
   446→
   447→      if (!res.ok) {
   448→        setLocalError(data.error || 'خطا در ورود');
   449→        setLoading(false);
   450→        return;
   451→      }
   452→
   453→      setAuthUser(data.user || null);
   454→
   455→      if (data.user?.role === 'admin') {
   456→        adminLogin(auth.username, auth.password);
   457→        setAdminOpen(true);
   458→        closeAuth();
   459→        setLoading(false);
   460→        return;
   461→      }
   462→
   463→      setAuthVerified(true);
   464→      setLoading(false);
   465→      toast.success(`خوش آمدید، ${data.user?.fullName || 'کاربر'} عزیز`);
   466→      setTimeout(() => { closeAuth(); setUserPanelOpen(true); }, 900);
   467→    } catch {
   468→      setLocalError('خطا در ارتباط با سرور');
   469→      setLoading(false);
   470→    }
   471→  };
   472→
   473→  const resetAndClose = () => { setLocalError(''); closeAuth(); };
   474→
   475→  const formatCountdown = (s: number) => {
   476→    const min = Math.floor(s / 60);
   477→    const sec = s % 60;
   478→    return `${min}:${sec.toString().padStart(2, '0')}`;
   479→  };
   480→
   481→  // Driver step labels
   482→  const driverStepLabels = ['موبایل', 'کد تأیید', 'شاهکار', 'اطلاعات', 'خودرو'];
   483→  const getDriverStepIndex = () => {
   484→    switch (driverSubStep) {
   485→      case 'shahkar': return 2;
   486→      case 'info': return 3;
   487→      case 'vehicle': return 4;
   488→      default: return 2;
   489→    }
   490→  };
   491→
   492→  return (
   493→    <Dialog open={auth.isOpen} onOpenChange={(val) => { if (!val) resetAndClose(); }}>
   494→      <DialogContent className="bg-[#1a1a1a] border-[#333] max-w-md max-h-[85vh] overflow-y-auto">
   495→        <DialogHeader>
   496→          <DialogTitle className="text-right text-[#fafafa]">
   497→            {step === 'success'
   498→              ? 'خوش آمدید!'
   499→              : step === 'register' && registerMode === 'driver' && driverSubStep === 'vehicle'
   500→                ? 'اطلاعات خودرو'
   501→              : step === 'register' && registerMode === 'driver' && driverSubStep === 'info'
   502→                ? 'اطلاعات شخصی راننده'
   503→              : step === 'register' && registerMode === 'driver' && driverSubStep === 'shahkar'
   504→                ? 'تأیید شاهکار'
   505→                : step === 'register'
   506→                  ? 'تکمیل ثبت‌نام'
   507→                : step === 'code'
   508→                  ? 'تأیید شماره موبایل'
   509→                  : legacyMode
   510→                    ? 'ورود با نام کاربری'
   511→                    : 'ورود / ثبت‌نام'}
   512→          </DialogTitle>
   513→          <DialogDescription className="text-right text-[#a1a1aa]">
   514→            {step === 'success' && 'حساب شما با موفقیت تأیید شد'}
   515→            {step === 'register' && registerMode === 'driver' && driverSubStep === 'select-type' && 'نوع حساب کاربری خود را انتخاب کنید'}
   516→            {step === 'register' && registerMode === 'driver' && driverSubStep === 'shahkar' && 'تأیید هویت شاهکار — تطبیق کد ملی با شماره موبایل'}
   517→            {step === 'register' && registerMode === 'driver' && driverSubStep === 'info' && 'اطلاعات شخصی راننده'}
   518→            {step === 'register' && registerMode === 'driver' && driverSubStep === 'vehicle' && 'اطلاعات خودرو'}
   519→            {step === 'register' && registerMode === 'passenger' && 'برای تکمیل ثبت‌نام، نام خود را وارد کنید'}
   520→            {step === 'code' && `کد تأیید ارسال شده به ${auth.phone} را وارد کنید`}
   521→            {step === 'phone' && !legacyMode && 'شماره موبایل خود را وارد کنید تا کد تأیید ارسال شود'}
   522→            {step === 'phone' && legacyMode && 'نام کاربری و رمز عبور خود را وارد کنید'}
   523→          </DialogDescription>
   524→        </DialogHeader>
   525→
   526→        <div className="mt-2">
   527→          <AnimatePresence mode="wait">
   528→            {/* ─── Success State ─────────────────────────── */}
   529→            {step === 'success' && (
   530→              <motion.div
   531→                key="success"
   532→                initial={{ opacity: 0, scale: 0.9 }}
   533→                animate={{ opacity: 1, scale: 1 }}
   534→                className="text-center py-6 space-y-4"
   535→              >
   536→                <div className="w-16 h-16 rounded-full bg-green-500/20 flex items-center justify-center mx-auto">
   537→                  {registerMode === 'driver' ? (
   538→                    <BadgeCheck className="h-8 w-8 text-green-400" />
   539→                  ) : (
   540→                    <Check className="h-8 w-8 text-green-400" />
   541→                  )}
   542→                </div>
   543→                <p className="text-[#fafafa] font-medium">
   544→                  {registerMode === 'driver'
   545→                    ? 'درخواست ثبت‌نام راننده شما ثبت شد'
   546→                    : auth.user?.fullName
   547→                      ? `${auth.user.fullName} عزیز، خوش آمدید!`
   548→                      : 'خوش آمدید!'}
   549→                </p>
   550→                {registerMode === 'driver' && (
   551→                  <p className="text-[#a1a1aa] text-xs">پس از بررسی اطلاعات، نتیجه از طریق پیامک اطلاع‌رسانی خواهد شد</p>
   552→                )}
   553→                <Button onClick={resetAndClose} className="bg-[#D4AF37] text-[#0a0a0a] hover:bg-[#E5C76B]">
   554→                  بستن
   555→                </Button>
   556→              </motion.div>
   557→            )}
   558→

   948→                </div>
   949→
   950→                {localError && <ErrorBox msg={localError} />}
   951→
   952→                <Button onClick={handleLegacyLogin} disabled={loading || !auth.username || !auth.password} className="w-full bg-[#D4AF37] text-[#0a0a0a] hover:bg-[#E5C76B] h-12 font-medium">
   953→                  {loading ? <Loader2 className="h-4 w-4 animate-spin" /> : <><LogIn className="h-4 w-4 ml-2" />ورود به حساب</>}

   959→                </div>
   960→              </motion.div>
   961→            )}
   962→          </AnimatePresence>
   963→        </div>
   964→      </DialogContent>
   965→    </Dialog>
   966→  );
   967→}
