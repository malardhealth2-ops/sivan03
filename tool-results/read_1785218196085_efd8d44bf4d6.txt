   101→// ─── Invalidate Map Size Handler ─────────────────────────────────────────────
   102→
   103→function MapInvalidator() {
   104→  const map = useMap();
   105→
   106→  useEffect(() => {
   107→    // Force Leaflet to recalculate its size after the container is fully rendered
   108→    const timer = setTimeout(() => {
   109→      map.invalidateSize();
   110→    }, 200);
   111→    return () => clearTimeout(timer);
   112→  }, [map]);
   113→
   114→  // Also listen for scroll/resize events that might change the container
   115→  useEffect(() => {
   116→    let frameId: number;
   117→    const handleUpdate = () => {
   118→      if (frameId) cancelAnimationFrame(frameId);
   119→      frameId = requestAnimationFrame(() => {
   120→        map.invalidateSize({ animate: false });
   121→      });
   122→    };
   123→
   124→    window.addEventListener('resize', handleUpdate);
   125→    const observer = new IntersectionObserver((entries) => {
   126→      for (const entry of entries) {
   127→        if (entry.isIntersecting) {
   128→          setTimeout(() => map.invalidateSize({ animate: false }), 100);
   129→        }
   130→      }
   131→    }, { threshold: 0.1 });
   132→
   133→    const container = map.getContainer();
   134→    if (container) observer.observe(container);
   135→
   136→    return () => {
   137→      window.removeEventListener('resize', handleUpdate);
   138→      observer.disconnect();
   139→      if (frameId) cancelAnimationFrame(frameId);
   140→    };
   141→  }, [map]);
   142→
   143→  return null;
   144→}

   146→// ─── Fit to Bounds Handler ──────────────────────────────────────────────────
   147→
   148→function FitBoundsHandler({
   149→  points,
   150→  routeData,
   151→}: {
   152→  points: { origin: MapPoint | null; destination: MapPoint | null };
   153→  routeData: RouteResponse | null;
   154→}) {
   155→  const map = useMap();
   156→
   157→  useEffect(() => {
   158→    if (routeData && routeData.routes.length > 0 && routeData.routes[0].path.length > 0) {
   159→      const polyline = L.polyline(routeData.routes[0].path);
   160→      const bounds = polyline.getBounds().pad(0.15);
   161→      map.flyToBounds(bounds, { duration: 1.2 });
   162→    } else if (points.origin && points.destination) {
   163→      const bounds = L.latLngBounds(
   164→        [points.origin.lat, points.origin.lng],
   165→        [points.destination.lat, points.destination.lng]
   166→      ).pad(0.3);
   167→      map.flyToBounds(bounds, { duration: 1 });
   168→    } else if (points.origin) {
   169→      map.flyTo([points.origin.lat, points.origin.lng], 12, { duration: 1 });
   170→    }
   171→  }, [points, routeData, map]);
   172→
   173→  return null;
   174→}

   297→export default function InteractiveMapInner() {
   298→  const [origin, setOrigin] = useState<MapPoint | null>(null);
   299→  const [destination, setDestination] = useState<MapPoint | null>(null);
   300→  const [selectionStep, setSelectionStep] = useState<SelectionStep>('origin');
   301→  const [routeData, setRouteData] = useState<RouteResponse | null>(null);
   302→  const [routeLoading, setRouteLoading] = useState(false);
   303→  const [routeError, setRouteError] = useState('');
   304→  const [activeRoute, setActiveRoute] = useState(0);
   305→
   306→  // Search state
   307→  const [originSearch, setOriginSearch] = useState('');
   308→  const [destSearch, setDestSearch] = useState('');
   309→  const [originResults, setOriginResults] = useState<
   310→    { display_name: string; lat: string; lon: string }[]
   311→  >([]);
   312→  const [destResults, setDestResults] = useState<
   313→    { display_name: string; lat: string; lon: string }[]
   314→  >([]);
   315→  const [searchLoading, setSearchLoading] = useState<'origin' | 'dest' | null>(null);
   316→  const searchTimer = useRef<ReturnType<typeof setTimeout>>();
   317→
   318→  // Handle map click
   319→  const handleSelectPoint = useCallback(
   320→    (lat: number, lng: number) => {
   321→      if (selectionStep === 'origin') {
   322→        setOrigin({ lat, lng });
   323→        setRouteData(null);
   324→        setActiveRoute(0);
   325→        setSelectionStep('destination');
   326→      } else if (selectionStep === 'destination') {
   327→        if (origin && Math.abs(origin.lat - lat) < 0.001 && Math.abs(origin.lng - lng) < 0.001) {
   328→          return;
   329→        }
   330→        setDestination({ lat, lng });
   331→        setSelectionStep('ready');
   332→      } else {
   333→        clearSelection();
   334→      }
   335→    },
   336→    [selectionStep, origin]
   337→  );
   338→
   339→  // Search function
   340→  const handleSearch = useCallback(
   341→    (query: string, type: 'origin' | 'dest') => {
   342→      if (searchTimer.current) clearTimeout(searchTimer.current);
   343→
   344→      if (!query || query.length < 2) {
   345→        if (type === 'origin') setOriginResults([]);
   346→        else setDestResults([]);
   347→        return;
   348→      }
   349→
   350→      setSearchLoading(type);
   351→
   352→      searchTimer.current = setTimeout(async () => {
   353→        try {
   354→          const encoded = encodeURIComponent(query + ', ایران');
   355→          const url = `https://nominatim.openstreetmap.org/search?q=${encoded}&format=json&limit=5&accept-language=fa&addressdetails=1`;
   356→
   357→          const res = await fetch(url, {
   358→            headers: {
   359→              'User-Agent': 'SivanVIPTaxi/1.0',
   360→            },
   361→          });
   362→
   363→          if (!res.ok) return;
   364→
   365→          const data = await res.json();
   366→          if (type === 'origin') {
   367→            setOriginResults(data || []);
   368→          } else {
   369→            setDestResults(data || []);
   370→          }
   371→        } catch {
   372→          // ignore search errors
   373→        } finally {
   374→          setSearchLoading(null);
   375→        }
   376→      }, 800);
   377→    },
   378→    []
   379→  );
   380→
   381→  // Handle search select
   382→  const handleSearchSelect = useCallback(
   383→    (lat: number, lng: number, name: string, type: 'origin' | 'dest') => {
   384→      if (type === 'origin') {
   385→        setOrigin({ lat, lng, name });
   386→        setOriginSearch(name);
   387→        setOriginResults([]);
   388→        if (!destination) {
   389→          setSelectionStep('destination');
   390→        } else {
   391→          setSelectionStep('ready');
   392→        }
   393→      } else {
   394→        setDestination({ lat, lng, name });
   395→        setDestSearch(name);
   396→        setDestResults([]);
   397→        if (origin) {
   398→          setSelectionStep('ready');
   399→        } else {
   400→          setSelectionStep('origin');
   401→        }
   402→      }
   403→    },
   404→    [origin, destination]
   405→  );
   406→
   407→  // Fetch route when both points selected
   408→  useEffect(() => {
   409→    if (origin && destination) {
   410→      setRouteLoading(true);
   411→      setRouteError('');
   412→
   413→      const timer = setTimeout(async () => {
   414→        try {
   415→          const res = await fetch(
   416→            `/api/map/route?originLat=${origin.lat}&originLng=${origin.lng}&destLat=${destination.lat}&destLng=${destination.lng}`
   417→          );
   418→
   419→          if (!res.ok) {
   420→            const err = await res.json().catch(() => ({ error: 'خطا در محاسبه مسیر' }));
   421→            throw new Error(err.error || 'خطا');
   422→          }
   423→
   424→          const data = await res.json();
   425→          setRouteData(data);
   426→
   427→          if (data.origin?.name && !origin.name) {
   428→            setOrigin((prev) => prev ? { ...prev, name: data.origin.name } : prev);
   429→            setOriginSearch(data.origin.name);
   430→          }
   431→          if (data.destination?.name && !destination.name) {
   432→            setDestination((prev) => prev ? { ...prev, name: data.destination.name } : prev);
   433→            setDestSearch(data.destination.name);
   434→          }
   435→        } catch (err) {
   436→          setRouteError(err instanceof Error ? err.message : 'خطا در محاسبه مسیر');
   437→          setRouteData(null);
   438→        } finally {
   439→          setRouteLoading(false);
   440→        }
   441→      }, 500);
   442→
   443→      return () => clearTimeout(timer);
   444→    }
   445→  }, [origin?.lat, origin?.lng, destination?.lat, destination?.lng]);
   446→
   447→  const clearSelection = () => {
   448→    setOrigin(null);
   449→    setDestination(null);
   450→    setRouteData(null);
   451→    setRouteError('');
   452→    setActiveRoute(0);
   453→    setOriginSearch('');
   454→    setDestSearch('');
   455→    setOriginResults([]);
   456→    setDestResults([]);
   457→    setSelectionStep('origin');
   458→  };
   459→
   460→  const swapPoints = () => {
   461→    if (!origin || !destination) return;
   462→    const tmpOrigin = origin;
   463→    const tmpOriginSearch = originSearch;
   464→    setOrigin(destination);
   465→    setDestination(tmpOrigin);
   466→    setOriginSearch(destSearch);
   467→    setDestSearch(tmpOriginSearch);
   468→    setRouteData(null);
   469→    setActiveRoute(0);
   470→  };
   471→
   472→  // Build polylines
   473→  const routePolylines = routeData?.routes.map((route) => route.path) || [];
   474→
   475→  return (
   476→    <section id="map" className="py-20 sm:py-24 bg-[#0a0a0a] relative">
   477→      {/* Decorative */}
   478→      <div className="absolute top-0 left-1/2 -translate-x-1/2 w-24 h-1 bg-gradient-to-l from-transparent via-[#D4AF37] to-transparent rounded-full" />
   479→
   480→      <div className="container mx-auto px-4 sm:px-6">
   481→        {/* Section Header */}
   482→        <motion.div
   483→          initial={{ opacity: 0, y: 20 }}
   484→          whileInView={{ opacity: 1, y: 0 }}
   485→          viewport={{ once: true, margin: '-100px' }}
   486→          transition={{ duration: 0.6 }}
   487→          className="text-center mb-10 sm:mb-14"
   488→        >
   489→          <Badge className="bg-[#D4AF37]/15 text-[#D4AF37] border-[#D4AF37]/30 mb-4 px-4 py-1.5">
   490→            <Navigation className="h-3.5 w-3.5 ml-1.5" />
   491→            نقشه تعاملی
   492→          </Badge>
   493→          <h2 className="text-2xl sm:text-3xl md:text-4xl font-bold text-[#fafafa] mb-4">
   494→            مسیر خود را روی <span className="text-gold-gradient">نقشه</span> انتخاب کنید
