   160→/** Restore generation state from the DB (called once on startup). */
   161→async function restoreState(): Promise<void> {
   162→  try {
   163→    let state = await db.blogGeneratorState.findUnique({ where: { id: 'main' } });
   164→    if (!state) {
   165→      // First run ever — create the row with defaults.
   166→      state = await db.blogGeneratorState.create({ data: { id: 'main' } });
   167→      console.log('[blog-generator] created initial DB state row');
   168→    }
   169→    cycleIndex = state.cycleIndex;
   170→    const parsed = JSON.parse(state.categoryTopicIndex || '{}');
   171→    for (const cat of Object.keys(categoryTopicIndex) as TopicCategory[]) {
   172→      if (typeof parsed[cat] === 'number') {
   173→        categoryTopicIndex[cat] = parsed[cat];
   174→      }
   175→    }
   176→    if (state.lastAutoGenAt) {
   177→      lastGeneratedAt = state.lastAutoGenAt.toISOString();
   178→    }
   179→    console.log(`[blog-generator] state restored: cycleIndex=${cycleIndex}, categoryTopicIndex=${JSON.stringify(categoryTopicIndex)}, lastAutoGenAt=${lastGeneratedAt || 'never'}, totalAutoGenerated=${state.totalAutoGenerated}`);
   180→  } catch (err) {
   181→    console.error('[blog-generator] failed to restore state from DB, starting fresh:', err);
   182→  }
   183→}

   185→/** Persist current generation state to the DB (called after each auto-generation). */
   186→async function saveState(): Promise<void> {
   187→  try {
   188→    await db.blogGeneratorState.upsert({
   189→      where: { id: 'main' },
   190→      update: {
   191→        cycleIndex,
   192→        categoryTopicIndex: JSON.stringify(categoryTopicIndex),
   193→        lastAutoGenAt: lastGeneratedAt ? new Date(lastGeneratedAt) : null,
   194→        totalAutoGenerated: { increment: 1 },
   195→      },
   196→      create: {
   197→        id: 'main',
   198→        cycleIndex,
   199→        categoryTopicIndex: JSON.stringify(categoryTopicIndex),
   200→        lastAutoGenAt: lastGeneratedAt ? new Date(lastGeneratedAt) : null,
   201→      },
   202→    });
   203→    console.log('[blog-generator] state saved to DB');
   204→  } catch (err) {
   205→    console.error('[blog-generator] failed to save state to DB:', err);
   206→  }
   207→}

   209→function pickTopic(): Topic {
   210→  const category = CATEGORY_ORDER[cycleIndex % CATEGORY_ORDER.length];
   211→  cycleIndex++;
   212→  const pool = TOPICS.filter((t) => t.category === category);
   213→  const idx = categoryTopicIndex[category] % pool.length;
   214→  categoryTopicIndex[category]++;
   215→  return pool[idx];
   216→}

   218→function makeSlug(title: string): string {
   219→  const base = title
   220→    .replace(/[«»"'!?.,:;()\[\]{}]/g, '')
   221→    .replace(/\s+/g, '-')
   222→    .replace(/-+/g, '-')
   223→    .trim()
   224→    .slice(0, 60);
   225→  const suffix = Date.now().toString(36).slice(-6);
   226→  return `${base}-${suffix}`;
   227→}

   229→function parseArticle(raw: string): {
   230→  title: string;
   231→  excerpt: string;
   232→  html: string;
   233→  tags: string[];
   234→  metaDescription: string;
   235→} | null {
   236→  if (!raw) return null;
   237→  const lines = raw.split('\n');
   238→
   239→  let title = '';
   240→  let excerpt = '';
   241→  let metaDescription = '';
   242→  let tags: string[] = [];
   243→  const htmlLines: string[] = [];
   244→
   245→  for (const line of lines) {
   246→    const trimmed = line.trim();
   247→    if (!trimmed) continue;
   248→    if (trimmed.startsWith('عنوان:')) {
   249→      title = trimmed.slice(6).trim().replace(/^["'«]|["'»]$/g, '');
   250→    } else if (trimmed.startsWith('خلاصه:')) {
   251→      excerpt = trimmed.slice(6).trim().replace(/^["'«]|["'»]$/g, '');
   252→    } else if (trimmed.startsWith('متا:')) {
   253→      metaDescription = trimmed.slice(4).trim().replace(/^["'«]|["'»]$/g, '');
   254→    } else if (trimmed.startsWith('برچسب‌ها:')) {
   255→      const rest = trimmed.replace(/^برچسب‌ها:\s*/, '');
   256→      tags = rest
   257→        .split(',')
   258→        .map((t) => t.trim().replace(/^["'«]|["'»]$/g, ''))
   259→        .filter(Boolean)
   260→        .slice(0, 6);
   261→    } else {
   262→      htmlLines.push(trimmed);
   263→    }
   264→  }
   265→
   266→  let html = htmlLines.join('\n').trim();
   267→  if (!html) return null;
   268→
   269→  if (!/<[a-z][\s\S]*>/i.test(html)) {
   270→    html = html
   271→      .split(/\n{2,}/)
   272→      .map((p) => `<p>${p.replace(/\n/g, '<br/>')}</p>`)
   273→      .join('\n');
   274→  }
   275→
   276→  if (!title) {
   277→    const h = html.match(/<h[1-3][^>]*>([\s\S]*?)<\/h[1-3]>/i);
   278→    title = h ? h[1].replace(/<[^>]+>/g, '').trim() : html.slice(0, 60).replace(/<[^>]+>/g, '');
   279→  }
   280→  if (!excerpt) {
   281→    const firstP = html.match(/<p[^>]*>([\s\S]*?)<\/p>/i);
   282→    excerpt = firstP ? firstP[1].replace(/<[^>]+>/g, '').trim().slice(0, 180) : title;
   283→  }
   284→  if (!metaDescription) {
   285→    metaDescription = excerpt.slice(0, 155);
   286→  }
   287→  if (tags.length === 0) {
   288→    tags = ['سفر', 'تاکسی VIP', 'سیوان'];
   289→  }
   290→
   291→  return { title, excerpt, html, tags, metaDescription };
   292→}

   335→async function generateImagePrompt(title: string, topic: Topic): Promise<string | null> {
   336→  if (!zai) return null;
   337→  const styleHint =
   338→    topic.category === 'luxury-cars' || topic.category === 'luxury-vs-economy'
   339→      ? 'professional automotive photography, moody cinematic lighting'
   340→      : topic.category === 'tourism'
   341→        ? 'professional travel photography, golden-hour cinematic lighting'
   342→        : topic.category === 'sivan-brand'
   343→          ? 'professional automotive photography, premium elegant cinematic lighting'
   344→          : 'professional travel photography, cinematic lighting';
   345→  try {
   346→    const completion = await zai.chat.completions.create({
   347→      messages: [
   348→        {
   349→          role: 'user',
   350→          content: `You are a professional art director for a luxury Persian travel & taxi blog. I will give you the title of a Persian article. Produce ONE single English sentence describing a striking, topic-specific cover photo that visually matches the article's ACTUAL subject.
   351→
   352→Rules (very important):
   353→- The image MUST visually depict the specific subject of the article (the exact place, exact car model, or exact concept) — NOT a generic scene.
   354→- Translate any Persian place/car name into its English equivalent so the image model understands it (e.g. "Hormuz Island", "Mercedes-Benz E-Class", "Tehran to Mashhad highway").
   355→- Photorealistic scene description only. No abstract art, no illustrations.
   356→- Absolutely NO text, words, letters, logos or watermark in the image.
   357→- Do NOT show clearly visible human faces.
   358→- Do NOT wrap the answer in quotes or markdown. Do NOT add any explanation or label. Output ONLY the one sentence.
   359→- Keep the sentence between 20 and 80 words.
   360→
   361→Article title (Persian): ${title}
   362→Article focus keyword (Persian): ${topic.keyword}
   363→
   364→End your sentence with exactly: ", ${styleHint}, high quality, no text, no watermark".
   365→
   366→Write the one-sentence English image prompt now:`,
   367→        },
   368→      ],
   369→      thinking: { type: 'disabled' },
   370→    });
   371→    const raw = (completion?.choices?.[0]?.message?.content || '').trim();
   372→    if (!raw) return null;
   373→    const cleaned = raw
   374→      .replace(/^```[a-z]*\n?/i, '')
   375→      .replace(/\n?```$/i, '')
   376→      .replace(/^["'«]|["'»]$/g, '')
   377→      .replace(/^(prompt|image prompt|description|cover photo)\s*:\s*/i, '')
   378→      .replace(/\s+/g, ' ')
   379→      .trim();
   380→    if (cleaned.length < 15 || cleaned.length > 600) return null;
   381→    if (!/no text/i.test(cleaned)) {
   382→      return `${cleaned}, ${styleHint}, high quality, no text, no watermark`;
   383→    }
   384→    return cleaned;
   385→  } catch (err) {
   386→    console.error('[blog-generator] image-prompt LLM call failed:', err);
   387→    return null;
   388→  }

   423→function deriveKeyword(topic: string): string {
   424→  const stopwords = new Set([
   425→    'در', 'از', 'به', 'با', 'و', 'یا', 'را', 'است', 'نیست', 'برای', 'تا', 'که',
   426→    'این', 'آن', 'هر', 'چه', 'چگونه', 'چرا', 'کی', 'کجا', 'کدام', 'هم', 'تاکنون',
   427→    'یک', 'دو', 'سه', 'نه', 'بله', 'خیر', 'اما', 'اگر', 'چون', 'بنابراین', 'پس',
   428→    'ترین', 'ها', 'های', 'ای', 'هایی', 'شود', 'شده', 'می', 'کرد', 'کند', 'باشد',

   430→  ]);
   431→  const words = topic
   432→    .replace(/[«»"'!?.,:;()\[\]{}،؛؟]/g, ' ')

   591→/**
   592→ * On startup, decide when the next generation should happen.
   593→ *
   594→ * Strategy:
   595→ * 1. Restore persisted state (cycleIndex, categoryTopicIndex, lastAutoGenAt)
   596→ * 2. If fewer than MIN_POSTS_FOR_IMMEDIATE_SKIP published posts exist → generate now (after 5s delay)
   597→ * 3. If lastAutoGenAt is recorded in DB and it's been >= 6h since then → generate now
   598→ * 4. If lastAutoGenAt is recorded but it's been < 6h → schedule the remaining time
   599→ * 5. If no lastAutoGenAt but enough posts exist → start a fresh 6h interval
   600→ */
   601→async function scheduleGenerationOnStartup() {
   602→  try {
   603→    const count = await db.blogPost.count({ where: { status: 'published' } });
   604→    console.log(`[blog-generator] current published posts: ${count}`);
   605→
   606→    if (count < MIN_POSTS_FOR_IMMEDIATE_SKIP) {
   607→      console.log(`[blog-generator] only ${count} published posts; generating one now (5s delay)...`);
   608→      setTimeout(() => generateArticle().catch((e) => console.error(e)), 5000);
   609→      return;
   610→    }
   611→
   612→    // Check persisted lastAutoGenAt from DB (restored into lastGeneratedAt)
   613→    if (lastGeneratedAt) {
   614→      const elapsed = Date.now() - new Date(lastGeneratedAt).getTime();
   615→      console.log(`[blog-generator] last auto-gen was ${Math.round(elapsed / 60000)}min ago`);
   616→      if (elapsed >= GENERATION_INTERVAL_MS) {
   617→        // Overdue — generate now
   618→        console.log('[blog-generator] overdue — generating now (5s delay)...');
   619→        setTimeout(() => generateArticle().catch((e) => console.error(e)), 5000);
   620→        return;
   621→      }
   622→      // Not yet due — wait the remaining time
   623→      const remaining = GENERATION_INTERVAL_MS - elapsed;
   624→      console.log(`[blog-generator] next generation in ${Math.round(remaining / 60000)}min (remaining from last auto-gen)`);
   625→      nextGenAt = new Date(Date.now() + remaining);
   626→      setTimeout(() => {
   627→        generateArticle().catch((e) => console.error('[blog-generator] scheduled error:', e));
   628→        // After this first (delayed) generation, set up the recurring interval
   629→        setInterval(() => {
   630→          generateArticle().catch((e) => console.error('[blog-generator] scheduled error:', e));
   631→        }, GENERATION_INTERVAL_MS);
   632→      }, remaining);
   633→      return;
   634→    }
   635→
   636→    // No lastAutoGenAt recorded but we have enough posts (edge case: posts
   637→    // were seeded or created manually, or DB state was lost). Start fresh.
   638→    console.log(`[blog-generator] no previous auto-gen recorded; starting fresh ${GENERATION_INTERVAL_MS / 3600000}h interval`);
   639→    nextGenAt = new Date(Date.now() + GENERATION_INTERVAL_MS);
   640→    setInterval(() => {
   641→      generateArticle().catch((e) => console.error('[blog-generator] scheduled error:', e));
   642→    }, GENERATION_INTERVAL_MS);
   643→  } catch (err) {
   644→    console.error('[blog-generator] startup scheduling failed:', err);
   645→    // Fallback: just start a fresh interval
   646→    setInterval(() => {
   647→      generateArticle().catch((e) => console.error('[blog-generator] fallback scheduled error:', e));
   648→    }, GENERATION_INTERVAL_MS);
   649→  }
   650→}
