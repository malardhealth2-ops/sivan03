   174→function makeSlug(title: string): string {
   175→  // Persian-safe slug: keep Persian letters/digits, replace spaces with dashes,
   176→  // strip punctuation, append a short unique suffix.
   177→  const base = title
   178→    .replace(/[«»"'!?.,:;()\[\]{}]/g, '')
   179→    .replace(/\s+/g, '-')
   180→    .replace(/-+/g, '-')
   181→    .trim()
   182→    .slice(0, 60);
   183→  const suffix = Date.now().toString(36).slice(-6);
   184→  return `${base}-${suffix}`;
   185→}

   187→function parseArticle(raw: string): {
   188→  title: string;
   189→  excerpt: string;
   190→  html: string;
   191→  tags: string[];
   192→  metaDescription: string;
   193→} | null {
   194→  if (!raw) return null;
   195→  const lines = raw.split('\n');
   196→
   197→  let title = '';
   198→  let excerpt = '';
   199→  let metaDescription = '';
   200→  let tags: string[] = [];
   201→  const htmlLines: string[] = [];
   202→
   203→  for (const line of lines) {
   204→    const trimmed = line.trim();
   205→    if (!trimmed) continue;
   206→    if (trimmed.startsWith('عنوان:')) {
   207→      title = trimmed.slice(6).trim().replace(/^["'«]|["'»]$/g, '');
   208→    } else if (trimmed.startsWith('خلاصه:')) {
   209→      excerpt = trimmed.slice(6).trim().replace(/^["'«]|["'»]$/g, '');
   210→    } else if (trimmed.startsWith('متا:')) {
   211→      metaDescription = trimmed.slice(4).trim().replace(/^["'«]|["'»]$/g, '');
   212→    } else if (trimmed.startsWith('برچسب‌ها:')) {
   213→      const rest = trimmed.replace(/^برچسب‌ها:\s*/, '');
   214→      tags = rest
   215→        .split(',')
   216→        .map((t) => t.trim().replace(/^["'«]|["'»]$/g, ''))
   217→        .filter(Boolean)
   218→        .slice(0, 6);
   219→    } else {
   220→      htmlLines.push(trimmed);
   221→    }
   222→  }
   223→
   224→  // Reconstruct content HTML.
   225→  let html = htmlLines.join('\n').trim();
   226→  if (!html) return null;
   227→
   228→  // If the model didn't use HTML tags, wrap lines in <p>.
   229→  if (!/<[a-z][\s\S]*>/i.test(html)) {
   230→    html = html
   231→      .split(/\n{2,}/)
   232→      .map((p) => `<p>${p.replace(/\n/g, '<br/>')}</p>`)
   233→      .join('\n');
   234→  }
   235→
   236→  if (!title) {
   237→    const h = html.match(/<h[1-3][^>]*>([\s\S]*?)<\/h[1-3]>/i);
   238→    title = h ? h[1].replace(/<[^>]+>/g, '').trim() : html.slice(0, 60).replace(/<[^>]+>/g, '');
   239→  }
   240→  if (!excerpt) {
   241→    const firstP = html.match(/<p[^>]*>([\s\S]*?)<\/p>/i);
   242→    excerpt = firstP ? firstP[1].replace(/<[^>]+>/g, '').trim().slice(0, 180) : title;
   243→  }
   244→  if (!metaDescription) {
   245→    metaDescription = excerpt.slice(0, 155);
   246→  }
   247→  if (tags.length === 0) {
   248→    tags = ['سفر', 'تاکسی VIP', 'سیوان'];
   249→  }
   250→
   251→  return { title, excerpt, html, tags, metaDescription };
   252→}

   297→async function generateCoverImage(
   298→  title: string,
   299→  slug: string,
   300→  topic: Topic
   301→): Promise<string | null> {
   302→  if (!zai) return null;
   303→  const scenes = COVER_SCENES[topic.category] || COVER_SCENES['travel-guide'];
   304→  const scene = scenes[Math.floor(Math.random() * scenes.length)];
   305→  try {
   306→    const response = await zai.images.generations.create({
   307→      prompt: scene,
   308→      size: '1344x768',
   309→    });
   310→    const base64 = response?.data?.[0]?.base64;
   311→    if (!base64) return null;
   312→    if (!fs.existsSync(BLOG_IMG_DIR)) fs.mkdirSync(BLOG_IMG_DIR, { recursive: true });
   313→    const filename = `${slug}.png`;
   314→    const filepath = path.join(BLOG_IMG_DIR, filename);
   315→    fs.writeFileSync(filepath, Buffer.from(base64, 'base64'));
   316→    return `/images/blog/${filename}`;
   317→  } catch (err) {
   318→    console.error('[blog-generator] image generation failed:', err);
   319→    // Fallback to an existing default car image so the post isn't imageless.
   320→    return '/images/luxury-car.png';
   321→  }
   322→}

   325→// words (skip very common Persian stopwords), join with space. Falls back to
   326→// the whole topic if filtering removes everything.
   327→function deriveKeyword(topic: string): string {
   328→  const stopwords = new Set([
   329→    'در', 'از', 'به', 'با', 'و', 'یا', 'را', 'است', 'نیست', 'برای', 'تا', 'که',
   330→    'این', 'آن', 'هر', 'چه', 'چگونه', 'چرا', 'کی', 'کجا', 'کدام', 'هم', 'تاکنون',
   331→    'یک', 'دو', 'سه', 'نه', 'بله', 'خیر', 'اما', 'اگر', 'چون', 'بنابراین', 'پس',
   332→    'ترین', 'ها', 'های', 'ای', 'هایی', 'شود', 'شده', 'می', 'کرد', 'کند', 'باشد',
   333→    'هستند', 'بود', 'بودن', 'ما', 'شما', 'او', 'آنها', 'خود', 'همین', 'مانند',
   334→  ]);
   335→  const words = topic
   336→    .replace(/[«»"'!?.,:;()\[\]{}،؛؟]/g, ' ')
   337→    .split(/\s+/)
   338→    .filter((w) => w.length > 1 && !stopwords.has(w));
   339→  const kw = words.slice(0, 5).join(' ').trim();
   340→  return kw || topic.slice(0, 40);
   341→}

   343→// Heuristically classify a free-form custom topic into one of the cover-image

   358→    return 'travel-tips';
   359→  }
   360→  if (/(امني|ایمنی|بي خطر|بی‌خطر|خطر|تصادف|كاكسه|کاسه|كمربند|کمربند| ايمني سفر|ایمنی سفر|شب|چك ليست|چک‌لیست|چک ليست)/.test(t)) {
   361→    return 'safety';
   362→  }
   363→  if (/(سيوان|سرويس دربستي|سرویس دربستی|آژانس|هزينه تاكسيمون|هزینه تاکسی|قيمت|قیمت|رزرو|اپليكيشن|اپلیکیشن|برند سيوان|برند سوان)/.test(t)) {
   364→    return 'sivan-brand';
   365→  }
   366→  return 'travel-guide';

   369→async function generateArticle(customTopic?: string): Promise<{ ok: boolean; title?: string; slug?: string; error?: string; custom?: boolean }> {
   370→  if (!zai) {
   371→    return { ok: false, error: 'ZAI SDK not initialized' };
   372→  }
   373→  if (isGenerating) {
   374→    return { ok: false, error: 'Generation already in progress' };
   375→  }
   376→  isGenerating = true;
   377→  lastError = null;
   378→
   379→  // For custom-topic generations we build a Topic from the admin's input and
   380→  // do NOT call pickTopic() — this keeps the 6-hour auto-rotation untouched
   381→  // (cycleIndex / categoryTopicIndex are only advanced by scheduled/on-demand
   382→  // auto-pick generations, never by custom ones).
   383→  const isCustom = !!customTopic && customTopic.trim().length > 0;
   384→  const topic: Topic = isCustom
   385→    ? {
   386→        title: customTopic!.trim().slice(0, 200),
   387→        keyword: deriveKeyword(customTopic!),
   388→        category: classifyTopic(customTopic!),
   389→      }
   390→    : pickTopic();
   391→  console.log(`[blog-generator] generating article for topic: ${topic.title} (keyword: ${topic.keyword}, category: ${topic.category}${isCustom ? ', CUSTOM' : ''})`);

   405→

   467→      data: {
   468→        title: parsed.title,
   469→        slug,
   470→        excerpt: parsed.excerpt,

   482→    isGenerating = false;
   483→    return { ok: true, title: post.title, slug: post.slug, custom: isCustom };

   486→    lastError = 'DB save failed';
   487→    isGenerating = false;
   488→    return { ok: false, error: 'DB save failed' };

   492→async function maybeGenerateOnStartup() {
   493→  try {
   494→    const count = await db.blogPost.count({ where: { status: 'published' } });
   495→    if (count < MIN_POSTS_FOR_IMMEDIATE_SKIP) {
   496→      console.log(`[blog-generator] only ${count} published posts; generating one now...`);
   497→      setTimeout(() => generateArticle().catch((e) => console.error(e)), 5000);
   498→      return;
   499→    }
   500→    // If the most recent post is older than 6h, generate one now.
   501→    const latest = await db.blogPost.findFirst({
   502→      where: { status: 'published' },
   503→      orderBy: { publishedAt: 'desc' },
   504→      select: { publishedAt: true },
   505→    });
   506→    if (latest?.publishedAt) {
   507→      const ageMs = Date.now() - new Date(latest.publishedAt).getTime();
   508→      if (ageMs > GENERATION_INTERVAL_MS) {
   509→        console.log(`[blog-generator] last post is ${Math.round(ageMs / 3600000)}h old; generating one now...`);
   510→        setTimeout(() => generateArticle().catch((e) => console.error(e)), 5000);
   511→      } else {
   512→        console.log(`[blog-generator] last post is recent; next auto-generation in 6h.`);
   513→      }
   514→    }
   515→  } catch (err) {
   516→    console.error('[blog-generator] startup check failed:', err);
   517→  }
   518→}

   539→          'Access-Control-Allow-Origin': '*',
   540→          'Access-Control-Allow-Methods': 'GET, POST, OPTIONS',
   541→          'Access-Control-Allow-Headers': 'Content-Type',
   542→        },
   543→      });
   544→    }
   545→
   546→    const corsHeaders = {
   547→      'Access-Control-Allow-Origin': '*',
   548→      'Access-Control-Allow-Methods': 'GET, POST, OPTIONS',
   549→      'Access-Control-Allow-Headers': 'Content-Type',
   550→    };
   551→
   552→    if (req.method === 'GET' && (url.pathname === '/health' || url.pathname === '/status')) {
   553→      const totalPosts = await getTotalPosts();

   565→        { headers: corsHeaders }

   570→      // (slow) LLM + image generation. Admin UI can poll /status afterwards.

   578→      // Custom-topic generation. Reads { topic: string } from the JSON body.
   579→      // This does NOT advance the 6-hour auto-rotation — generateArticle()
   580→      // skips pickTopic() when a customTopic is supplied, so the scheduled

   587→        body = await req.json();
   588→      } catch {
   589→        return Response.json({ started: false, message: 'بدنه درخواست نامعتبر است' }, { status: 400, headers: corsHeaders });
   590→      }
   591→      const topic = typeof body?.topic === 'string' ? body.topic.trim() : '';
   592→      if (!topic || topic.length < 3) {
   593→        return Response.json({ started: false, message: 'موضوع مقاله باید حداقل ۳ کاراکتر باشد' }, { status: 400, headers: corsHeaders });
   594→      }

   597→      }

   600→      return Response.json({ started: true, custom: true, topic }, { headers: corsHeaders });
   601→    }
   602→    return new Response('Not Found', { status: 404, headers: corsHeaders });
   603→  },
