     2→     2→

     4→     4→import { motion, AnimatePresence } from 'framer-motion';
     5→     5→import {
     6→     6→  User,

    20→    20→import {

    22→    22→  DialogContent,
    23→    23→  DialogHeader,

    25→    25→  DialogDescription,
    26→    26→} from '@/components/ui/dialog';
    27→    27→import { Button } from '@/components/ui/button';
    28→    28→import { Input } from '@/components/ui/input';
    29→    29→import { InputOTP, InputOTPGroup, InputOTPSlot, InputOTPSeparator } from '@/components/ui/input-otp';

    32→    32→import { toast } from 'sonner';
    33→    33→
    34→    34→// ─── OTP Step tracking ─────────────────────────────────────────────
    35→    35→
    36→    36→type OTPStep = 'phone' | 'code' | 'register' | 'success';
    37→    37→
    38→    38→export function AuthModal() {
    39→    39→  const {
    40→    40→    auth,
    41→    41→    closeAuth,
    42→    42→    setAuthUsername,
    43→    43→    setAuthFullName,
    44→    44→    setAuthPhone,
    45→    45→    setAuthUser,
    46→    46→    setAuthVerified,
    47→    47→    adminLogin,
    48→    48→    setAdminOpen,
    49→    49→    openAuth,
    50→    50→    setUserPanelOpen,
    51→    51→  } = useAppStore();
    52→    52→
    53→    53→  const [step, setStep] = useState<OTPStep>('phone');
    54→    54→  const [loading, setLoading] = useState(false);
    55→    55→  const [localError, setLocalError] = useState('');
    56→    56→  const [otpSent, setOtpSent] = useState(false);
    57→    57→  const [isDemo, setIsDemo] = useState(false);
    58→    58→  const [demoOtp, setDemoOtp] = useState('');
    59→    59→  const [countdown, setCountdown] = useState(0);
    60→    60→  const [isExistingUser, setIsExistingUser] = useState(false);
    61→    61→  const [existingUser, setExistingUser] = useState<{ id: string; fullName: string; username: string; role: string } | null>(null);
    62→    62→  const countdownRef = useRef<ReturnType<typeof setInterval>>();
    63→    63→  const inputRef = useRef<HTMLInputElement>(null);
    64→    64→
    65→    65→  // Reset when modal opens/closes
    66→    66→  useEffect(() => {
    67→    67→    if (auth.isOpen) {
    68→    68→      setStep('phone');
    69→    69→      setLocalError('');
    70→    70→      setOtpSent(false);
    71→    71→      setIsDemo(false);
    72→    72→      setDemoOtp('');
    73→    73→      setCountdown(0);
    74→    74→      setIsExistingUser(false);
    75→    75→      setExistingUser(null);
    76→    76→      // Focus phone input after mount
    77→    77→      setTimeout(() => inputRef.current?.focus(), 200);
    78→    78→    }
    79→    79→  }, [auth.isOpen]);
    80→    80→
    81→    81→  // Countdown timer
    82→    82→  useEffect(() => {
    83→    83→    if (countdown > 0) {
    84→    84→      countdownRef.current = setInterval(() => {
    85→    85→        setCountdown((c) => {
    86→    86→          if (c <= 1) {
    87→    87→            clearInterval(countdownRef.current);
    88→    88→            return 0;
    89→    89→          }
    90→    90→          return c - 1;
    91→    91→        });
    92→    92→      }, 1000);
    93→    93→      return () => clearInterval(countdownRef.current);
    94→    94→    }
    95→    95→  }, [countdown]);
    96→    96→
    97→    97→  // ─── Step 1: Send OTP ──────────────────────────────────────
    98→    98→
    99→    99→  const handleSendOTP = async () => {
   100→   100→    const phone = auth.phone.trim();
   101→   101→    if (!phone || !/^09[0-9]{9}$/.test(phone)) {
   102→   102→      setLocalError('شماره موبایل معتبر وارد کنید (مثال: ۰۹۱۲۳۴۵۶۷۸۹)');
   103→   103→      return;
   104→   104→    }
   105→   105→    setLocalError('');
   106→   106→    setLoading(true);
   107→   107→
   108→   108→    try {
   109→   109→      const res = await fetch('/api/auth/send-otp', {
   110→   110→        method: 'POST',
   111→   111→        headers: { 'Content-Type': 'application/json' },
   112→   112→        body: JSON.stringify({ phone }),
   113→   113→      });
   114→   114→
   115→   115→      const data = await res.json();
   116→   116→
   117→   117→      if (!res.ok) {
   118→   118→        setLocalError(data.error || 'خطا در ارسال کد تأیید');
   119→   119→        setLoading(false);
   120→   120→        return;
   121→   121→      }
   122→   122→
   123→   123→      setOtpSent(true);
   124→   124→      setIsDemo(!!data.isDemo);
   125→   125→      if (data.isDemo && data.otp) {
   126→   126→        setDemoOtp(data.otp);
   127→   127→      }
   128→   128→      setCountdown(120);
   129→   129→      setStep('code');
   130→   130→      toast.success('کد تأیید ارسال شد');
   131→   131→    } catch {
   132→   132→      setLocalError('خطا در ارتباط با سرور');
   133→   133→    } finally {
   134→   134→      setLoading(false);
   135→   135→    }
   136→   136→  };
   137→   137→
   138→   138→  // ─── Step 2: Verify OTP ─────────────────────────────────────
   139→   139→
   140→   140→  const handleVerifyOTP = async (code: string) => {
   141→   141→    if (!code || code.length !== 6) return;
   142→   142→    setLocalError('');
   143→   143→    setLoading(true);
   144→   144→
   145→   145→    try {
   146→   146→      const res = await fetch('/api/auth/verify-otp', {
   147→   147→        method: 'POST',
   148→   148→        headers: { 'Content-Type': 'application/json' },
   149→   149→        body: JSON.stringify({ phone: auth.phone, code }),
   150→   150→      });
   151→   151→
   152→   152→      const data = await res.json();
   153→   153→
   154→   154→      if (!res.ok) {
   155→   155→        setLocalError(data.error || 'کد تأیید نامعتبر');
   156→   156→        setLoading(false);
   157→   157→        return;
   158→   158→      }
   159→   159→
   160→   160→      if (data.isNewUser) {
   161→   161→        // New user — go to registration step
   162→   162→        setStep('register');
   163→   163→        setLoading(false);
   164→   164→      } else {
   165→   165→        // Existing user — login directly
   166→   166→        setIsExistingUser(true);
   167→   167→        setExistingUser(data.user);
   168→   168→        setAuthUser({
   169→   169→          id: data.user.id,
   170→   170→          username: data.user.username || data.user.phone,
   171→   171→          fullName: data.user.fullName,
   172→   172→          role: data.user.role,
   173→   173→        });
   174→   174→        setAuthVerified(true);
   175→   175→        setStep('success');
   176→   176→
   177→   177→        // Admin handling
   178→   178→        if (data.user?.role === 'admin') {
   179→   179→          adminLogin(data.user.username, '');
   180→   180→          setTimeout(() => {
   181→   181→            setAdminOpen(true);
   182→   182→            closeAuth();
   183→   183→          }, 1200);
   184→   184→        } else {
   185→   185→          setTimeout(() => {
   186→   186→            closeAuth();
   187→   187→            setUserPanelOpen(true);
   188→   188→          }, 1200);
   189→   189→        }
   190→   190→
   191→   191→        toast.success(`خوش آمدید، ${data.user?.fullName || 'کاربر'} عزیز`);
   192→   192→      }
   193→   193→    } catch {
   194→   194→      setLocalError('خطا در ارتباط با سرور');
   195→   195→    } finally {
   196→   196→      setLoading(false);
   197→   197→    }
   198→   198→  };
   199→   199→
   200→   200→  // ─── Step 3: Complete Registration ──────────────────────────
   201→   201→
   202→   202→  const handleCompleteRegistration = async () => {
   203→   203→    const fullName = auth.fullName.trim();
   204→   204→    if (!fullName || fullName.length < 2) {
   205→   205→      setLocalError('نام و نام خانوادگی را کامل وارد کنید');
   206→   206→      return;
   207→   207→    }
   208→   208→    setLocalError('');
   209→   209→    setLoading(true);
   210→   210→
   211→   211→    try {
   212→   212→      const res = await fetch('/api/auth/register', {
   213→   213→        method: 'POST',
   214→   214→        headers: { 'Content-Type': 'application/json' },
   215→   215→        body: JSON.stringify({
   216→   216→          username: `user_${auth.phone}`,
   217→   217→          fullName,
   218→   218→          password: `pass_${Date.now()}`, // Auto-generated since auth is OTP-based
   219→   219→          phone: auth.phone,
   220→   220→        }),
   221→   221→      });
   222→   222→
   223→   223→      const data = await res.json();
   224→   224→
   225→   225→      if (!res.ok) {
   226→   226→        setLocalError(data.error || 'خطا در ثبت‌نام');
   227→   227→        setLoading(false);
   228→   228→        return;
   229→   229→      }
   230→   230→
   231→   231→      setAuthUser({
   232→   232→        id: data.user.id,
   233→   233→        username: data.user.username,
   234→   234→        fullName: data.user.fullName,
   235→   235→        role: data.user.role,
   236→   236→      });
   237→   237→      setAuthVerified(true);
   238→   238→      setStep('success');
   239→   239→
   240→   240→      toast.success(`ثبت‌نام موفق! خوش آمدید، ${data.user.fullName} عزیز`);
   241→   241→      setTimeout(() => {
   242→   242→        closeAuth();
   243→   243→        setUserPanelOpen(true);
   244→   244→      }, 1200);
   245→   245→    } catch {
   246→   246→      setLocalError('خطا در ارتباط با سرور');
   247→   247→    } finally {
   248→   248→      setLoading(false);
   249→   249→    }
   250→   250→  };
   251→   251→
   252→   252→  // ─── Legacy login via username/password ─────────────────────
   253→   253→
   254→   254→  const [legacyMode, setLegacyMode] = useState(false);
   255→   255→  const [showPass, setShowPass] = useState(false);
   256→   256→
   257→   257→  const handleLegacyLogin = async () => {
   258→   258→    if (!auth.username || !auth.password) {
   259→   259→      setLocalError('نام کاربری و رمز عبور را وارد کنید');
   260→   260→      return;
   261→   261→    }
   262→   262→    setLocalError('');
   263→   263→    setLoading(true);
   264→   264→
   265→   265→    try {
   266→   266→      const res = await fetch('/api/auth/login', {
   267→   267→        method: 'POST',
   268→   268→        headers: { 'Content-Type': 'application/json' },
   269→   269→        body: JSON.stringify({ username: auth.username, password: auth.password }),
   270→   270→      });
   271→   271→
   272→   272→      const data = await res.json();
   273→   273→
   274→   274→      if (!res.ok) {
   275→   275→        setLocalError(data.error || 'خطا در ورود');
   276→   276→        setLoading(false);
   277→   277→        return;
   278→   278→      }
   279→   279→
   280→   280→      setAuthUser(data.user || null);
   281→   281→
   282→   282→      if (data.user?.role === 'admin') {
   283→   283→        adminLogin(auth.username, auth.password);
   284→   284→        setAdminOpen(true);
   285→   285→        closeAuth();
   286→   286→        setLoading(false);
   287→   287→        return;
   288→   288→      }
   289→   289→
   290→   290→      setAuthVerified(true);
   291→   291→      setLoading(false);
   292→   292→      toast.success(`خوش آمدید، ${data.user?.fullName || 'کاربر'} عزیز`);
   293→   293→      setTimeout(() => {
   294→   294→        closeAuth();
   295→   295→        setUserPanelOpen(true);
   296→   296→      }, 900);
   297→   297→    } catch {
   298→   298→      setLocalError('خطا در ارتباط با سرور');
   299→   299→      setLoading(false);
   300→   300→    }
   301→   301→  };
   302→   302→
   303→   303→  const resetAndClose = () => {
   304→   304→    setLocalError('');
   305→   305→    closeAuth();
   306→   306→  };
   307→   307→
   308→   308→  const formatCountdown = (s: number) => {
   309→   309→    const min = Math.floor(s / 60);
   310→   310→    const sec = s % 60;
   311→   311→    return `${min}:${sec.toString().padStart(2, '0')}`;
   312→   312→  };
   313→   313→
   314→   314→  return (
   315→   315→    <Dialog open={auth.isOpen} onOpenChange={(val) => { if (!val) resetAndClose(); }}>
   316→   316→      <DialogContent className="bg-[#1a1a1a] border-[#333] max-w-md">
   317→   317→        <DialogHeader>
   318→   318→          <DialogTitle className="text-right text-[#fafafa]">
   319→   319→            {step === 'success'
   320→   320→              ? 'خوش آمدید!'
   321→   321→              : step === 'register'
   322→   322→                ? 'تکمیل ثبت‌نام'
   323→   323→                : step === 'code'
   324→   324→                  ? 'تأیید شماره موبایل'
   325→   325→                  : legacyMode
   326→   326→                    ? 'ورود با نام کاربری'
   327→   327→                    : 'ورود / ثبت‌نام'}
   328→   328→          </DialogTitle>
   329→   329→          <DialogDescription className="text-right text-[#a1a1aa]">
   330→   330→            {step === 'success'
   331→   331→              ? 'حساب شما با موفقیت تأیید شد'
   332→   332→              : step === 'register'
   333→   333→                ? 'برای تکمیل ثبت‌نام، نام خود را وارد کنید'
   334→   334→                : step === 'code'
   335→   335→                  ? `کد تأیید ارسال شده به ${auth.phone} را وارد کنید`
   336→   336→                  : legacyMode
   337→   337→                    ? 'نام کاربری و رمز عبور خود را وارد کنید'
   338→   338→                    : 'شماره موبایل خود را وارد کنید تا کد تأیید ارسال شود'}
   339→   339→          </DialogDescription>
   340→   340→        </DialogHeader>
   341→   341→
   342→   342→        <div className="mt-2">
   343→   343→          <AnimatePresence mode="wait">
   344→   344→            {/* ─── Success State ─────────────────────────── */}
   345→   345→            {step === 'success' && (
   346→   346→              <motion.div
   347→   347→                key="success"
   348→   348→                initial={{ opacity: 0, scale: 0.9 }}
   349→   349→                animate={{ opacity: 1, scale: 1 }}
   350→   350→                className="text-center py-6 space-y-4"
   351→   351→              >
   352→   352→                <div className="w-16 h-16 rounded-full bg-green-500/20 flex items-center justify-center mx-auto">
   353→   353→                  <Check className="h-8 w-8 text-green-400" />
   354→   354→                </div>
   355→   355→                <p className="text-[#fafafa] font-medium">
   356→   356→                  {auth.user?.fullName ? `${auth.user.fullName} عزیز، خوش آمدید!` : 'خوش آمدید!'}
   357→   357→                </p>
   358→   358→                <p className="text-[#a1a1aa] text-xs">در حال ورود به پنل کاربری...</p>
   359→   359→              </motion.div>
   360→   360→            )}
   361→   361→
   362→   362→            {/* ─── Phone Entry ───────────────────────────── */}
   363→   363→            {step === 'phone' && !legacyMode && (
   364→   364→              <motion.div
   365→   365→                key="phone"
   366→   366→                initial={{ opacity: 0, x: -20 }}
   367→   367→                animate={{ opacity: 1, x: 0 }}
   368→   368→                exit={{ opacity: 0, x: 20 }}
   369→   369→                className="space-y-4"
   370→   370→              >
   371→   371→                <div className="flex justify-center">
   372→   372→                  <div className="w-16 h-16 rounded-full bg-[#D4AF37]/10 flex items-center justify-center">
   373→   373→                    <Smartphone className="h-8 w-8 text-[#D4AF37]" />
   374→   374→                  </div>
   375→   375→                </div>
   376→   376→
   377→   377→                <div className="space-y-2">
   378→   378→                  <Label className="text-[#fafafa] text-sm">
   379→   379→                    <Phone className="h-3.5 w-3.5 ml-1.5 text-[#D4AF37] inline" />
   380→   380→                    شماره موبایل
   381→   381→                  </Label>
   382→   382→                  <Input
   383→   383→                    ref={inputRef}
   384→   384→                    type="tel"
   385→   385→                    placeholder="۰۹۱۲۳۴۵۶۷۸۹"
   386→   386→                    value={auth.phone}
   387→   387→                    onChange={(e) => {
   388→   388→                      const val = e.target.value.replace(/[^0-9]/g, '');
   389→   389→                      setAuthPhone(val);
   390→   390→                      setLocalError('');
   391→   391→                    }}
   392→   392→                    onKeyDown={(e) => e.key === 'Enter' && handleSendOTP()}
   393→   393→                    className="bg-[#0a0a0a] border-[#333] text-[#fafafa] placeholder:text-[#888] h-12 text-lg tracking-wider"
   394→   394→                    dir="ltr"
   395→   395→                    autoFocus
   396→   396→                    maxLength={11}
   397→   397→                  />
   398→   398→                  <p className="text-[10px] text-[#666]">
   399→   399→                    کد تأیید به این شماره ارسال می‌شود
   400→   400→                  </p>
   401→   401→                </div>
   402→   402→
   403→   403→                {localError && (
   404→   404→                  <motion.div
   405→   405→                    initial={{ opacity: 0, y: -5 }}
   406→   406→                    animate={{ opacity: 1, y: 0 }}
   407→   407→                    className="flex items-center gap-2 p-3 rounded-lg bg-red-500/10 border border-red-500/20 text-red-400 text-sm"
   408→   408→                  >
   409→   409→                    <AlertTriangle className="h-4 w-4 shrink-0" />
   410→   410→                    {localError}
   411→   411→                  </motion.div>
   412→   412→                )}
   413→   413→
   414→   414→                <Button
   415→   415→                  onClick={handleSendOTP}
   416→   416→                  disabled={loading || auth.phone.length !== 11}
   417→   417→                  className="w-full bg-[#D4AF37] text-[#0a0a0a] hover:bg-[#E5C76B] h-12 font-medium"
   418→   418→                >
   419→   419→                  {loading ? (
   420→   420→                    <Loader2 className="h-4 w-4 animate-spin" />
   421→   421→                  ) : (
   422→   422→                    <>
   423→   423→                      <Fingerprint className="h-4 w-4 ml-2" />
   424→   424→                      ارسال کد تأیید
   425→   425→                    </>
   426→   426→                  )}
   427→   427→                </Button>
   428→   428→
   429→   429→                {/* Switch to legacy login */}
   430→   430→                <div className="text-center text-xs text-[#a1a1aa] pt-1">
   431→   431→                  ورود با نام کاربری؟{' '}
   432→   432→                  <button
   433→   433→                    type="button"
   434→   434→                    onClick={() => { setLegacyMode(true); setLocalError(''); }}
   435→   435→                    className="text-[#D4AF37] hover:text-[#E5C76B] font-medium underline-offset-2 hover:underline"
   436→   436→                  >
   437→   437→                    کلیک کنید
   438→   438→                  </button>
   439→   439→                </div>
   440→   440→              </motion.div>
   441→   441→            )}
   442→   442→
   443→   443→            {/* ─── OTP Code Entry ─────────────────────────── */}
   444→   444→            {step === 'code' && (
   445→   445→              <motion.div
   446→   446→                key="code"
   447→   447→                initial={{ opacity: 0, x: 20 }}
   448→   448→                animate={{ opacity: 1, x: 0 }}
   449→   449→                exit={{ opacity: 0, x: -20 }}
   450→   450→                className="space-y-4"
   451→   451→              >
   452→   452→                <div className="flex justify-center">
   453→   453→                  <div className="w-16 h-16 rounded-full bg-[#D4AF37]/10 flex items-center justify-center">
   454→   454→                    <KeyRound className="h-8 w-8 text-[#D4AF37]" />
   455→   455→                  </div>
   456→   456→                </div>
   457→   457→
   458→   458→                <div className="flex flex-col items-center gap-3">
   459→   459→                  <Label className="text-[#fafafa] text-sm text-center">
   460→   460→                    کد تأیید ۶ رقمی
   461→   461→                  </Label>
   462→   462→                  <InputOTP
   463→   463→                    maxLength={6}
   464→   464→                    onComplete={(code) => handleVerifyOTP(code)}
   465→   465→                    disabled={loading}
   466→   466→                    containerClassName="justify-center"
   467→   467→                    dir="ltr"
   468→   468→                  >
   469→   469→                    <InputOTPGroup>
   470→   470→                      <InputOTPSlot index={0} className="bg-[#0a0a0a] border-[#333] text-[#fafafa] h-12 w-12" />
   471→   471→                      <InputOTPSlot index={1} className="bg-[#0a0a0a] border-[#333] text-[#fafafa] h-12 w-12" />
   472→   472→                      <InputOTPSlot index={2} className="bg-[#0a0a0a] border-[#333] text-[#fafafa] h-12 w-12" />
   473→   473→                    </InputOTPGroup>
