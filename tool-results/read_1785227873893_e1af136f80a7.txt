   136→// ─── Invalidate Map Size Handler ─────────────────────────────────────────────
   137→
   138→function MapInvalidator() {
   139→  const map = useMap();
   140→
   141→  useEffect(() => {
   142→    // Force Leaflet to recalculate its size after the container is fully rendered
   143→    const timer = setTimeout(() => {
   144→      map.invalidateSize();
   145→    }, 200);
   146→    return () => clearTimeout(timer);
   147→  }, [map]);
   148→
   149→  // Also listen for scroll/resize events that might change the container
   150→  useEffect(() => {
   151→    let frameId: number;
   152→    const handleUpdate = () => {
   153→      if (frameId) cancelAnimationFrame(frameId);
   154→      frameId = requestAnimationFrame(() => {
   155→        map.invalidateSize({ animate: false });
   156→      });
   157→    };
   158→
   159→    window.addEventListener('resize', handleUpdate);
   160→    const observer = new IntersectionObserver((entries) => {
   161→      for (const entry of entries) {
   162→        if (entry.isIntersecting) {
   163→          setTimeout(() => map.invalidateSize({ animate: false }), 100);
   164→        }
   165→      }
   166→    }, { threshold: 0.1 });
   167→
   168→    const container = map.getContainer();
   169→    if (container) observer.observe(container);
   170→
   171→    return () => {
   172→      window.removeEventListener('resize', handleUpdate);
   173→      observer.disconnect();
   174→      if (frameId) cancelAnimationFrame(frameId);
   175→    };
   176→  }, [map]);
   177→
   178→  return null;
   179→}

   181→// ─── Fit to Bounds Handler ──────────────────────────────────────────────────
   182→
   183→function FitBoundsHandler({
   184→  points,
   185→  routeData,
   186→}: {
   187→  points: { origin: MapPoint | null; destination: MapPoint | null };
   188→  routeData: RouteResponse | null;
   189→}) {
   190→  const map = useMap();
   191→
   192→  useEffect(() => {
   193→    if (routeData && routeData.routes.length > 0 && routeData.routes[0].path.length > 0) {
   194→      const polyline = L.polyline(routeData.routes[0].path);
   195→      const bounds = polyline.getBounds().pad(0.15);
   196→      map.flyToBounds(bounds, { duration: 1.2 });
   197→    } else if (points.origin && points.destination) {
   198→      const bounds = L.latLngBounds(
   199→        [points.origin.lat, points.origin.lng],
   200→        [points.destination.lat, points.destination.lng]
   201→      ).pad(0.3);
   202→      map.flyToBounds(bounds, { duration: 1 });
   203→    } else if (points.origin) {
   204→      map.flyTo([points.origin.lat, points.origin.lng], 12, { duration: 1 });
   205→    }
   206→  }, [points, routeData, map]);
   207→
   208→  return null;
   209→}

   419→export default function InteractiveMapInner() {
   420→  const [origin, setOrigin] = useState<MapPoint | null>(null);
   421→  const [destination, setDestination] = useState<MapPoint | null>(null);
   422→  const [selectionStep, setSelectionStep] = useState<SelectionStep>('origin');
   423→  const [routeData, setRouteData] = useState<RouteResponse | null>(null);
   424→  const [routeLoading, setRouteLoading] = useState(false);
   425→  const [routeError, setRouteError] = useState('');
   426→  const [activeRoute, setActiveRoute] = useState(0);
   427→
   428→  // Search state
   429→  const [originSearch, setOriginSearch] = useState('');
   430→  const [destSearch, setDestSearch] = useState('');
   431→  const [originResults, setOriginResults] = useState<
   432→    { display_name: string; lat: string; lon: string }[]
   433→  >([]);
   434→  const [destResults, setDestResults] = useState<
   435→    { display_name: string; lat: string; lon: string }[]
   436→  >([]);
   437→  const [searchLoading, setSearchLoading] = useState<'origin' | 'dest' | null>(null);
   438→  const searchTimer = useRef<ReturnType<typeof setTimeout>>();
   439→  const [identifyingPlace, setIdentifyingPlace] = useState<'origin' | 'dest' | null>(null);
   440→
   441→  // Identify place name using VLM when clicking on map
   442→  const identifyPlace = useCallback(async (lat: number, lng: number, type: 'origin' | 'dest') => {
   443→    setIdentifyingPlace(type);
   444→    try {
   445→      const res = await fetch(`/api/map/identify-place?lat=${lat}&lng=${lng}`);
   446→      if (res.ok) {
   447→        const data = await res.json();
   448→        const name = data.name || '';
   449→        if (name) {
   450→          if (type === 'origin') {
   451→            setOriginSearch(name);
   452→            setOrigin(prev => prev ? { ...prev, name } : null);
   453→          } else {
   454→            setDestSearch(name);
   455→            setDestination(prev => prev ? { ...prev, name } : null);
   456→          }
   457→        }
   458→      }
   459→    } catch {
   460→      // Silently fail - the route API will also try to get place names
   461→    } finally {
   462→      setIdentifyingPlace(null);
   463→    }
   464→  }, []);
   465→
   466→  // Handle map click
   467→  const handleSelectPoint = useCallback(
   468→    (lat: number, lng: number) => {
   469→      if (selectionStep === 'origin') {
   470→        setOrigin({ lat, lng });
   471→        setRouteData(null);
   472→        setActiveRoute(0);
   473→        setSelectionStep('destination');
   474→        identifyPlace(lat, lng, 'origin');
   475→      } else if (selectionStep === 'destination') {
   476→        if (origin && Math.abs(origin.lat - lat) < 0.001 && Math.abs(origin.lng - lng) < 0.001) {
   477→          return;
   478→        }
   479→        setDestination({ lat, lng });
   480→        setSelectionStep('ready');
   481→        identifyPlace(lat, lng, 'destination');
   482→      } else {
   483→        clearSelection();
   484→      }
   485→    },
   486→    [selectionStep, origin, identifyPlace]
   487→  );
   488→
   489→  // Search function
   490→  const handleSearch = useCallback(
   491→    (query: string, type: 'origin' | 'dest') => {
   492→      if (searchTimer.current) clearTimeout(searchTimer.current);
   493→
   494→      if (!query || query.length < 2) {
   495→        if (type === 'origin') setOriginResults([]);
   496→        else setDestResults([]);
   497→        return;
   498→      }
   499→
   500→      setSearchLoading(type);
   501→
   502→      searchTimer.current = setTimeout(async () => {
   503→        try {
   504→          const encoded = encodeURIComponent(query + ', ایران');
   505→          const url = `https://nominatim.openstreetmap.org/search?q=${encoded}&format=json&limit=5&accept-language=fa&addressdetails=1`;
   506→
   507→          const res = await fetch(url, {
   508→            headers: {
   509→              'User-Agent': 'SivanVIPTaxi/1.0',
   510→            },
   511→          });
   512→
   513→          if (!res.ok) return;
   514→
   515→          const data = await res.json();
   516→          if (type === 'origin') {
   517→            setOriginResults(data || []);
   518→          } else {
   519→            setDestResults(data || []);
   520→          }
   521→        } catch {
   522→          // ignore search errors
   523→        } finally {
   524→          setSearchLoading(null);
   525→        }
   526→      }, 800);
   527→    },
   528→    []
   529→  );
   530→
   531→  // Handle search select
   532→  const handleSearchSelect = useCallback(
   533→    (lat: number, lng: number, name: string, type: 'origin' | 'dest') => {
   534→      if (type === 'origin') {
   535→        setOrigin({ lat, lng, name });
   536→        setOriginSearch(name);
   537→        setOriginResults([]);
   538→        if (!destination) {
   539→          setSelectionStep('destination');
   540→        } else {
   541→          setSelectionStep('ready');
   542→        }
   543→      } else {
   544→        setDestination({ lat, lng, name });
   545→        setDestSearch(name);
   546→        setDestResults([]);
   547→        if (origin) {
   548→          setSelectionStep('ready');
   549→        } else {
   550→          setSelectionStep('origin');
   551→        }
   552→      }
   553→    },
   554→    [origin, destination]
   555→  );
   556→
   557→  // Fetch route when both points selected
   558→  useEffect(() => {
   559→    if (origin && destination) {
   560→      setRouteLoading(true);
   561→      setRouteError('');
   562→
   563→      const timer = setTimeout(async () => {
   564→        try {
   565→          const res = await fetch(
   566→            `/api/map/route?originLat=${origin.lat}&originLng=${origin.lng}&destLat=${destination.lat}&destLng=${destination.lng}`
   567→          );
   568→
   569→          if (!res.ok) {
   570→            const err = await res.json().catch(() => ({ error: 'خطا در محاسبه مسیر' }));
   571→            throw new Error(err.error || 'خطا');
   572→          }
   573→
   574→          const data = await res.json();
   575→          setRouteData(data);
   576→
   577→          // Always update names from route API (may be more accurate)
   578→          if (data.origin?.name) {
   579→            setOrigin((prev) => prev ? { ...prev, name: data.origin.name } : prev);
   580→            setOriginSearch(data.origin.name);
   581→          }
   582→          if (data.destination?.name) {
   583→            setDestination((prev) => prev ? { ...prev, name: data.destination.name } : prev);
   584→            setDestSearch(data.destination.name);
   585→          }
   586→        } catch (err) {
   587→          // Don't show error anymore - direct distance is always available
