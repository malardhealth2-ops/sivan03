    21→// ─── Admin Login ───
    22→function AdminLoginScreen() {
    23→  const { adminLogin, admin, setAdminOpen } = useAppStore();
    24→  const [username, setUsername] = useState('');
    25→  const [password, setPassword] = useState('');
    26→  const [showPass, setShowPass] = useState(false);
    27→  const [loading, setLoading] = useState(false);
    28→
    29→  const handleLogin = async () => {
    30→    setLoading(true);
    31→    await new Promise((r) => setTimeout(r, 800));
