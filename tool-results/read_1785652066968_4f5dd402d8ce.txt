   136→// ─── Invalidate Map Size Handler ─────────────────────────────────────────────

   138→function MapInvalidator() {
   139→  const map = useMap();
   140→
   141→  useEffect(() => {
   142→    // Force Leaflet to recalculate its size after the container is fully rendered
   143→    const timer = setTimeout(() => {
   144→      map.invalidateSize();
   145→    }, 200);
   146→    return () => clearTimeout(timer);
   147→  }, [map]);
   148→
   149→  // Also listen for scroll/resize events that might change the container
   150→  useEffect(() => {
   151→    let frameId: number;
   152→    const handleUpdate = () => {
   153→      if (frameId) cancelAnimationFrame(frameId);
   154→      frameId = requestAnimationFrame(() => {
   155→        map.invalidateSize({ animate: false });
   156→      });
   157→    };
   158→
   159→    window.addEventListener('resize', handleUpdate);
   160→    const observer = new IntersectionObserver((entries) => {
   161→      for (const entry of entries) {
   162→        if (entry.isIntersecting) {
   163→          setTimeout(() => map.invalidateSize({ animate: false }), 100);
   164→        }
   165→      }
   166→    }, { threshold: 0.1 });
   167→
   168→    const container = map.getContainer();
   169→    if (container) observer.observe(container);
   170→
   171→    return () => {
   172→      window.removeEventListener('resize', handleUpdate);
   173→      observer.disconnect();
   174→      if (frameId) cancelAnimationFrame(frameId);
   175→    };
   176→  }, [map]);
   177→
   178→  return null;
   179→}

   181→// ─── Fit to Bounds Handler ──────────────────────────────────────────────────
   182→
   183→function FitBoundsHandler({
   184→  points,
   185→  routeData,
   186→}: {
   187→  points: { origin: MapPoint | null; destination: MapPoint | null };
   188→  routeData: RouteResponse | null;
   189→}) {
   190→  const map = useMap();
   191→
   192→  useEffect(() => {
   193→    if (routeData && routeData.routes.length > 0 && routeData.routes[0].path.length > 0) {
   194→      const polyline = L.polyline(routeData.routes[0].path);
   195→      const bounds = polyline.getBounds().pad(0.15);
   196→      map.flyToBounds(bounds, { duration: 1.2 });
   197→    } else if (points.origin && points.destination) {
   198→      const bounds = L.latLngBounds(
   199→        [points.origin.lat, points.origin.lng],
   200→        [points.destination.lat, points.destination.lng]
   201→      ).pad(0.3);
   202→      map.flyToBounds(bounds, { duration: 1 });
   203→    } else if (points.origin) {
   204→      map.flyTo([points.origin.lat, points.origin.lng], 12, { duration: 1 });
   205→    }
   206→  }, [points, routeData, map]);
   207→
   208→  return null;
   209→}

   431→export default function InteractiveMapInner() {
   432→  const [origin, setOrigin] = useState<MapPoint | null>(null);
   433→  const [destination, setDestination] = useState<MapPoint | null>(null);
   434→  const [selectionStep, setSelectionStep] = useState<SelectionStep>('origin');
   435→  const [routeData, setRouteData] = useState<RouteResponse | null>(null);
   436→  const [routeLoading, setRouteLoading] = useState(false);
   437→  const [routeError, setRouteError] = useState('');
   438→  const [activeRoute, setActiveRoute] = useState(0);
   439→
   440→  // Search state
   441→  const [originSearch, setOriginSearch] = useState('');
   442→  const [destSearch, setDestSearch] = useState('');
   443→  const [originResults, setOriginResults] = useState<
   444→    { display_name: string; lat: string; lon: string }[]
   445→  >([]);
   446→  const [destResults, setDestResults] = useState<
   447→    { display_name: string; lat: string; lon: string }[]
   448→  >([]);
   449→  const [searchLoading, setSearchLoading] = useState<'origin' | 'dest' | null>(null);
   450→  const searchTimer = useRef<ReturnType<typeof setTimeout>>();
   451→  const [identifyingPlace, setIdentifyingPlace] = useState<'origin' | 'dest' | null>(null);
   452→
   453→  // Identify place name using VLM when clicking on map
   454→  const identifyPlace = useCallback(async (lat: number, lng: number, type: 'origin' | 'dest') => {
   455→    setIdentifyingPlace(type);
   456→    try {
   457→      const res = await fetch(`/api/map/identify-place?lat=${lat}&lng=${lng}`);
   458→      if (res.ok) {
   459→        const data = await res.json();
   460→        const name = data.name || '';
   461→        if (name) {
   462→          if (type === 'origin') {
   463→            setOriginSearch(name);
   464→            setOrigin(prev => prev ? { ...prev, name } : null);
   465→          } else {
   466→            setDestSearch(name);
   467→            setDestination(prev => prev ? { ...prev, name } : null);
   468→          }
   469→        }
   470→      }
   471→    } catch {
   472→      // Silently fail - the route API will also try to get place names
   473→    } finally {
   474→      setIdentifyingPlace(null);
   475→    }
   476→  }, []);
   477→
   478→  // Handle map click
   479→  const handleSelectPoint = useCallback(
   480→    (lat: number, lng: number) => {
   481→      if (selectionStep === 'origin') {
   482→        setOrigin({ lat, lng });
   483→        setRouteData(null);
   484→        setActiveRoute(0);
   485→        setSelectionStep('destination');
   486→        identifyPlace(lat, lng, 'origin');
   487→      } else if (selectionStep === 'destination') {
   488→        if (origin && Math.abs(origin.lat - lat) < 0.001 && Math.abs(origin.lng - lng) < 0.001) {
   489→          return;
   490→        }
   491→        setDestination({ lat, lng });
   492→        setSelectionStep('ready');
   493→        identifyPlace(lat, lng, 'dest');
   494→      } else {
   495→        clearSelection();
   496→      }
   497→    },
   498→    [selectionStep, origin, identifyPlace]
   499→  );
   500→
   501→  // Search function
   502→  const handleSearch = useCallback(
   503→    (query: string, type: 'origin' | 'dest') => {
   504→      if (searchTimer.current) clearTimeout(searchTimer.current);
   505→
   506→      if (!query || query.length < 2) {
   507→        if (type === 'origin') setOriginResults([]);
   508→        else setDestResults([]);
   509→        return;
   510→      }
   511→
   512→      setSearchLoading(type);
   513→
   514→      searchTimer.current = setTimeout(async () => {
   515→        try {
   516→          const encoded = encodeURIComponent(query + ', ایران');
   517→          const url = `https://nominatim.openstreetmap.org/search?q=${encoded}&format=json&limit=5&accept-language=fa&addressdetails=1`;
   518→
   519→          const res = await fetch(url, {
   520→            headers: {
   521→              'User-Agent': 'SivanVIPTaxi/1.0',
   522→            },
   523→          });
   524→
   525→          if (!res.ok) return;
   526→
   527→          const data = await res.json();
   528→          if (type === 'origin') {
   529→            setOriginResults(data || []);
   530→          } else {
   531→            setDestResults(data || []);
   532→          }
   533→        } catch {
   534→          // ignore search errors
   535→        } finally {
   536→          setSearchLoading(null);
   537→        }
   538→      }, 800);
   539→    },
   540→    []
   541→  );
   542→
   543→  // Handle search select
   544→  const handleSearchSelect = useCallback(
   545→    (lat: number, lng: number, name: string, type: 'origin' | 'dest') => {
   546→      if (type === 'origin') {
   547→        setOrigin({ lat, lng, name });
   548→        setOriginSearch(name);
   549→        setOriginResults([]);
   550→        if (!destination) {
   551→          setSelectionStep('destination');
   552→        } else {
   553→          setSelectionStep('ready');
   554→        }
   555→      } else {
   556→        setDestination({ lat, lng, name });
   557→        setDestSearch(name);
   558→        setDestResults([]);
   559→        if (origin) {
   560→          setSelectionStep('ready');
   561→        } else {
   562→          setSelectionStep('origin');
   563→        }
   564→      }
   565→    },
   566→    [origin, destination]
   567→  );
   568→
   569→  // Fetch route when both points selected
   570→  useEffect(() => {
   571→    if (origin && destination) {
   572→      setRouteLoading(true);
   573→      setRouteError('');
   574→
   575→      const timer = setTimeout(async () => {
   576→        try {
   577→          const res = await fetch(
   578→            `/api/map/route?originLat=${origin.lat}&originLng=${origin.lng}&destLat=${destination.lat}&destLng=${destination.lng}`
   579→          );
