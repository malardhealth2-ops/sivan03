     1→'use client';
     2→
     3→import { useState, useEffect, useRef, useCallback } from 'react';
     4→import { motion, AnimatePresence } from 'framer-motion';
     5→import {
     6→  User,
     7→  Loader2,
     8→  Check,
     9→  AlertTriangle,
    10→  UserPlus,
    11→  LogIn,
    12→  Shield,
    13→  Phone,
    14→  Smartphone,
    15→  KeyRound,
    16→  ArrowLeft,
    17→  Timer,
    18→  Fingerprint,
    19→} from 'lucide-react';
    20→import {
    21→  Dialog,
    22→  DialogContent,
    23→  DialogHeader,
    24→  DialogTitle,
    25→  DialogDescription,
    26→} from '@/components/ui/dialog';
    27→import { Button } from '@/components/ui/button';
    28→import { Input } from '@/components/ui/input';
    29→import { InputOTP, InputOTPGroup, InputOTPSlot, InputOTPSeparator } from '@/components/ui/input-otp';
    30→import { Label } from '@/components/ui/label';
    31→import { useAppStore } from '@/lib/store';
    32→import { toast } from 'sonner';
    33→
    34→// ─── OTP Step tracking ─────────────────────────────────────────────
    35→
    36→type OTPStep = 'phone' | 'code' | 'register' | 'success';
    37→
    38→export function AuthModal() {
    39→  const {
    40→    auth,
    41→    closeAuth,
    42→    setAuthUsername,
    43→    setAuthFullName,
    44→    setAuthPhone,
    45→    setAuthUser,
    46→    setAuthVerified,
    47→    adminLogin,
    48→    setAdminOpen,
    49→    openAuth,
    50→    setUserPanelOpen,
    51→  } = useAppStore();
    52→
    53→  const [step, setStep] = useState<OTPStep>('phone');
    54→  const [loading, setLoading] = useState(false);
    55→  const [localError, setLocalError] = useState('');
    56→  const [otpSent, setOtpSent] = useState(false);
    57→  const [isDemo, setIsDemo] = useState(false);
    58→  const [demoOtp, setDemoOtp] = useState('');
    59→  const [countdown, setCountdown] = useState(0);
    60→  const [isExistingUser, setIsExistingUser] = useState(false);
    61→  const [existingUser, setExistingUser] = useState<{ id: string; fullName: string; username: string; role: string } | null>(null);
    62→  const countdownRef = useRef<ReturnType<typeof setInterval>>();
    63→  const inputRef = useRef<HTMLInputElement>(null);
    64→
    65→  // Reset when modal opens/closes
    66→  useEffect(() => {
    67→    if (auth.isOpen) {
    68→      setStep('phone');
    69→      setLocalError('');
    70→      setOtpSent(false);
    71→      setIsDemo(false);
    72→      setDemoOtp('');
    73→      setCountdown(0);
    74→      setIsExistingUser(false);
    75→      setExistingUser(null);
    76→      // Focus phone input after mount
    77→      setTimeout(() => inputRef.current?.focus(), 200);
    78→    }
    79→  }, [auth.isOpen]);
    80→
    81→  // Countdown timer
    82→  useEffect(() => {
    83→    if (countdown > 0) {
    84→      countdownRef.current = setInterval(() => {
    85→        setCountdown((c) => {
    86→          if (c <= 1) {
    87→            clearInterval(countdownRef.current);
    88→            return 0;
    89→          }
    90→          return c - 1;
    91→        });
    92→      }, 1000);
    93→      return () => clearInterval(countdownRef.current);
    94→    }
    95→  }, [countdown]);
    96→
    97→  // ─── Step 1: Send OTP ──────────────────────────────────────
    98→
    99→  const handleSendOTP = async () => {
   100→    const phone = auth.phone.trim();
   101→    if (!phone || !/^09[0-9]{9}$/.test(phone)) {
   102→      setLocalError('شماره موبایل معتبر وارد کنید (مثال: ۰۹۱۲۳۴۵۶۷۸۹)');
   103→      return;
   104→    }
   105→    setLocalError('');
   106→    setLoading(true);
   107→
   108→    try {
   109→      const res = await fetch('/api/auth/send-otp', {
   110→        method: 'POST',
   111→        headers: { 'Content-Type': 'application/json' },
   112→        body: JSON.stringify({ phone }),
   113→      });
   114→
   115→      const data = await res.json();
   116→
   117→      if (!res.ok) {
   118→        setLocalError(data.error || 'خطا در ارسال کد تأیید');
   119→        setLoading(false);
   120→        return;
   121→      }
   122→
   123→      setOtpSent(true);
   124→      setIsDemo(!!data.isDemo);
   125→      if (data.isDemo && data.otp) {
   126→        setDemoOtp(data.otp);
   127→      }
   128→      setCountdown(120);
   129→      setStep('code');
   130→      toast.success('کد تأیید ارسال شد');
   131→    } catch {
   132→      setLocalError('خطا در ارتباط با سرور');
   133→    } finally {
   134→      setLoading(false);
   135→    }
   136→  };
   137→
   138→  // ─── Step 2: Verify OTP ─────────────────────────────────────
   139→
   140→  const handleVerifyOTP = async (code: string) => {
   141→    if (!code || code.length !== 6) return;
   142→    setLocalError('');
   143→    setLoading(true);
   144→
   145→    try {
   146→      const res = await fetch('/api/auth/verify-otp', {
   147→        method: 'POST',
   148→        headers: { 'Content-Type': 'application/json' },
   149→        body: JSON.stringify({ phone: auth.phone, code }),
   150→      });
   151→
   152→      const data = await res.json();
   153→
   154→      if (!res.ok) {
   155→        setLocalError(data.error || 'کد تأیید نامعتبر');
   156→        setLoading(false);
   157→        return;
   158→      }
   159→
   160→      if (data.isNewUser) {
   161→        // New user — go to registration step
   162→        setStep('register');
   163→        setLoading(false);
   164→      } else {
   165→        // Existing user — login directly
   166→        setIsExistingUser(true);
   167→        setExistingUser(data.user);
   168→        setAuthUser({
   169→          id: data.user.id,
   170→          username: data.user.username || data.user.phone,
   171→          fullName: data.user.fullName,
   172→          role: data.user.role,
   173→        });
   174→        setAuthVerified(true);
   175→        setStep('success');
   176→
   177→        // Admin handling
   178→        if (data.user?.role === 'admin') {
   179→          adminLogin(data.user.username, '');
   180→          setTimeout(() => {
   181→            setAdminOpen(true);
   182→            closeAuth();
   183→          }, 1200);
   184→        } else {
   185→          setTimeout(() => {
   186→            closeAuth();
   187→            setUserPanelOpen(true);
   188→          }, 1200);
   189→        }
   190→
   191→        toast.success(`خوش آمدید، ${data.user?.fullName || 'کاربر'} عزیز`);
   192→      }
   193→    } catch {
   194→      setLocalError('خطا در ارتباط با سرور');
   195→    } finally {
   196→      setLoading(false);
   197→    }
   198→  };
   199→
   200→  // ─── Step 3: Complete Registration ──────────────────────────
   201→
   202→  const handleCompleteRegistration = async () => {
   203→    const fullName = auth.fullName.trim();
   204→    if (!fullName || fullName.length < 2) {
   205→      setLocalError('نام و نام خانوادگی را کامل وارد کنید');
   206→      return;
   207→    }
   208→    setLocalError('');
   209→    setLoading(true);
   210→
   211→    try {
   212→      const res = await fetch('/api/auth/register', {
   213→        method: 'POST',
   214→        headers: { 'Content-Type': 'application/json' },
   215→        body: JSON.stringify({
   216→          username: `user_${auth.phone}`,
   217→          fullName,
   218→          password: `pass_${Date.now()}`, // Auto-generated since auth is OTP-based
   219→          phone: auth.phone,
   220→        }),
   221→      });
   222→
   223→      const data = await res.json();
   224→
   225→      if (!res.ok) {
   226→        setLocalError(data.error || 'خطا در ثبت‌نام');
   227→        setLoading(false);
   228→        return;
   229→      }
   230→
   231→      setAuthUser({
   232→        id: data.user.id,
   233→        username: data.user.username,
   234→        fullName: data.user.fullName,
   235→        role: data.user.role,
   236→      });
   237→      setAuthVerified(true);
   238→      setStep('success');
   239→
   240→      toast.success(`ثبت‌نام موفق! خوش آمدید، ${data.user.fullName} عزیز`);
   241→      setTimeout(() => {
   242→        closeAuth();
   243→        setUserPanelOpen(true);
   244→      }, 1200);
   245→    } catch {
   246→      setLocalError('خطا در ارتباط با سرور');
   247→    } finally {
   248→      setLoading(false);
   249→    }
   250→  };
   251→
   252→  // ─── Legacy login via username/password ─────────────────────
   253→
   254→  const [legacyMode, setLegacyMode] = useState(false);
   255→  const [showPass, setShowPass] = useState(false);
   256→
   257→  const handleLegacyLogin = async () => {
   258→    if (!auth.username || !auth.password) {
   259→      setLocalError('نام کاربری و رمز عبور را وارد کنید');
   260→      return;
   261→    }
   262→    setLocalError('');
   263→    setLoading(true);
   264→
   265→    try {
   266→      const res = await fetch('/api/auth/login', {
   267→        method: 'POST',
   268→        headers: { 'Content-Type': 'application/json' },
   269→        body: JSON.stringify({ username: auth.username, password: auth.password }),
   270→      });
   271→
   272→      const data = await res.json();
   273→
   274→      if (!res.ok) {
   275→        setLocalError(data.error || 'خطا در ورود');
   276→        setLoading(false);
   277→        return;
   278→      }
   279→
   280→      setAuthUser(data.user || null);
   281→
   282→      if (data.user?.role === 'admin') {
   283→        adminLogin(auth.username, auth.password);
   284→        setAdminOpen(true);
   285→        closeAuth();
   286→        setLoading(false);
   287→        return;
   288→      }
   289→
   290→      setAuthVerified(true);
   291→      setLoading(false);
   292→      toast.success(`خوش آمدید، ${data.user?.fullName || 'کاربر'} عزیز`);
   293→      setTimeout(() => {
   294→        closeAuth();
   295→        setUserPanelOpen(true);
   296→      }, 900);
   297→    } catch {
   298→      setLocalError('خطا در ارتباط با سرور');
   299→      setLoading(false);
   300→    }
   301→  };
   302→
   303→  const resetAndClose = () => {
   304→    setLocalError('');
   305→    closeAuth();
   306→  };
   307→
   308→  const formatCountdown = (s: number) => {
   309→    const min = Math.floor(s / 60);
   310→    const sec = s % 60;
   311→    return `${min}:${sec.toString().padStart(2, '0')}`;
   312→  };
   313→
   314→  return (
   315→    <Dialog open={auth.isOpen} onOpenChange={(val) => { if (!val) resetAndClose(); }}>
   316→      <DialogContent className="bg-[#1a1a1a] border-[#333] max-w-md">
   317→        <DialogHeader>
   318→          <DialogTitle className="text-right text-[#fafafa]">
   319→            {step === 'success'
   320→              ? 'خوش آمدید!'
   321→              : step === 'register'
   322→                ? 'تکمیل ثبت‌نام'
   323→                : step === 'code'
   324→                  ? 'تأیید شماره موبایل'
   325→                  : legacyMode
   326→                    ? 'ورود با نام کاربری'
   327→                    : 'ورود / ثبت‌نام'}
   328→          </DialogTitle>
   329→          <DialogDescription className="text-right text-[#a1a1aa]">
   330→            {step === 'success'
   331→              ? 'حساب شما با موفقیت تأیید شد'
   332→              : step === 'register'
   333→                ? 'برای تکمیل ثبت‌نام، نام خود را وارد کنید'
   334→                : step === 'code'
   335→                  ? `کد تأیید ارسال شده به ${auth.phone} را وارد کنید`
   336→                  : legacyMode
   337→                    ? 'نام کاربری و رمز عبور خود را وارد کنید'
   338→                    : 'شماره موبایل خود را وارد کنید تا کد تأیید ارسال شود'}
   339→          </DialogDescription>
   340→        </DialogHeader>
   341→
   342→        <div className="mt-2">
   343→          <AnimatePresence mode="wait">
   344→            {/* ─── Success State ─────────────────────────── */}
   345→            {step === 'success' && (
   346→              <motion.div
   347→                key="success"
   348→                initial={{ opacity: 0, scale: 0.9 }}
   349→                animate={{ opacity: 1, scale: 1 }}
   350→                className="text-center py-6 space-y-4"
   351→              >
   352→                <div className="w-16 h-16 rounded-full bg-green-500/20 flex items-center justify-center mx-auto">
   353→                  <Check className="h-8 w-8 text-green-400" />
   354→                </div>
   355→                <p className="text-[#fafafa] font-medium">
   356→                  {auth.user?.fullName ? `${auth.user.fullName} عزیز، خوش آمدید!` : 'خوش آمدید!'}
   357→                </p>
   358→                <p className="text-[#a1a1aa] text-xs">در حال ورود به پنل کاربری...</p>
   359→              </motion.div>
   360→            )}
   361→
   362→            {/* ─── Phone Entry ───────────────────────────── */}
   363→            {step === 'phone' && !legacyMode && (
   364→              <motion.div
   365→                key="phone"
   366→                initial={{ opacity: 0, x: -20 }}
   367→                animate={{ opacity: 1, x: 0 }}
   368→                exit={{ opacity: 0, x: 20 }}
   369→                className="space-y-4"
   370→              >
   371→                <div className="flex justify-center">
   372→                  <div className="w-16 h-16 rounded-full bg-[#D4AF37]/10 flex items-center justify-center">
   373→                    <Smartphone className="h-8 w-8 text-[#D4AF37]" />
   374→                  </div>
   375→                </div>
   376→
   377→                <div className="space-y-2">
   378→                  <Label className="text-[#fafafa] text-sm">
   379→                    <Phone className="h-3.5 w-3.5 ml-1.5 text-[#D4AF37] inline" />
   380→                    شماره موبایل
   381→                  </Label>
   382→                  <Input
   383→                    ref={inputRef}
   384→                    type="tel"
   385→                    placeholder="۰۹۱۲۳۴۵۶۷۸۹"
   386→                    value={auth.phone}
   387→                    onChange={(e) => {
   388→                      const val = e.target.value.replace(/[^0-9]/g, '');
   389→                      setAuthPhone(val);
   390→                      setLocalError('');
   391→                    }}
   392→                    onKeyDown={(e) => e.key === 'Enter' && handleSendOTP()}
   393→                    className="bg-[#0a0a0a] border-[#333] text-[#fafafa] placeholder:text-[#888] h-12 text-lg tracking-wider"
   394→                    dir="ltr"
   395→                    autoFocus
   396→                    maxLength={11}
   397→                  />
   398→                  <p className="text-[10px] text-[#666]">
   399→                    کد تأیید به این شماره ارسال می‌شود
   400→                  </p>
   401→                </div>
   402→
   403→                {localError && (
   404→                  <motion.div
   405→                    initial={{ opacity: 0, y: -5 }}
   406→                    animate={{ opacity: 1, y: 0 }}
   407→                    className="flex items-center gap-2 p-3 rounded-lg bg-red-500/10 border border-red-500/20 text-red-400 text-sm"
   408→                  >
   409→                    <AlertTriangle className="h-4 w-4 shrink-0" />
   410→                    {localError}
   411→                  </motion.div>
   412→                )}
   413→
   414→                <Button
   415→                  onClick={handleSendOTP}
   416→                  disabled={loading || auth.phone.length !== 11}
   417→                  className="w-full bg-[#D4AF37] text-[#0a0a0a] hover:bg-[#E5C76B] h-12 font-medium"
   418→                >
   419→                  {loading ? (
   420→                    <Loader2 className="h-4 w-4 animate-spin" />
   421→                  ) : (
   422→                    <>
   423→                      <Fingerprint className="h-4 w-4 ml-2" />
   424→                      ارسال کد تأیید
   425→                    </>
   426→                  )}
   427→                </Button>
   428→
   429→                {/* Switch to legacy login */}
   430→                <div className="text-center text-xs text-[#a1a1aa] pt-1">
   431→                  ورود با نام کاربری؟{' '}
   432→                  <button
   433→                    type="button"
   434→                    onClick={() => { setLegacyMode(true); setLocalError(''); }}
   435→                    className="text-[#D4AF37] hover:text-[#E5C76B] font-medium underline-offset-2 hover:underline"
   436→                  >
   437→                    کلیک کنید
   438→                  </button>
   439→                </div>
   440→              </motion.div>
   441→            )}
   442→
   443→            {/* ─── OTP Code Entry ─────────────────────────── */}
   444→            {step === 'code' && (
   445→              <motion.div
   446→                key="code"
   447→                initial={{ opacity: 0, x: 20 }}
   448→                animate={{ opacity: 1, x: 0 }}
   449→                exit={{ opacity: 0, x: -20 }}
   450→                className="space-y-4"
   451→              >
   452→                <div className="flex justify-center">
   453→                  <div className="w-16 h-16 rounded-full bg-[#D4AF37]/10 flex items-center justify-center">
   454→                    <KeyRound className="h-8 w-8 text-[#D4AF37]" />
   455→                  </div>
   456→                </div>
   457→
   458→                <div className="flex flex-col items-center gap-3">
   459→                  <Label className="text-[#fafafa] text-sm text-center">
   460→                    کد تأیید ۶ رقمی
   461→                  </Label>
   462→                  <InputOTP
   463→                    maxLength={6}
   464→                    onComplete={(code) => handleVerifyOTP(code)}
   465→                    disabled={loading}
   466→                    containerClassName="justify-center"
   467→                    dir="ltr"
   468→                  >
   469→                    <InputOTPGroup>
   470→                      <InputOTPSlot index={0} className="bg-[#0a0a0a] border-[#333] text-[#fafafa] h-12 w-12" />
   471→                      <InputOTPSlot index={1} className="bg-[#0a0a0a] border-[#333] text-[#fafafa] h-12 w-12" />
   472→                      <InputOTPSlot index={2} className="bg-[#0a0a0a] border-[#333] text-[#fafafa] h-12 w-12" />
   473→                    </InputOTPGroup>
   474→                    <InputOTPSeparator className="text-[#666]" />
   475→                    <InputOTPGroup>
   476→                      <InputOTPSlot index={3} className="bg-[#0a0a0a] border-[#333] text-[#fafafa] h-12 w-12" />
   477→                      <InputOTPSlot index={4} className="bg-[#0a0a0a] border-[#333] text-[#fafafa] h-12 w-12" />
   478→                      <InputOTPSlot index={5} className="bg-[#0a0a0a] border-[#333] text-[#fafafa] h-12 w-12" />
   479→                    </InputOTPGroup>
   480→                  </InputOTP>
   481→
   482→                  {/* Demo OTP display */}
   483→                  {isDemo && demoOtp && (
   484→                    <div className="bg-amber-500/10 border border-amber-500/30 rounded-lg px-4 py-2 text-center">
   485→                      <p className="text-amber-400 text-xs mb-1">حالت آزمایشی — کد تأیید:</p>
   486→                      <p className="text-amber-300 font-mono text-lg tracking-widest font-bold">{demoOtp}</p>
   487→                    </div>
   488→                  )}
   489→
   490→                  {/* Countdown / Resend */}
   491→                  <div className="flex items-center gap-2 text-xs">
   492→                    {countdown > 0 ? (
   493→                      <>
   494→                        <Timer className="h-3 w-3 text-[#a1a1aa]" />
   495→                        <span className="text-[#a1a1aa]">
   496→                          ارسال مجدد تا {formatCountdown(countdown)}
   497→                        </span>
   498→                      </>
   499→                    ) : (
   500→                      <button
   501→                        type="button"
   502→                        onClick={handleSendOTP}
   503→                        disabled={loading}
   504→                        className="text-[#D4AF37] hover:text-[#E5C76B] font-medium"
   505→                      >
   506→                        ارسال مجدد کد تأیید
   507→                      </button>
   508→                    )}
   509→                  </div>
   510→                </div>
   511→
   512→                {localError && (
   513→                  <motion.div
   514→                    initial={{ opacity: 0, y: -5 }}
   515→                    animate={{ opacity: 1, y: 0 }}
   516→                    className="flex items-center gap-2 p-3 rounded-lg bg-red-500/10 border border-red-500/20 text-red-400 text-sm"
   517→                  >
   518→                    <AlertTriangle className="h-4 w-4 shrink-0" />
   519→                    {localError}
   520→                  </motion.div>
   521→                )}
   522→
   523→                {loading && (
   524→                  <div className="flex items-center justify-center gap-2 text-sm text-[#a1a1aa]">
   525→                    <Loader2 className="h-4 w-4 animate-spin text-[#D4AF37]" />
   526→                    در حال تأیید...
   527→                  </div>
   528→                )}
   529→
   530→                {/* Back button */}
   531→                <Button
   532→                  variant="ghost"
   533→                  size="sm"
   534→                  onClick={() => { setStep('phone'); setLocalError(''); }}
   535→                  className="w-full text-[#a1a1aa] hover:text-[#fafafa]"
   536→                >
   537→                  <ArrowLeft className="h-4 w-4 ml-1" />
   538→                  تغییر شماره موبایل
   539→                </Button>
   540→              </motion.div>
   541→            )}
   542→
   543→            {/* ─── Registration (New User) ────────────────── */}
   544→            {step === 'register' && (
   545→              <motion.div
   546→                key="register"
   547→                initial={{ opacity: 0, x: 20 }}
   548→                animate={{ opacity: 1, x: 0 }}
   549→                exit={{ opacity: 0, x: -20 }}
   550→                className="space-y-4"
   551→              >
   552→                <div className="flex justify-center">
   553→                  <div className="w-16 h-16 rounded-full bg-[#D4AF37]/10 flex items-center justify-center">
   554→                    <UserPlus className="h-8 w-8 text-[#D4AF37]" />
   555→                  </div>
   556→                </div>
   557→
   558→                <div className="flex items-center gap-2 p-3 bg-green-500/10 border border-green-500/30 rounded-xl text-sm">
   559→                  <Check className="h-4 w-4 text-green-400 shrink-0" />
   560→                  <span className="text-green-300">شماره موبایل {auth.phone} تأیید شد</span>
   561→                </div>
   562→
