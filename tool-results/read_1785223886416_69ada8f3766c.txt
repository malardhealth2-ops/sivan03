   132→// ─── Invalidate Map Size Handler ─────────────────────────────────────────────
   133→
   134→function MapInvalidator() {
   135→  const map = useMap();
   136→
   137→  useEffect(() => {
   138→    // Force Leaflet to recalculate its size after the container is fully rendered
   139→    const timer = setTimeout(() => {
   140→      map.invalidateSize();
   141→    }, 200);
   142→    return () => clearTimeout(timer);
   143→  }, [map]);
   144→
   145→  // Also listen for scroll/resize events that might change the container
   146→  useEffect(() => {
   147→    let frameId: number;
   148→    const handleUpdate = () => {
   149→      if (frameId) cancelAnimationFrame(frameId);
   150→      frameId = requestAnimationFrame(() => {
   151→        map.invalidateSize({ animate: false });
   152→      });
   153→    };
   154→
   155→    window.addEventListener('resize', handleUpdate);
   156→    const observer = new IntersectionObserver((entries) => {
   157→      for (const entry of entries) {
   158→        if (entry.isIntersecting) {
   159→          setTimeout(() => map.invalidateSize({ animate: false }), 100);
   160→        }
   161→      }
   162→    }, { threshold: 0.1 });
   163→
   164→    const container = map.getContainer();
   165→    if (container) observer.observe(container);
   166→
   167→    return () => {
   168→      window.removeEventListener('resize', handleUpdate);
   169→      observer.disconnect();
   170→      if (frameId) cancelAnimationFrame(frameId);
   171→    };
   172→  }, [map]);
   173→
   174→  return null;
   175→}

   177→// ─── Fit to Bounds Handler ──────────────────────────────────────────────────
   178→
   179→function FitBoundsHandler({
   180→  points,
   181→  routeData,
   182→}: {
   183→  points: { origin: MapPoint | null; destination: MapPoint | null };
   184→  routeData: RouteResponse | null;
   185→}) {
   186→  const map = useMap();
   187→
   188→  useEffect(() => {
   189→    if (routeData && routeData.routes.length > 0 && routeData.routes[0].path.length > 0) {
   190→      const polyline = L.polyline(routeData.routes[0].path);
   191→      const bounds = polyline.getBounds().pad(0.15);
   192→      map.flyToBounds(bounds, { duration: 1.2 });
   193→    } else if (points.origin && points.destination) {
   194→      const bounds = L.latLngBounds(
   195→        [points.origin.lat, points.origin.lng],
   196→        [points.destination.lat, points.destination.lng]
   197→      ).pad(0.3);
   198→      map.flyToBounds(bounds, { duration: 1 });
   199→    } else if (points.origin) {
   200→      map.flyTo([points.origin.lat, points.origin.lng], 12, { duration: 1 });
   201→    }
   202→  }, [points, routeData, map]);
   203→
   204→  return null;
   205→}

   328→export default function InteractiveMapInner() {
   329→  const [origin, setOrigin] = useState<MapPoint | null>(null);
   330→  const [destination, setDestination] = useState<MapPoint | null>(null);
   331→  const [selectionStep, setSelectionStep] = useState<SelectionStep>('origin');
   332→  const [routeData, setRouteData] = useState<RouteResponse | null>(null);
   333→  const [routeLoading, setRouteLoading] = useState(false);
   334→  const [routeError, setRouteError] = useState('');
   335→  const [activeRoute, setActiveRoute] = useState(0);
   336→
   337→  // Search state
   338→  const [originSearch, setOriginSearch] = useState('');
   339→  const [destSearch, setDestSearch] = useState('');
   340→  const [originResults, setOriginResults] = useState<
   341→    { display_name: string; lat: string; lon: string }[]
   342→  >([]);
   343→  const [destResults, setDestResults] = useState<
   344→    { display_name: string; lat: string; lon: string }[]
   345→  >([]);
   346→  const [searchLoading, setSearchLoading] = useState<'origin' | 'dest' | null>(null);
   347→  const searchTimer = useRef<ReturnType<typeof setTimeout>>();
   348→  const [identifyingPlace, setIdentifyingPlace] = useState<'origin' | 'dest' | null>(null);
   349→
   350→  // Identify place name using VLM when clicking on map
   351→  const identifyPlace = useCallback(async (lat: number, lng: number, type: 'origin' | 'dest') => {
   352→    setIdentifyingPlace(type);
   353→    try {
   354→      const res = await fetch(`/api/map/identify-place?lat=${lat}&lng=${lng}`);
   355→      if (res.ok) {
   356→        const data = await res.json();
   357→        const name = data.name || '';
   358→        if (name) {
   359→          if (type === 'origin') {
   360→            setOriginSearch(name);
   361→            setOrigin(prev => prev ? { ...prev, name } : null);
   362→          } else {
   363→            setDestSearch(name);
   364→            setDestination(prev => prev ? { ...prev, name } : null);
   365→          }
   366→        }
   367→      }
   368→    } catch {
   369→      // Silently fail - the route API will also try to get place names
   370→    } finally {
   371→      setIdentifyingPlace(null);
   372→    }
   373→  }, []);
   374→
   375→  // Handle map click
   376→  const handleSelectPoint = useCallback(
   377→    (lat: number, lng: number) => {
   378→      if (selectionStep === 'origin') {
   379→        setOrigin({ lat, lng });
   380→        setRouteData(null);
   381→        setActiveRoute(0);
   382→        setSelectionStep('destination');
   383→        identifyPlace(lat, lng, 'origin');
   384→      } else if (selectionStep === 'destination') {
   385→        if (origin && Math.abs(origin.lat - lat) < 0.001 && Math.abs(origin.lng - lng) < 0.001) {
   386→          return;
   387→        }
   388→        setDestination({ lat, lng });
   389→        setSelectionStep('ready');
   390→        identifyPlace(lat, lng, 'destination');
   391→      } else {
   392→        clearSelection();
   393→      }
   394→    },
   395→    [selectionStep, origin, identifyPlace]
   396→  );
   397→
   398→  // Search function
   399→  const handleSearch = useCallback(
   400→    (query: string, type: 'origin' | 'dest') => {
   401→      if (searchTimer.current) clearTimeout(searchTimer.current);
   402→
   403→      if (!query || query.length < 2) {
   404→        if (type === 'origin') setOriginResults([]);
   405→        else setDestResults([]);
   406→        return;
   407→      }
   408→
   409→      setSearchLoading(type);
   410→
   411→      searchTimer.current = setTimeout(async () => {
   412→        try {
   413→          const encoded = encodeURIComponent(query + ', ایران');
   414→          const url = `https://nominatim.openstreetmap.org/search?q=${encoded}&format=json&limit=5&accept-language=fa&addressdetails=1`;
   415→
   416→          const res = await fetch(url, {
   417→            headers: {
   418→              'User-Agent': 'SivanVIPTaxi/1.0',
   419→            },
   420→          });
   421→
   422→          if (!res.ok) return;
   423→
   424→          const data = await res.json();
   425→          if (type === 'origin') {
   426→            setOriginResults(data || []);
   427→          } else {
   428→            setDestResults(data || []);
   429→          }
   430→        } catch {
   431→          // ignore search errors
   432→        } finally {
   433→          setSearchLoading(null);
   434→        }
   435→      }, 800);
   436→    },
   437→    []
   438→  );
   439→
   440→  // Handle search select
   441→  const handleSearchSelect = useCallback(
   442→    (lat: number, lng: number, name: string, type: 'origin' | 'dest') => {
   443→      if (type === 'origin') {
   444→        setOrigin({ lat, lng, name });
   445→        setOriginSearch(name);
   446→        setOriginResults([]);
   447→        if (!destination) {
   448→          setSelectionStep('destination');
   449→        } else {
   450→          setSelectionStep('ready');
   451→        }
   452→      } else {
   453→        setDestination({ lat, lng, name });
   454→        setDestSearch(name);
   455→        setDestResults([]);
   456→        if (origin) {
   457→          setSelectionStep('ready');
   458→        } else {
   459→          setSelectionStep('origin');
   460→        }
   461→      }
   462→    },
   463→    [origin, destination]
   464→  );
   465→
   466→  // Fetch route when both points selected
   467→  useEffect(() => {
   468→    if (origin && destination) {
   469→      setRouteLoading(true);
   470→      setRouteError('');
   471→
   472→      const timer = setTimeout(async () => {
   473→        try {
   474→          const res = await fetch(
   475→            `/api/map/route?originLat=${origin.lat}&originLng=${origin.lng}&destLat=${destination.lat}&destLng=${destination.lng}`
   476→          );
   477→
   478→          if (!res.ok) {
   479→            const err = await res.json().catch(() => ({ error: 'خطا در محاسبه مسیر' }));
   480→            throw new Error(err.error || 'خطا');
   481→          }
   482→
   483→          const data = await res.json();
   484→          setRouteData(data);
   485→
   486→          // Always update names from route API (may be more accurate)
   487→          if (data.origin?.name) {
   488→            setOrigin((prev) => prev ? { ...prev, name: data.origin.name } : prev);
   489→            setOriginSearch(data.origin.name);
   490→          }
   491→          if (data.destination?.name) {
   492→            setDestination((prev) => prev ? { ...prev, name: data.destination.name } : prev);
   493→            setDestSearch(data.destination.name);
   494→          }
   495→        } catch (err) {
   496→          // Don't show error anymore - direct distance is always available
   497→          console.error('Route fetch warning:', err);
   498→          setRouteData(null);
   499→        } finally {
   500→          setRouteLoading(false);
   501→        }
   502→      }, 500);
   503→
   504→      return () => clearTimeout(timer);
   505→    }
   506→  }, [origin?.lat, origin?.lng, destination?.lat, destination?.lng]);
   507→
   508→  const clearSelection = () => {
   509→    setOrigin(null);
   510→    setDestination(null);
   511→    setRouteData(null);
   512→    setRouteError('');
   513→    setActiveRoute(0);
   514→    setOriginSearch('');
   515→    setDestSearch('');
   516→    setOriginResults([]);
   517→    setDestResults([]);
   518→    setSelectionStep('origin');
   519→  };
   520→
   521→  const swapPoints = () => {
   522→    if (!origin || !destination) return;
   523→    const tmpOrigin = origin;
   524→    const tmpOriginSearch = originSearch;
   525→    setOrigin(destination);
   526→    setDestination(tmpOrigin);
   527→    setOriginSearch(destSearch);
   528→    setDestSearch(tmpOriginSearch);
   529→    setRouteData(null);
   530→    setActiveRoute(0);
   531→  };
   532→
   533→  // Build polylines
   534→  const routePolylines = routeData?.routes.map((route) => route.path) || [];
   535→
   536→  return (
   537→    <section id="map" className="py-20 sm:py-24 bg-[#0a0a0a] relative">
   538→      {/* Decorative */}
   539→      <div className="absolute top-0 left-1/2 -translate-x-1/2 w-24 h-1 bg-gradient-to-l from-transparent via-[#D4AF37] to-transparent rounded-full" />
   540→
   541→      <div className="container mx-auto px-4 sm:px-6">
   542→        {/* Section Header */}
   543→        <motion.div
   544→          initial={{ opacity: 0, y: 20 }}
   545→          whileInView={{ opacity: 1, y: 0 }}
   546→          viewport={{ once: true, margin: '-100px' }}
   547→          transition={{ duration: 0.6 }}
   548→          className="text-center mb-10 sm:mb-14"
   549→        >
   550→          <Badge className="bg-[#D4AF37]/15 text-[#D4AF37] border-[#D4AF37]/30 mb-4 px-4 py-1.5">
   551→            <Navigation className="h-3.5 w-3.5 ml-1.5" />
   552→            نقشه تعاملی
   553→          </Badge>
   554→          <h2 className="text-2xl sm:text-3xl md:text-4xl font-bold text-[#fafafa] mb-4">
   555→            مسیر خود را روی <span className="text-gold-gradient">نقشه</span> انتخاب کنید
   556→          </h2>
   557→          <p className="text-[#a1a1aa] max-w-2xl mx-auto">
   558→            با کلیک روی نقشه مبدا و مقصد خود را مشخص کنید. مسیر حرکت، فاصله و مسیرهای جایگزین نمایش داده می‌شود
   559→          </p>
   560→        </motion.div>
   561→
   562→        <motion.div
   563→          initial={{ opacity: 0, y: 30 }}
   564→          whileInView={{ opacity: 1, y: 0 }}
   565→          viewport={{ once: true, margin: '-100px' }}
   566→          transition={{ duration: 0.7 }}
   567→          className="flex flex-col gap-6"
   568→        >
   569→          {/* Search & Controls Bar */}
   570→          <div className="relative z-10 bg-[#1a1a1a]/90 backdrop-blur-xl rounded-2xl border border-[#D4AF37]/20 p-4 sm:p-6">
   571→            {/* Selection Step Indicator */}
