   441→function parseSectionData(raw: { title?: string; subtitle?: string; body?: string } | undefined): Record<string, string> {
   442→  const result: Record<string, string> = {};
   443→  if (!raw) return result;
   444→  result.title = raw.title || '';
   445→  result.subtitle = raw.subtitle || '';
   446→  if (raw.body) {
   447→    try {
   448→      const parsed = JSON.parse(raw.body);
   449→      if (parsed && typeof parsed === 'object') {
   450→        for (const [k, v] of Object.entries(parsed)) {
   451→          if (typeof v === 'string') result[k] = v;
   452→          else if (Array.isArray(v)) result[k] = (v as string[]).join('\n');
   453→        }
   454→      } else {
   455→        result.body = raw.body;
   456→      }
   457→    } catch {
   458→      result.body = raw.body;
   459→    }
   460→  }
   461→  return result;
   462→}

   464→function serializeSectionData(data: Record<string, string>): { title: string; subtitle: string; body: string } {
   465→  const title = data.title || '';
   466→  const subtitle = data.subtitle || '';
   467→  const extra: Record<string, string | string[]> = {};
   468→  for (const [k, v] of Object.entries(data)) {
   469→    if (k === 'title' || k === 'subtitle') continue;
   470→    if (k === 'items' || k === 'body') {
   471→      extra[k] = v;
   472→    } else {
   473→      extra[k] = v;
   474→    }
   475→  }
   476→  const hasExtra = Object.keys(extra).length > 0;
   477→  return { title, subtitle, body: hasExtra ? JSON.stringify(extra) : '' };
   478→}

   594→function AIBlogGenerator({ onComplete }: { onComplete: () => void }) {
   595→  const [status, setStatus] = useState<{
   596→    running: boolean;
   597→    lastGeneratedAt: string | null;
   598→    lastError: string | null;
   599→    totalPosts: number;
   600→  } | null>(null);

   608→      if (!res.ok) throw new Error('status fetch failed');
   609→      const data = await res.json();
   610→      setStatus({
   611→        running: !!data.running,
   612→        lastGeneratedAt: data.lastGeneratedAt || null,
   613→        lastError: data.lastError || null,
   614→        totalPosts: data.totalPosts ?? 0,

   623→  useEffect(() => {
   624→    fetchStatus();
   625→    const t = setInterval(fetchStatus, 5000);
   626→    return () => clearInterval(t);

   630→  const prevRunning = React.useRef(false);

   639→  const handleGenerate = async () => {
   640→    setTriggering(true);
   641→    try {
   642→      const res = await fetch('/api/blog-generator/generate', { method: 'POST' });
   643→      const data = await res.json();
   644→      if (data.started) {
   645→        toast.success('تولید مقاله با هوش مصنوعی شروع شد — چند دقیقه طول می‌کشد');

   648→      } else {
   649→        toast.info(data.message || 'تولید قبلی هنوز در حال انجام است');
   650→      }
   651→    } catch {
   652→      toast.error('خطا در ارتباط با سرویس تولید مقاله');

   654→      setTriggering(false);
   655→    }

   658→  const handleCustomGenerate = async () => {
   659→    const topic = customTopic.trim();
   660→    if (!topic) {
   661→      toast.error('لطفاً موضوع مقاله را وارد کنید');
   662→      return;
   663→    }
   664→    if (topic.length < 3) {
   665→      toast.error('موضوع مقاله باید حداقل ۳ کاراکتر باشد');

   667→    }
   668→    if (topic.length > 200) {
   669→      toast.error('موضوع مقاله نباید بیشتر از ۲۰۰ کاراکتر باشد');
   670→      return;
   671→    }
   672→    setCustomTriggering(true);
   673→    try {
   674→      const res = await fetch('/api/blog-generator/generate-custom', {
   675→        method: 'POST',
   676→        headers: { 'Content-Type': 'application/json' },
   677→        body: JSON.stringify({ topic }),
   678→      });

  1901→function urlBase64ToUint8Array(base64String: string): Uint8Array {
  1902→  const padding = '='.repeat((4 - (base64String.length % 4)) % 4);
  1903→  const base64 = (base64String + padding).replace(/-/g, '+').replace(/_/g, '/');
  1904→  const rawData = atob(base64);
  1905→  const output = new Uint8Array(rawData.length);
  1906→  for (let i = 0; i < rawData.length; ++i) output[i] = rawData.charCodeAt(i);
  1907→  return output;
  1908→}

  1910→/** Plays a short two-tone "ding" using the Web Audio API (no audio file needed). */
  1911→function playNotifSound() {
  1912→  try {
  1913→    const AudioCtx = window.AudioContext || (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;
  1914→    if (!AudioCtx) return;
  1915→    const ctx = new AudioCtx();
  1916→    const now = ctx.currentTime;
  1917→
  1918→    const playTone = (freq: number, start: number, duration: number) => {
  1919→      const osc = ctx.createOscillator();
  1920→      const gain = ctx.createGain();
  1921→      osc.type = 'sine';
  1922→      osc.frequency.setValueAtTime(freq, now + start);
  1923→      gain.gain.setValueAtTime(0, now + start);
  1924→      gain.gain.linearRampToValueAtTime(0.25, now + start + 0.02);
  1925→      gain.gain.exponentialRampToValueAtTime(0.001, now + start + duration);
  1926→      osc.connect(gain);
  1927→      gain.connect(ctx.destination);
  1928→      osc.start(now + start);
  1929→      osc.stop(now + start + duration + 0.05);
  1930→    };
  1931→
  1932→    // Pleasant two-tone ascending chime (gold/bell-like)
  1933→    playTone(880, 0, 0.18);     // A5
  1934→    playTone(1318.5, 0.12, 0.3); // E6
  1935→
  1936→    setTimeout(() => { try { ctx.close(); } catch { /* ignore */ } }, 600);
  1937→  } catch { /* audio not available — silent */ }
  1938→}
