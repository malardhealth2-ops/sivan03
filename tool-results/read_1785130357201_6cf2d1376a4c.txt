    36→// ─── Admin Login ───
    37→function AdminLoginScreen() {
    38→  const { adminLogin, admin, setAdminOpen, setLoginError } = useAppStore();
    39→  const [username, setUsername] = useState('');
    40→  const [password, setPassword] = useState('');
    41→  const [showPass, setShowPass] = useState(false);
    42→  const [loading, setLoading] = useState(false);
    43→
    44→  const handleLogin = async () => {
    45→    if (!username || !password) return;
    46→    setLoading(true);
    47→    try {
    48→      const res = await fetch('/api/auth/login', {
    49→        method: 'POST',
    50→        headers: { 'Content-Type': 'application/json' },
    51→        body: JSON.stringify({ username, password }),
    52→      });
    53→      const data = await res.json();
    54→      if (!res.ok) {
    55→        setLoginError(data.error || 'نام کاربری یا رمز عبور اشتباه است');
    56→        setLoading(false);
    57→        return;
    58→      }
    59→      if (data.user?.role !== 'admin') {
    60→        setLoginError('این حساب کاربری دسترسی مدیریت ندارد');
    61→        setLoading(false);
    62→        return;
    63→      }
    64→      adminLogin(username, password);
    65→    } catch {
    66→      setLoginError('خطا در ارتباط با سرور');
    67→      setLoading(false);
    68→      return;
    69→    }
    70→    setLoading(false);
    71→  };
    72→
    73→  return (
    74→    <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="fixed inset-0 z-[100] bg-[#0a0a0a] flex items-center justify-center">
    75→      <div className="absolute inset-0 bg-gradient-to-br from-[#0a0a0a] via-[#111] to-[#0a0a0a]" />
    76→      <button onClick={() => setAdminOpen(false)} className="absolute top-6 left-6 p-2 rounded-lg hover:bg-[#2d2d2d] text-[#a1a1aa] hover:text-[#fafafa] transition-colors z-10">
    77→        <X className="h-6 w-6" />
    78→      </button>
    79→      <motion.div initial={{ opacity: 0, y: 30, scale: 0.95 }} animate={{ opacity: 1, y: 0, scale: 1 }} transition={{ duration: 0.5 }} className="relative z-10 w-full max-w-md mx-4">
    80→        <div className="bg-[#1a1a1a] border border-[#333] rounded-2xl p-8 shadow-2xl">
    81→          <div className="text-center mb-8">
    82→            <img src="/logo.png" alt="سیوان" className="h-16 w-16 rounded-2xl mx-auto mb-4 ring-2 ring-[#D4AF37]/40" />
    83→            <h1 className="text-2xl font-bold text-[#fafafa] mb-2">پنل مدیریت سیوان</h1>
    84→            <p className="text-[#a1a1aa] text-sm">نام کاربری و رمز عبور خود را وارد کنید</p>
    85→          </div>
    86→          <div className="space-y-5">
    87→            <div className="space-y-2">
    88→              <Label className="text-[#a1a1aa] text-sm">نام کاربری</Label>
    89→              <Input value={username} onChange={(e) => setUsername(e.target.value)} placeholder="نام کاربری" className="bg-[#0a0a0a] border-[#333] text-[#fafafa] placeholder:text-[#888] h-12" dir="ltr" autoFocus onKeyDown={(e) => e.key === 'Enter' && handleLogin()} />
    90→            </div>
    91→            <div className="space-y-2">
    92→              <Label className="text-[#a1a1aa] text-sm">رمز عبور</Label>
    93→              <div className="relative">
    94→                <Input type={showPass ? 'text' : 'password'} value={password} onChange={(e) => setPassword(e.target.value)} placeholder="رمز عبور" className="bg-[#0a0a0a] border-[#333] text-[#fafafa] placeholder:text-[#888] h-12" dir="ltr" onKeyDown={(e) => e.key === 'Enter' && handleLogin()} />
    95→                <button type="button" onClick={() => setShowPass(!showPass)} className="absolute left-3 top-1/2 -translate-y-1/2 text-[#a1a1aa] hover:text-[#fafafa]">{showPass ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}</button>
    96→              </div>
    97→            </div>
    98→            {admin.loginError && (
    99→              <motion.div initial={{ opacity: 0, y: -5 }} animate={{ opacity: 1, y: 0 }} className="flex items-center gap-2 p-3 rounded-lg bg-red-500/10 border border-red-500/20 text-red-400 text-sm"><AlertTriangle className="h-4 w-4 shrink-0" />{admin.loginError}</motion.div>
   100→            )}
   101→            <Button onClick={handleLogin} disabled={loading || !username || !password} className="w-full h-12 bg-[#D4AF37] text-[#0a0a0a] hover:bg-[#E5C76B] font-bold text-base rounded-xl">
   102→              {loading ? <Loader2 className="h-5 w-5 animate-spin" /> : 'ورود به پنل مدیریت'}
   103→            </Button>
   104→          </div>
   105→        </div>
   106→      </motion.div>
   107→    </motion.div>
   108→  );
   109→}

   111→// ─── Admin Dashboard ───
   112→function AdminDashboard() {
   113→  const { admin, adminLogout, setAdminOpen, setAdminActiveTab } = useAppStore();
   114→  const [jalaliClock, setJalaliClock] = useState('');
   115→
   116→  useEffect(() => {
   117→    const update = () => {
   118→      setJalaliClock(getTehranTimeString() + ' - ' + formatJalaaliDate(getTehranTime()));
   119→    };
   120→    update();
   121→    const i = setInterval(update, 1000);
   122→    return () => clearInterval(i);
   123→  }, []);
   124→
   125→  // ── Live email polling: foreground toast + background SW notification ──
   126→  // Polls /api/admin/emails every 25s; when a NEW email arrives (unseen id),
   127→  // shows a toast (foreground) + a service-worker notification (background)
   128→  // + a short notification sound. This complements the web-push system
   129→  // (which works even when the tab is fully closed).
   130→  useEffect(() => {
   131→    let seenIds = new Set<string>();
   132→    let firstLoad = true;
   133→
   134→    const checkNewEmails = async () => {
   135→      try {
   136→        const res = await fetch('/api/admin/emails?limit=5');
   137→        if (!res.ok) return;
   138→        const data = await res.json();
   139→        const items: { id: string; subject: string; toEmail: string; source: string; refId?: string }[] = data.items || [];
   140→        if (items.length === 0) return;
   141→
   142→        if (firstLoad) {
   143→          items.forEach((it) => seenIds.add(it.id));
   144→          firstLoad = false;
   145→          return;
   146→        }
   147→
   148→        const fresh = items.filter((it) => !seenIds.has(it.id));
   149→        if (fresh.length === 0) return;
   150→
   151→        fresh.forEach((it) => seenIds.add(it.id));
   152→
   153→        const isBooking = it.source === 'booking';
   154→        const title = isBooking ? `🚕 رزرو جدید: ${it.refId || ''}` : `📧 ایمیل جدید: ${it.subject}`;
   155→        const body = isBooking ? `کد رهگیری ${it.refId || ''} — برای مشاهده کلیک کنید` : `${it.subject} — برای ${it.toEmail}`;
   156→
   157→        // Foreground: toast + sound
   158→        toast(title, { description: body, duration: 8000 });
   159→        playNotifSound();
   160→
   161→        // Background: service worker notification (shows even if tab is not focused)
   162→        if ('serviceWorker' in navigator && 'Notification' in window && Notification.permission === 'granted') {
   163→          try {
   164→            const reg = await navigator.serviceWorker.ready;
   165→            reg.showNotification(title, {
   166→              body,
   167→              icon: '/icon-192.png',
   168→              badge: '/icon-192.png',
   169→              tag: `email-${it.id}`,
   170→              renotify: true,
   171→              dir: 'rtl',
   172→              lang: 'fa',
   173→              data: { url: '/' },
   174→              actions: [{ action: 'open', title: 'مشاهده' }, { action: 'close', title: 'بستن' }],
   175→              vibrate: [120, 60, 120],
   176→            });
   177→          } catch { /* ignore */ }
   178→        }
   179→      } catch { /* ignore */ }
   180→    };
   181→
   182→    checkNewEmails();
   183→    const iv = setInterval(checkNewEmails, 25000);
   184→    return () => clearInterval(iv);
   185→  }, []);
   186→
   187→  const tabs: { id: TabId; label: string; icon: typeof LayoutDashboard }[] = [
   188→    { id: 'dashboard', label: 'داشبورد', icon: LayoutDashboard },
   189→    { id: 'trips', label: 'سفرها', icon: Car },
   190→    { id: 'passengers', label: 'مسافران', icon: Users },
   191→    { id: 'drivers', label: 'رانندگان', icon: UserCheck },
   192→    { id: 'content', label: 'مدیریت محتوا', icon: Pencil },
   193→    { id: 'blog', label: 'بلاگ', icon: FileText },
   194→    { id: 'pricing', label: 'قیمت‌گذاری', icon: Calculator },
   195→    { id: 'emails', label: 'ایمیل‌ها', icon: Mail },
   196→    { id: 'notifications', label: 'اعلان‌ها', icon: Bell },
   197→    { id: 'settings', label: 'تنظیمات', icon: Settings },
   198→  ];
   199→
   200→  return (
   201→    <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="fixed inset-0 z-[100] bg-[#0a0a0a] flex">
   202→      <aside className="hidden md:flex w-64 bg-[#111111] border-l border-[#333] flex-col">
   203→        <div className="p-5 border-b border-[#333]">
   204→          <div className="flex items-center gap-3">
   205→            <img src="/logo.png" alt="سیوان" className="h-10 w-10 rounded-xl ring-1 ring-[#D4AF37]/30" />
   206→            <div><h3 className="text-[#fafafa] font-bold text-sm">پنل مدیریت</h3><p className="text-[#a1a1aa] text-xs">سیوان VIP</p></div>
   207→          </div>
   208→        </div>
   209→        <nav className="flex-1 p-3 space-y-1 overflow-y-auto">
   210→          {tabs.map((tab) => (
   211→            <button key={tab.id} onClick={() => setAdminActiveTab(tab.id)} className={`w-full flex items-center gap-3 px-4 py-3 rounded-xl text-sm transition-all ${admin.activeTab === tab.id ? 'bg-[#D4AF37] text-[#0a0a0a] font-bold shadow-lg shadow-[#D4AF37]/20' : 'text-[#a1a1aa] hover:text-[#fafafa] hover:bg-[#2d2d2d]'}`}>
   212→              <tab.icon className="h-4 w-4" />{tab.label}
   213→            </button>
   214→          ))}
   215→        </nav>
   216→        <div className="p-4 border-t border-[#333]">
   217→          <div className="flex items-center gap-3 mb-3 px-2">
   218→            <div className="w-8 h-8 rounded-full bg-[#D4AF37]/10 flex items-center justify-center"><span className="text-[#D4AF37] text-xs font-bold">A</span></div>
   219→            <div><p className="text-[#fafafa] text-sm">{admin.adminUsername}</p><p className="text-[#a1a1aa] text-[10px]">مدیر سیستم</p></div>
   220→          </div>
   221→          <Button onClick={adminLogout} variant="ghost" className="w-full text-red-400 hover:text-red-300 hover:bg-red-500/10 justify-start gap-2"><LogOut className="h-4 w-4" />خروج</Button>
   222→        </div>
   223→      </aside>
   224→
   225→      <div className="flex-1 flex flex-col overflow-hidden">
   226→        <header className="h-16 bg-[#111111] border-b border-[#333] flex items-center justify-between px-6">
   227→          <div className="flex items-center gap-3">
   228→            <span className="text-[#fafafa] font-bold">مدیریت سیوان</span>
   229→            <Separator orientation="vertical" className="h-6 bg-[#333]" />
   230→            <span className="text-[#a1a1aa] text-xs">{jalaliClock}</span>
   231→          </div>
   232→          <div className="flex items-center gap-3">
   233→            <div className="md:hidden">
   234→              <Select value={admin.activeTab} onValueChange={(v) => setAdminActiveTab(v as TabId)}>
   235→                <SelectTrigger className="w-36 bg-[#1a1a1a] border-[#333] text-[#fafafa] h-9 text-xs"><SelectValue /></SelectTrigger>
   236→                <SelectContent className="bg-[#1a1a1a] border-[#333]">
   237→                  {tabs.map((t) => (<SelectItem key={t.id} value={t.id} className="text-[#fafafa]">{t.label}</SelectItem>))}

   240→            </div>

   242→            <button onClick={() => setAdminOpen(false)} className="p-2 rounded-lg hover:bg-[#2d2d2d] text-[#a1a1aa] hover:text-[#fafafa] transition-colors"><X className="h-5 w-5" /></button>
   243→          </div>
   244→        </header>
   245→        <main className="flex-1 overflow-y-auto p-6">
   246→          <AnimatePresence mode="wait">
   247→            {admin.activeTab === 'dashboard' && <DashboardTab key="dash" />}
   248→            {admin.activeTab === 'trips' && <TripsTab key="trips" />}

   257→          </AnimatePresence>
   258→        </main>
   259→      </div>
   260→    </motion.div>
   261→  );
   262→}

   439→function parseSectionData(raw: { title?: string; subtitle?: string; body?: string } | undefined): Record<string, string> {
   440→  const result: Record<string, string> = {};
   441→  if (!raw) return result;
   442→  result.title = raw.title || '';
   443→  result.subtitle = raw.subtitle || '';
   444→  if (raw.body) {
   445→    try {
   446→      const parsed = JSON.parse(raw.body);
   447→      if (parsed && typeof parsed === 'object') {
   448→        for (const [k, v] of Object.entries(parsed)) {
   449→          if (typeof v === 'string') result[k] = v;
   450→          else if (Array.isArray(v)) result[k] = (v as string[]).join('\n');
   451→        }
   452→      } else {
   453→        result.body = raw.body;
   454→      }
   455→    } catch {
   456→      result.body = raw.body;
   457→    }
   458→  }
   459→  return result;
   460→}
   461→
   462→function serializeSectionData(data: Record<string, string>): { title: string; subtitle: string; body: string } {
   463→  const title = data.title || '';
   464→  const subtitle = data.subtitle || '';
   465→  const extra: Record<string, string | string[]> = {};
   466→  for (const [k, v] of Object.entries(data)) {
   467→    if (k === 'title' || k === 'subtitle') continue;
   468→    if (k === 'items' || k === 'body') {
   469→      extra[k] = v;
   470→    } else {
   471→      extra[k] = v;
   472→    }
   473→  }
   474→  const hasExtra = Object.keys(extra).length > 0;
   475→  return { title, subtitle, body: hasExtra ? JSON.stringify(extra) : '' };
   476→}

  1652→function urlBase64ToUint8Array(base64String: string): Uint8Array {
  1653→  const padding = '='.repeat((4 - (base64String.length % 4)) % 4);
  1654→  const base64 = (base64String + padding).replace(/-/g, '+').replace(/_/g, '/');
  1655→  const rawData = atob(base64);
  1656→  const output = new Uint8Array(rawData.length);
  1657→  for (let i = 0; i < rawData.length; ++i) output[i] = rawData.charCodeAt(i);
  1658→  return output;
  1659→}

  1661→/** Plays a short two-tone "ding" using the Web Audio API (no audio file needed). */
  1662→function playNotifSound() {
  1663→  try {
  1664→    const AudioCtx = window.AudioContext || (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;
  1665→    if (!AudioCtx) return;
  1666→    const ctx = new AudioCtx();
  1667→    const now = ctx.currentTime;
  1668→
  1669→    const playTone = (freq: number, start: number, duration: number) => {
  1670→      const osc = ctx.createOscillator();
  1671→      const gain = ctx.createGain();
  1672→      osc.type = 'sine';
  1673→      osc.frequency.setValueAtTime(freq, now + start);
  1674→      gain.gain.setValueAtTime(0, now + start);
  1675→      gain.gain.linearRampToValueAtTime(0.25, now + start + 0.02);
  1676→      gain.gain.exponentialRampToValueAtTime(0.001, now + start + duration);
  1677→      osc.connect(gain);
  1678→      gain.connect(ctx.destination);
  1679→      osc.start(now + start);
  1680→      osc.stop(now + start + duration + 0.05);
  1681→    };
  1682→
  1683→    // Pleasant two-tone ascending chime (gold/bell-like)
  1684→    playTone(880, 0, 0.18);     // A5
  1685→    playTone(1318.5, 0.12, 0.3); // E6
  1686→
  1687→    setTimeout(() => { try { ctx.close(); } catch { /* ignore */ } }, 600);
  1688→  } catch { /* audio not available — silent */ }
  1689→}
