   140→function MapInvalidator() {
   141→  const map = useMap();
   142→
   143→  useEffect(() => {
   144→    // Force Leaflet to recalculate its size after the container is fully rendered
   145→    const timer = setTimeout(() => {
   146→      map.invalidateSize();
   147→    }, 200);
   148→    return () => clearTimeout(timer);
   149→  }, [map]);
   150→
   151→  // Also listen for scroll/resize events that might change the container
   152→  useEffect(() => {
   153→    let frameId: number;
   154→    const handleUpdate = () => {
   155→      if (frameId) cancelAnimationFrame(frameId);
   156→      frameId = requestAnimationFrame(() => {
   157→        map.invalidateSize({ animate: false });
   158→      });
   159→    };
   160→
   161→    window.addEventListener('resize', handleUpdate);
   162→    const observer = new IntersectionObserver((entries) => {
   163→      for (const entry of entries) {
   164→        if (entry.isIntersecting) {
   165→          setTimeout(() => map.invalidateSize({ animate: false }), 100);
   166→        }
   167→      }
   168→    }, { threshold: 0.1 });
   169→
   170→    const container = map.getContainer();
   171→    if (container) observer.observe(container);
   172→
   173→    return () => {
   174→      window.removeEventListener('resize', handleUpdate);
   175→      observer.disconnect();
   176→      if (frameId) cancelAnimationFrame(frameId);
   177→    };
   178→  }, [map]);
   179→
   180→  return null;
   181→}

   185→function FitBoundsHandler({
   186→  points,
   187→  routeData,
   188→}: {
   189→  points: { origin: MapPoint | null; destination: MapPoint | null };
   190→  routeData: RouteResponse | null;
   191→}) {
   192→  const map = useMap();
   193→
   194→  useEffect(() => {
   195→    if (routeData && routeData.routes.length > 0 && routeData.routes[0].path.length > 0) {
   196→      const polyline = L.polyline(routeData.routes[0].path);
   197→      const bounds = polyline.getBounds().pad(0.15);
   198→      map.flyToBounds(bounds, { duration: 1.2 });
   199→    } else if (points.origin && points.destination) {
   200→      const bounds = L.latLngBounds(
   201→        [points.origin.lat, points.origin.lng],
   202→        [points.destination.lat, points.destination.lng]
   203→      ).pad(0.3);
   204→      map.flyToBounds(bounds, { duration: 1 });
   205→    } else if (points.origin) {
   206→      map.flyTo([points.origin.lat, points.origin.lng], 12, { duration: 1 });
   207→    }
   208→  }, [points, routeData, map]);
   209→
   210→  return null;
   211→}

   601→          setRouteLoading(false);
   602→        }
   603→      }, 500);

   605→      return () => clearTimeout(timer);

   607→  }, [origin?.lat, origin?.lng, destination?.lat, destination?.lng]);
