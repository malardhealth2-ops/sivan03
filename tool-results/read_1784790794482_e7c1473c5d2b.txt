   788→   788→    is_active BOOLEAN DEFAULT true,
   789→   789→    applicable_to ENUM('all', 'new_users', 'specific_trip_type'),
   790→   790→    created_by UUID REFERENCES users(id),
   791→   791→    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
   792→   792→);
   793→   793→
   794→   794→-- جدول لاگ فعالیت‌ها (Activity Logs)
   795→   795→CREATE TABLE activity_logs (
   796→   796→    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   797→   797→    user_id UUID REFERENCES users(id),
   798→   798→    action VARCHAR(100) NOT NULL,
   799→   799→    entity_type VARCHAR(50), -- e.g., 'trip', 'user', 'post'
   800→   800→    entity_id UUID,
   801→   801→    old_values JSONB,
   802→   802→    new_values JSONB,
   803→   803→    ip_address INET,
   804→   804→    user_agent TEXT,
   805→   805→    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
   806→   806→);
   807→   807→```
