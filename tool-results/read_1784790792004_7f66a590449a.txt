   227→interface DriverRegistration {

   534→### **جداول اصلی (Core Tables):**
   535→
   536→```sql
   537→-- جدول کاربران (Users)
   538→CREATE TABLE users (
   539→    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   540→    phone VARCHAR(11) UNIQUE NOT NULL,
   541→    email VARCHAR(255) UNIQUE,
   542→    password_hash VARCHAR(255) NOT NULL,
   543→    full_name VARCHAR(100) NOT NULL,
   544→    role ENUM('passenger', 'driver', 'admin') NOT NULL,
   545→    avatar_url TEXT,
   546→    is_active BOOLEAN DEFAULT true,
   547→    is_verified BOOLEAN DEFAULT false,
   548→    last_login_at TIMESTAMP WITH TIME ZONE,
   549→    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
   550→    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
   551→);
   552→
   553→-- جدول مسافران (Passengers)
   554→CREATE TABLE passengers (
   555→    id UUID PRIMARY KEY REFERENCES users(id),
   556→    national_id VARCHAR(10) UNIQUE,

   558→    birth_date DATE,
   559→    wallet_balance DECIMAL(12, 2) DEFAULT 0,

   562→    referral_code VARCHAR(10) UNIQUE,
   563→    referred_by UUID REFERENCES passengers(id)

   565→
   566→-- جدول رانندگان (Drivers)
   567→CREATE TABLE drivers (
   568→    id UUID PRIMARY KEY REFERENCES users(id),
   569→    national_id VARCHAR(10) UNIQUE NOT NULL,
   570→    father_name VARCHAR(50),
   571→    license_number VARCHAR(20),
   572→    license_expiry DATE,
   573→    health_cert_expiry DATE,
   574→    verification_status ENUM('pending', 'approved', 'rejected', 'suspended') DEFAULT 'pending',
   575→    rating DECIMAL(2, 1) DEFAULT 5.0,
   576→    total_trips INTEGER DEFAULT 0,
   577→    acceptance_rate DECIMAL(5, 2) DEFAULT 100.00,
   578→    completion_rate DECIMAL(5, 2) DEFAULT 100.00,
   579→    current_latitude DECIMAL(10, 8),
   580→    current_longitude DECIMAL(11, 8),
   581→    online_status ENUM('online', 'offline', 'busy') DEFAULT 'offline',
   582→    balance DECIMAL(12, 2) DEFAULT 0, -- موجودی قابل تسویه
   583→    sheba_number VARCHAR(26), -- IRXXXXXXXXXXXXXXX
   584→    bank_name VARCHAR(50),
   585→    account_owner VARCHAR(100),
   586→    joined_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
   587→);
   588→
   589→-- جدول خودروها (Vehicles)
   590→CREATE TABLE vehicles (
   591→    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   592→    driver_id UUID NOT NULL REFERENCES drivers(id),
   593→    type ENUM('sedan', 'suv', 'van', 'electric') NOT NULL,
   594→    brand VARCHAR(50) NOT NULL,
   595→    model VARCHAR(50) NOT NULL,
   596→    year INTEGER NOT NULL,
   597→    color VARCHAR(30) NOT NULL,
   598→    plate_number VARCHAR(10) UNIQUE NOT NULL,
   599→    insurance_expiry DATE,
   600→    tech_diagnosis_expiry DATE,
   601→    status ENUM('active', 'inactive', 'under_maintenance') DEFAULT 'active',
   602→    images JSONB, -- array of image URLs
   603→    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
   604→);
   605→
   606→-- جدول سفرها (Trips)
   607→CREATE TABLE trips (
   608→    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   609→    passenger_id UUID NOT NULL REFERENCES passengers(id),
   610→    driver_id UUID REFERENCES drivers(id),
   611→    
   612→    -- اطلاعات مسیر
   613→    origin_address TEXT NOT NULL,
   614→    origin_lat DECIMAL(10, 8) NOT NULL,
   615→    origin_lng DECIMAL(11, 8) NOT NULL,
   616→    destination_address TEXT NOT NULL,
   617→    destination_lat DECIMAL(10, 8) NOT NULL,
   618→    destination_lng DECIMAL(11, 8) NOT NULL,
   619→    
   620→    -- جزئیات سفر
   621→    trip_type ENUM('economy', 'vip', 'luxury', 'van', 'electric') NOT NULL,
   622→    status ENUM(
   623→        'pending',        -- منتظر پذیرش راننده
   624→        'accepted',       -- پذیرفته شده توسط راننده
   625→        'driver_arrived', -- راننده رسید
   626→        'in_progress',    -- در حال حرکت
   627→        'completed',      -- تکمیل شده
   628→        'cancelled',      -- لغو شده
   629→        'no_show'         -- مسافر حاضر نشد
   630→    ) DEFAULT 'pending',
   631→    
   632→    -- زمان‌بندی
   633→    requested_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
   634→    accepted_at TIMESTAMP WITH TIME ZONE,
   635→    picked_up_at TIMESTAMP WITH TIME ZONE,
   636→    completed_at TIMESTAMP WITH TIME ZONE,
   637→    cancelled_at TIMESTAMP WITH TIME ZONE,
   638→    scheduled_for TIMESTAMP WITH TIME ZONE, -- برای رزرو آینده
   639→    
   640→    -- مالی
   641→    base_fare DECIMAL(10, 2),
   642→    distance_fare DECIMAL(10, 2),
   643→    time_fare DECIMAL(10, 2),
   644→    discount_amount DECIMAL(10, 2) DEFAULT 0,
   645→    total_fare DECIMAL(10, 2) NOT NULL,
   646→    commission_amount DECIMAL(10, 2), -- سهم سیستم
   647→    driver_earnings DECIMAL(10, 2),   -- سهم راننده
   648→    payment_method ENUM('cash', 'online', 'wallet') DEFAULT 'cash',
   649→    payment_status ENUM('pending', 'paid', 'refunded') DEFAULT 'pending',
   650→    
   651→    -- اطلاعات اضافی
   652→    distance_km DECIMAL(8, 2),
   653→    duration_minutes INTEGER,
   654→    passenger_count INTEGER DEFAULT 1,
   655→    notes TEXT,
   656→    cancel_reason TEXT,
   657→    cancelled_by UUID REFERENCES users(id),
   658→    
   659→    -- امتیازدهی
   660→    passenger_rating DECIMAL(2, 1),
   661→    driver_rating DECIMAL(2, 1),
   662→    passenger_review TEXT,
   663→    driver_review TEXT,
   664→    
   665→    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
   666→    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
   667→);
   668→
   669→-- جدول تراکنش‌ها (Transactions)
   670→CREATE TABLE transactions (
   671→    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   672→    user_id UUID NOT NULL REFERENCES users(id),
   673→    trip_id UUID REFERENCES trips(id),
   674→    type ENUM('credit', 'debit', 'commission', 'refund', 'topup', 'withdrawal') NOT NULL,
   675→    amount DECIMAL(12, 2) NOT NULL,
   676→    description TEXT,
   677→    reference_id VARCHAR(100), -- Reference from payment gateway
   678→    status ENUM('pending', 'completed', 'failed') DEFAULT 'pending',
   679→    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
   680→);
   681→
   682→-- جدول کیف پول (Wallet)
   683→CREATE TABLE wallets (
   684→    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   685→    user_id UUID UNIQUE NOT NULL REFERENCES users(id),
   686→    balance DECIMAL(12, 2) DEFAULT 0,
   687→    frozen_balance DECIMAL(12, 2) DEFAULT 0, -- برای تراکنش‌های در حال پردازش
   688→    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
   689→);
   690→
   691→-- جدول بلاگ (Blog Posts)
   692→CREATE TABLE blog_posts (
   693→    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   694→    title VARCHAR(255) NOT NULL,
   695→    slug VARCHAR(255) UNIQUE NOT NULL,
   696→    excerpt TEXT,
   697→    content TEXTB NOT NULL, -- Markdown or HTML
   698→    featured_image_url TEXT,
   699→    author_id UUID REFERENCES users(id),
   700→    category_id UUID REFERENCES blog_categories(id),
   701→    tags TEXT[],
   702→    status ENUM('draft', 'published', 'archived') DEFAULT 'draft',
   703→    published_at TIMESTAMP WITH TIME ZONE,
   704→    view_count INTEGER DEFAULT 0,
   705→    like_count INTEGER DEFAULT 0,
   706→    comment_count INTEGER DEFAULT 0,
   707→    meta_title VARCHAR(255),
   708→    meta_description TEXT,
   709→    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
   710→    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
   711→);
   712→
   713→-- جدول نظرات بلاگ (Comments)
   714→CREATE TABLE blog_comments (
   715→    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   716→    post_id UUID NOT NULL REFERENCES blog_posts(id),
   717→    user_id UUID REFERENCES users(id), -- can be null for guest
   718→    parent_id UUID REFERENCES blog_comments(id), -- for replies
   719→    content TEXT NOT NULL,
   720→    is_approved BOOLEAN DEFAULT false,
   721→    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
   722→);
   723→
   724→-- جدول تیکت‌های پشتیبانی (Support Tickets)
   725→CREATE TABLE support_tickets (
   726→    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   727→    user_id UUID NOT NULL REFERENCES users(id),
   728→    subject VARCHAR(255) NOT NULL,
   729→    description TEXT NOT NULL,
   730→    priority ENUM('low', 'medium', 'high', 'urgent') DEFAULT 'medium',
   731→    status ENUM('open', 'in_progress', 'waiting', 'resolved', 'closed') DEFAULT 'open',
   732→    assigned_to UUID REFERENCES users(id), -- admin/support agent
   733→    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
   734→    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
   735→    closed_at TIMESTAMP WITH TIME ZONE
   736→);
   737→
   738→-- جدول پیام‌های تیکت (Ticket Messages)
   739→CREATE TABLE ticket_messages (
   740→    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   741→    ticket_id UUID NOT NULL REFERENCES support_tickets(id),
   742→    sender_id UUID NOT NULL REFERENCES users(id),
   743→    message TEXT NOT NULL,
   744→    attachments JSONB, -- array of file URLs
   745→    is_read BOOLEAN DEFAULT false,
   746→    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
   747→);
   748→
   749→-- جدول کدهای OTP (One Time Passwords)
   750→CREATE TABLE otp_codes (
   751→    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   752→    phone VARCHAR(11) NOT NULL,
   753→    code VARCHAR(6) NOT NULL,
   754→    purpose ENUM('registration', 'login', 'password_reset', 'transaction') NOT NULL,
   755→    expires_at TIMESTAMP WITH TIME ZONE NOT NULL,
   756→    used_at TIMESTAMP WITH TIME ZONE,
   757→    attempts INTEGER DEFAULT 0,
   758→    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
   759→);
   760→
   761→-- جدول نوتیفیکیشن‌ها (Notifications)
   762→CREATE TABLE notifications (
   763→    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   764→    user_id UUID NOT NULL REFERENCES users(id),
   765→    title VARCHAR(255) NOT NULL,
   766→    message TEXT NOT NULL,
   767→    type ENUM('info', 'success', 'warning', 'error') DEFAULT 'info',
   768→    channel ENUM('sms', 'email', 'push', 'in_app') DEFAULT 'in_app',
   769→    data JSONB, -- additional metadata
   770→    is_read BOOLEAN DEFAULT false,
   771→    sent_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
   772→    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
   773→);
   774→
   775→-- جدول تخفیف‌ها (Coupons/Discounts)
   776→CREATE TABLE coupons (
   777→    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   778→    code VARCHAR(20) UNIQUE NOT NULL,
   779→    description TEXT,
   780→    discount_type ENUM('percentage', 'fixed') NOT NULL,
   781→    discount_value DECIMAL(10, 2) NOT NULL,
   782→    min_order_amount DECIMAL(10, 2) DEFAULT 0,
   783→    max_discount_amount DECIMAL(10, 2),
   784→    usage_limit INTEGER,
   785→    usage_count INTEGER DEFAULT 0,
   786→    valid_from TIMESTAMP WITH TIME ZONE NOT NULL,
   787→    valid_until TIMESTAMP WITH TIME ZONE NOT NULL,
   788→    is_active BOOLEAN DEFAULT true,
   789→    applicable_to ENUM('all', 'new_users', 'specific_trip_type'),
   790→    created_by UUID REFERENCES users(id),
   791→    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
   792→);
   793→
   794→-- جدول لاگ فعالیت‌ها (Activity Logs)
   795→CREATE TABLE activity_logs (
   796→    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   797→    user_id UUID REFERENCES users(id),
   798→    action VARCHAR(100) NOT NULL,
   799→    entity_type VARCHAR(50), -- e.g., 'trip', 'user', 'post'
   800→    entity_id UUID,
   801→    old_values JSONB,
   802→    new_values JSONB,
   803→    ip_address INET,
   804→    user_agent TEXT,
   805→    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
   806→);
   807→```

   813→### **احراز هویت (Auth API):**
   814→```yaml
   815→POST /api/auth/send-otp:
   816→  description: ارسال کد OTP به شماره موبایل

   818→  response: { "message": "OTP sent", "expires_in": 120 }
   819→
   820→POST /api/auth/verify-otp:
   821→  description: تأیید کد OTP
   822→  body: { "phone": "...", "code": "123456" }
   823→  response: { "token": "jwt_token", "user": {...}, "is_new_user": bool }
   824→
   825→POST /api/auth/register/passenger:
   826→  description: ثبت نام مسافر
   827→  body: { "phone", "otp", "full_name", "email?", "password" }
   828→  response: { "user": {...}, "token": "..." }
   829→
   830→POST /api/auth/register/driver:
   831→  description: ثبت نام راننده (multi-step)
   832→  body: { /* all driver registration fields */ }
   833→  response: { "message": "Registration submitted for review", "driver_id": "..." }
   834→
   835→POST /api/auth/login:
   836→  description: ورود با رمز عبور
   837→  body: { "phone/email", "password" }
   838→  response: { "token": "...", "refresh_token": "...", "user": {...} }
   839→
   840→POST /api/auth/refresh-token:
   841→  description: تازه کردن توکن
   842→  body: { "refresh_token": "..." }
   843→  response: { "token": "new_jwt" }
   844→
   845→POST /api/auth/logout:
   846→  description: خروج
   847→  headers: Authorization: Bearer <token>
   848→  response: { "message": "Logged out successfully" }
   849→
   850→POST /api/auth/forgot-password:
   851→  description: فراموشی رمز عبور
   852→  body: { "phone": "..." }
   853→  response: { "message": "Reset code sent" }
   854→
   855→PUT /api/auth/change-password:
   856→  description: تغییر رمز عبور
   857→  headers: Authorization: Bearer <token>
   858→  body: { "old_password", "new_password" }
   859→  response: { "message": "Password changed" }
   860→```
