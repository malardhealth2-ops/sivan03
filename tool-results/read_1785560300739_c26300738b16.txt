   174→function makeSlug(title: string): string {
   175→  // Persian-safe slug: keep Persian letters/digits, replace spaces with dashes,
   176→  // strip punctuation, append a short unique suffix.
   177→  const base = title
   178→    .replace(/[«»"'!?.,:;()\[\]{}]/g, '')
   179→    .replace(/\s+/g, '-')
   180→    .replace(/-+/g, '-')
   181→    .trim()
   182→    .slice(0, 60);
   183→  const suffix = Date.now().toString(36).slice(-6);
   184→  return `${base}-${suffix}`;
   185→}

   187→function parseArticle(raw: string): {
   188→  title: string;
   189→  excerpt: string;
   190→  html: string;
   191→  tags: string[];
   192→  metaDescription: string;
   193→} | null {
   194→  if (!raw) return null;
   195→  const lines = raw.split('\n');
   196→
   197→  let title = '';
   198→  let excerpt = '';
   199→  let metaDescription = '';
   200→  let tags: string[] = [];
   201→  const htmlLines: string[] = [];
   202→
   203→  for (const line of lines) {
   204→    const trimmed = line.trim();
   205→    if (!trimmed) continue;
   206→    if (trimmed.startsWith('عنوان:')) {
   207→      title = trimmed.slice(6).trim().replace(/^["'«]|["'»]$/g, '');
   208→    } else if (trimmed.startsWith('خلاصه:')) {
   209→      excerpt = trimmed.slice(6).trim().replace(/^["'«]|["'»]$/g, '');
   210→    } else if (trimmed.startsWith('متا:')) {
   211→      metaDescription = trimmed.slice(4).trim().replace(/^["'«]|["'»]$/g, '');
   212→    } else if (trimmed.startsWith('برچسب‌ها:')) {
   213→      const rest = trimmed.replace(/^برچسب‌ها:\s*/, '');
   214→      tags = rest
   215→        .split(',')
   216→        .map((t) => t.trim().replace(/^["'«]|["'»]$/g, ''))
   217→        .filter(Boolean)
   218→        .slice(0, 6);
   219→    } else {
   220→      htmlLines.push(trimmed);
   221→    }
   222→  }
   223→
   224→  // Reconstruct content HTML.
   225→  let html = htmlLines.join('\n').trim();
   226→  if (!html) return null;
   227→
   228→  // If the model didn't use HTML tags, wrap lines in <p>.
   229→  if (!/<[a-z][\s\S]*>/i.test(html)) {
   230→    html = html
   231→      .split(/\n{2,}/)
   232→      .map((p) => `<p>${p.replace(/\n/g, '<br/>')}</p>`)
   233→      .join('\n');
   234→  }
   235→
   236→  if (!title) {
   237→    const h = html.match(/<h[1-3][^>]*>([\s\S]*?)<\/h[1-3]>/i);
   238→    title = h ? h[1].replace(/<[^>]+>/g, '').trim() : html.slice(0, 60).replace(/<[^>]+>/g, '');
   239→  }
   240→  if (!excerpt) {
   241→    const firstP = html.match(/<p[^>]*>([\s\S]*?)<\/p>/i);
   242→    excerpt = firstP ? firstP[1].replace(/<[^>]+>/g, '').trim().slice(0, 180) : title;
   243→  }
   244→  if (!metaDescription) {
   245→    metaDescription = excerpt.slice(0, 155);
   246→  }
   247→  if (tags.length === 0) {
   248→    tags = ['سفر', 'تاکسی VIP', 'سیوان'];
   249→  }
   250→
   251→  return { title, excerpt, html, tags, metaDescription };
   252→}

   319→          content: `You are a professional art director for a luxury Persian travel & taxi blog. I will give you the title of a Persian article. Produce ONE single English sentence describing a striking, topic-specific cover photo that visually matches the article's ACTUAL subject.

   321→Rules (very important):
   322→- The image MUST visually depict the specific subject of the article (the exact place, exact car model, or exact concept) — NOT a generic scene.
   323→- Translate any Persian place/car name into its English equivalent so the image model understands it (e.g. "Hormuz Island", "Mercedes-Benz E-Class", "Tehran to Mashhad highway").
   324→- Photorealistic scene description only. No abstract art, no illustrations.
   325→- Absolutely NO text, words, letters, logos or watermark in the image.
   326→- Do NOT show clearly visible human faces.
   327→- Do NOT wrap the answer in quotes or markdown. Do NOT add any explanation or label. Output ONLY the one sentence.
   328→- Keep the sentence between 20 and 80 words.
   329→
   330→Article title (Persian): ${title}
   331→Article focus keyword (Persian): ${topic.keyword}
   332→
   333→End your sentence with exactly: ", ${styleHint}, high quality, no text, no watermark".
   334→
   335→Write the one-sentence English image prompt now:`,
   336→        },
   337→      ],
   338→      thinking: { type: 'disabled' },
   339→    });
   340→    const raw = (completion?.choices?.[0]?.message?.content || '').trim();
   341→    if (!raw) return null;
   342→    // Clean up any wrapping the model might add despite instructions.
   343→    const cleaned = raw
   344→      .replace(/^```[a-z]*\n?/i, '')
   345→      .replace(/\n?```$/i, '')
   346→      .replace(/^["'«]|["'»]$/g, '')
   347→      .replace(/^(prompt|image prompt|description|cover photo)\s*:\s*/i, '')
   348→      .replace(/\s+/g, ' ')
   349→      .trim();
   350→    if (cleaned.length < 15 || cleaned.length > 600) return null;
   351→    // Guarantee the quality/no-text suffix is present.
   352→    if (!/no text/i.test(cleaned)) {
   353→      return `${cleaned}, ${styleHint}, high quality, no text, no watermark`;
   354→    }
   355→    return cleaned;
   356→  } catch (err) {
   357→    console.error('[blog-generator] image-prompt LLM call failed:', err);
   358→    return null;
   359→  }

   363→  title: string,
   364→  slug: string,
   365→  topic: Topic
   366→): Promise<string | null> {
   367→  if (!zai) return null;

   369→  // matches the article's subject (not just its broad category).
   370→  let prompt = await generateImagePrompt(title, topic);

   372→    console.log(`[blog-generator] topic-specific image prompt: "${prompt.slice(0, 120)}${prompt.length > 120 ? '…' : ''}"`);
   373→  } else {
   374→    // Fallback: pick a generic scene from the category-aware pool.
   375→    const scenes = COVER_SCENES[topic.category] || COVER_SCENES['travel-guide'];
   376→    prompt = scenes[Math.floor(Math.random() * scenes.length)];
   377→    console.log(`[blog-generator] LLM prompt failed; using fallback ${topic.category} scene`);
   378→  }
   379→  try {
   380→    const response = await zai.images.generations.create({
   381→      prompt,
   382→      size: '1344x768',
   383→    });

   392→    console.error('[blog-generator] image generation failed:', err);

   398→// Derive a keyword from a free-form custom topic: take the most meaningful
   399→// words (skip very common Persian stopwords), join with space. Falls back to
   400→// the whole topic if filtering removes everything.
   401→function deriveKeyword(topic: string): string {
   402→  const stopwords = new Set([
   403→    'در', 'از', 'به', 'با', 'و', 'یا', 'را', 'است', 'نیست', 'برای', 'تا', 'که',
   404→    'این', 'آن', 'هر', 'چه', 'چگونه', 'چرا', 'کی', 'کجا', 'کدام', 'هم', 'تاکنون',
   405→    'یک', 'دو', 'سه', 'نه', 'بله', 'خیر', 'اما', 'اگر', 'چون', 'بنابراین', 'پس',
   406→    'ترین', 'ها', 'های', 'ای', 'هایی', 'شود', 'شده', 'می', 'کرد', 'کند', 'باشد',
   407→    'هستند', 'بود', 'بودن', 'ما', 'شما', 'او', 'آنها', 'خود', 'همین', 'مانند',
   408→  ]);
   409→  const words = topic
   410→    .replace(/[«»"'!?.,:;()\[\]{}،؛؟]/g, ' ')
   411→    .split(/\s+/)
   412→    .filter((w) => w.length > 1 && !stopwords.has(w));
   413→  const kw = words.slice(0, 5).join(' ').trim();
   414→  return kw || topic.slice(0, 40);
   415→}

   439→  }

   443→async function generateArticle(customTopic?: string): Promise<{ ok: boolean; title?: string; slug?: string; error?: string; custom?: boolean }> {
   444→  if (!zai) {
   445→    return { ok: false, error: 'ZAI SDK not initialized' };

   454→  // do NOT call pickTopic() — this keeps the 6-hour auto-rotation untouched

   456→  // auto-pick generations, never by custom ones).
   457→  const isCustom = !!customTopic && customTopic.trim().length > 0;
   458→  const topic: Topic = isCustom
   459→    ? {
   460→        title: customTopic!.trim().slice(0, 200),

   535→
   536→  const slug = makeSlug(parsed.title);
   537→  const featuredImageUrl = await generateCoverImage(parsed.title, slug, topic);
   538→
   539→  try {
   540→    const post = await db.blogPost.create({
   541→      data: {
   542→        title: parsed.title,
   543→        slug,
   544→        excerpt: parsed.excerpt,
   545→        content: parsed.html,
   546→        featuredImageUrl,

   549→        tags: JSON.stringify(parsed.tags),

   551→        categoryId: null,
   552→      },

   556→    isGenerating = false;

   561→    isGenerating = false;
   562→    return { ok: false, error: 'DB save failed' };

   566→async function maybeGenerateOnStartup() {
   567→  try {
   568→    const count = await db.blogPost.count({ where: { status: 'published' } });
   569→    if (count < MIN_POSTS_FOR_IMMEDIATE_SKIP) {
   570→      console.log(`[blog-generator] only ${count} published posts; generating one now...`);
   571→      setTimeout(() => generateArticle().catch((e) => console.error(e)), 5000);
   572→      return;
   573→    }
   574→    // If the most recent post is older than 6h, generate one now.
   575→    const latest = await db.blogPost.findFirst({
   576→      where: { status: 'published' },
   577→      orderBy: { publishedAt: 'desc' },
   578→      select: { publishedAt: true },
   579→    });
   580→    if (latest?.publishedAt) {
   581→      const ageMs = Date.now() - new Date(latest.publishedAt).getTime();
   582→      if (ageMs > GENERATION_INTERVAL_MS) {
   583→        console.log(`[blog-generator] last post is ${Math.round(ageMs / 3600000)}h old; generating one now...`);
   584→        setTimeout(() => generateArticle().catch((e) => console.error(e)), 5000);
   585→      } else {
   586→        console.log(`[blog-generator] last post is recent; next auto-generation in 6h.`);
   587→      }
   588→    }

   590→    console.error('[blog-generator] startup check failed:', err);
