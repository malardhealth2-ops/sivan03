   136→// ─── Invalidate Map Size Handler ─────────────────────────────────────────────

   138→function MapInvalidator() {
   139→  const map = useMap();
   140→
   141→  useEffect(() => {
   142→    // Force Leaflet to recalculate its size after the container is fully rendered
   143→    const timer = setTimeout(() => {
   144→      map.invalidateSize();
   145→    }, 200);
   146→    return () => clearTimeout(timer);
   147→  }, [map]);
   148→
   149→  // Also listen for scroll/resize events that might change the container
   150→  useEffect(() => {
   151→    let frameId: number;
   152→    const handleUpdate = () => {
   153→      if (frameId) cancelAnimationFrame(frameId);
   154→      frameId = requestAnimationFrame(() => {
   155→        map.invalidateSize({ animate: false });
   156→      });
   157→    };
   158→
   159→    window.addEventListener('resize', handleUpdate);
   160→    const observer = new IntersectionObserver((entries) => {
   161→      for (const entry of entries) {
   162→        if (entry.isIntersecting) {
   163→          setTimeout(() => map.invalidateSize({ animate: false }), 100);
   164→        }
   165→      }
   166→    }, { threshold: 0.1 });
   167→
   168→    const container = map.getContainer();
   169→    if (container) observer.observe(container);
   170→
   171→    return () => {
   172→      window.removeEventListener('resize', handleUpdate);
   173→      observer.disconnect();
   174→      if (frameId) cancelAnimationFrame(frameId);
   175→    };
   176→  }, [map]);
   177→
   178→  return null;
   179→}

   181→// ─── Fit to Bounds Handler ──────────────────────────────────────────────────
   182→
   183→function FitBoundsHandler({
   184→  points,
   185→  routeData,
   186→}: {
   187→  points: { origin: MapPoint | null; destination: MapPoint | null };
   188→  routeData: RouteResponse | null;
   189→}) {
   190→  const map = useMap();
   191→
   192→  useEffect(() => {
   193→    if (routeData && routeData.routes.length > 0 && routeData.routes[0].path.length > 0) {
   194→      const polyline = L.polyline(routeData.routes[0].path);
   195→      const bounds = polyline.getBounds().pad(0.15);
   196→      map.flyToBounds(bounds, { duration: 1.2 });
   197→    } else if (points.origin && points.destination) {
   198→      const bounds = L.latLngBounds(
   199→        [points.origin.lat, points.origin.lng],
   200→        [points.destination.lat, points.destination.lng]
   201→      ).pad(0.3);
   202→      map.flyToBounds(bounds, { duration: 1 });
   203→    } else if (points.origin) {
   204→      map.flyTo([points.origin.lat, points.origin.lng], 12, { duration: 1 });
   205→    }
   206→  }, [points, routeData, map]);
   207→
   208→  return null;
   209→}

   432→export default function InteractiveMapInner() {
   433→  const [origin, setOrigin] = useState<MapPoint | null>(null);
   434→  const [destination, setDestination] = useState<MapPoint | null>(null);
   435→  const [selectionStep, setSelectionStep] = useState<SelectionStep>('origin');
   436→  const [routeData, setRouteData] = useState<RouteResponse | null>(null);
   437→  const [routeLoading, setRouteLoading] = useState(false);
   438→  const [routeError, setRouteError] = useState('');
   439→  const [activeRoute, setActiveRoute] = useState(0);
   440→
   441→  // Search state
   442→  const [originSearch, setOriginSearch] = useState('');
   443→  const [destSearch, setDestSearch] = useState('');
   444→  const [originResults, setOriginResults] = useState<
   445→    { display_name: string; lat: string; lon: string }[]
   446→  >([]);
   447→  const [destResults, setDestResults] = useState<
   448→    { display_name: string; lat: string; lon: string }[]
   449→  >([]);
   450→  const [searchLoading, setSearchLoading] = useState<'origin' | 'dest' | null>(null);
   451→  const searchTimer = useRef<ReturnType<typeof setTimeout>>();
   452→  const [identifyingPlace, setIdentifyingPlace] = useState<'origin' | 'dest' | null>(null);
   453→
   454→  // Identify place name using VLM when clicking on map
   455→  const identifyPlace = useCallback(async (lat: number, lng: number, type: 'origin' | 'dest') => {
   456→    setIdentifyingPlace(type);
   457→    try {
   458→      const res = await fetch(`/api/map/identify-place?lat=${lat}&lng=${lng}`);
   459→      if (res.ok) {
   460→        const data = await res.json();
   461→        const name = data.name || '';
   462→        if (name) {
   463→          if (type === 'origin') {
   464→            setOriginSearch(name);
   465→            setOrigin(prev => prev ? { ...prev, name } : null);
   466→          } else {
   467→            setDestSearch(name);
   468→            setDestination(prev => prev ? { ...prev, name } : null);
   469→          }
   470→        }
   471→      }
   472→    } catch {
   473→      // Silently fail - the route API will also try to get place names
   474→    } finally {
   475→      setIdentifyingPlace(null);
   476→    }
   477→  }, []);
   478→
   479→  // Handle map click
   480→  const handleSelectPoint = useCallback(
   481→    (lat: number, lng: number) => {
   482→      if (selectionStep === 'origin') {
   483→        setOrigin({ lat, lng });
   484→        setRouteData(null);
   485→        setActiveRoute(0);
   486→        setSelectionStep('destination');
   487→        identifyPlace(lat, lng, 'origin');
   488→      } else if (selectionStep === 'destination') {
   489→        if (origin && Math.abs(origin.lat - lat) < 0.001 && Math.abs(origin.lng - lng) < 0.001) {
   490→          return;
   491→        }
   492→        setDestination({ lat, lng });
   493→        setSelectionStep('ready');
   494→        identifyPlace(lat, lng, 'destination');
   495→      } else {
   496→        clearSelection();
   497→      }
   498→    },
   499→    [selectionStep, origin, identifyPlace]
   500→  );
   501→
   502→  // Search function
   503→  const handleSearch = useCallback(
   504→    (query: string, type: 'origin' | 'dest') => {
   505→      if (searchTimer.current) clearTimeout(searchTimer.current);
   506→
   507→      if (!query || query.length < 2) {
   508→        if (type === 'origin') setOriginResults([]);
   509→        else setDestResults([]);
   510→        return;
   511→      }
   512→
   513→      setSearchLoading(type);
   514→
   515→      searchTimer.current = setTimeout(async () => {
   516→        try {
   517→          const encoded = encodeURIComponent(query + ', ایران');
   518→          const url = `https://nominatim.openstreetmap.org/search?q=${encoded}&format=json&limit=5&accept-language=fa&addressdetails=1`;
   519→
   520→          const res = await fetch(url, {
   521→            headers: {
   522→              'User-Agent': 'SivanVIPTaxi/1.0',
   523→            },
   524→          });
   525→
   526→          if (!res.ok) return;
   527→
   528→          const data = await res.json();
   529→          if (type === 'origin') {
   530→            setOriginResults(data || []);
   531→          } else {
   532→            setDestResults(data || []);
   533→          }
   534→        } catch {
   535→          // ignore search errors
   536→        } finally {
   537→          setSearchLoading(null);
   538→        }
   539→      }, 800);
   540→    },
   541→    []
   542→  );
   543→
   544→  // Handle search select
   545→  const handleSearchSelect = useCallback(
   546→    (lat: number, lng: number, name: string, type: 'origin' | 'dest') => {
   547→      if (type === 'origin') {
   548→        setOrigin({ lat, lng, name });
   549→        setOriginSearch(name);
   550→        setOriginResults([]);
   551→        if (!destination) {
   552→          setSelectionStep('destination');
   553→        } else {
   554→          setSelectionStep('ready');
   555→        }
   556→      } else {
   557→        setDestination({ lat, lng, name });
   558→        setDestSearch(name);
   559→        setDestResults([]);
   560→        if (origin) {
   561→          setSelectionStep('ready');
   562→        } else {
   563→          setSelectionStep('origin');
   564→        }
   565→      }
   566→    },
   567→    [origin, destination]
   568→  );
   569→
   570→  // Fetch route when both points selected
   571→  useEffect(() => {
   572→    if (origin && destination) {
   573→      setRouteLoading(true);
   574→      setRouteError('');
   575→
   576→      const timer = setTimeout(async () => {
   577→        try {
   578→          const res = await fetch(
   579→            `/api/map/route?originLat=${origin.lat}&originLng=${origin.lng}&destLat=${destination.lat}&destLng=${destination.lng}`
   580→          );
