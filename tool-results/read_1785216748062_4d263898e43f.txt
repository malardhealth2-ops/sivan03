    77→function getCityDisplayName(city: { province: string; city: string; district?: string; neighborhood?: string }): string {
    78→  if (city.neighborhood) {
    79→    return `${city.neighborhood}، ${city.district}، ${city.province}`;
    80→  }
    81→  if (city.district) {
    82→    return `${city.district}، ${city.province}`;
    83→  }
    84→  if (city.city) {
    85→    return `${city.city}، ${city.province}`;
    86→  }
    87→  return city.province;
    88→}

    94→export function BookingModal() {
    95→  const {
    96→    booking,
    97→    setBookingStep,
    98→    updateBookingForm,
    99→    setEstimatedPrice,
   100→    setBookingCode,
   101→    setBookingSubmitting,
   102→    resetBooking,
   103→  } = useAppStore();
   104→
   105→  const [open, setOpen] = useState(false);
   106→  const [distanceLoading, setDistanceLoading] = useState(false);
   107→  const [distanceError, setDistanceError] = useState('');
   108→  const [showCustomTime, setShowCustomTime] = useState(false);
   109→
   110→  const shouldOpen = booking.currentStep >= 0 && booking.currentStep <= 3;
   111→  const dialogOpen = open || shouldOpen;
   112→
   113→  const handleClose = (val: boolean) => {
   114→    if (!val) {
   115→      setOpen(false);
   116→      resetBooking();
   117→    }
   118→  };
   119→
   120→  const nextStep = () => {
   121→    if (booking.currentStep < 3) {
   122→      setBookingStep(booking.currentStep + 1);
   123→    }
   124→  };
   125→
   126→  const prevStep = () => {
   127→    if (booking.currentStep > 0) {
   128→      setBookingStep(booking.currentStep - 1);
   129→    }
   130→  };
   131→
   132→  // Fetch distance when both cities are selected
   133→  const fetchDistance = useCallback(async () => {
   134→    const { originCity, destCity } = booking.formData;
   135→    if (!originCity.city || !destCity.city || originCity.city === destCity.city) {
   136→      updateBookingForm({ distanceKm: null, durationMin: null });
   137→      setDistanceError('');
   138→      return;
   139→    }
   140→
   141→    setDistanceLoading(true);
   142→    setDistanceError('');
   143→
   144→    try {
   145→      const originQuery = originCity.neighborhood || originCity.district || originCity.city;
   146→      const destQuery = destCity.neighborhood || destCity.district || destCity.city;
   147→
   148→      const res = await fetch(
   149→        `/api/distance?origin=${encodeURIComponent(originQuery)}&destination=${encodeURIComponent(destQuery)}`
   150→      );
   151→
   152→      if (!res.ok) {
   153→        const err = await res.json();
   154→        throw new Error(err.error || 'خطا در محاسبه فاصله');
   155→      }
   156→
   157→      const data = await res.json();
   158→      updateBookingForm({
   159→        distanceKm: data.distanceKm,
   160→        durationMin: data.durationMin,
   161→      });
   162→    } catch (err) {
   163→      setDistanceError(err instanceof Error ? err.message : 'خطا در محاسبه فاصله');
   164→      updateBookingForm({ distanceKm: null, durationMin: null });
   165→    } finally {
   166→      setDistanceLoading(false);
   167→    }
   168→  }, [booking.formData.originCity, booking.formData.destCity, updateBookingForm]);
   169→
   170→  useEffect(() => {
   171→    const { originCity, destCity } = booking.formData;
   172→    if (originCity.city && destCity.city && originCity.city !== destCity.city) {
   173→      const timer = setTimeout(fetchDistance, 500);
   174→      return () => clearTimeout(timer);
   175→    }
   176→  }, [booking.formData.originCity.city, booking.formData.destCity.city, fetchDistance]);
   177→
   178→  const handleSubmit = async () => {
   179→    setBookingSubmitting(true);
   180→    try {
   181→      const { originCity, destCity, distanceKm, tripType, roundTrip } = booking.formData;
   182→      const originDisplay = getCityDisplayName(originCity);
   183→      const destDisplay = getCityDisplayName(destCity);
   184→
   185→      // Fetch price
   186→      let price = 500000; // fallback base
   187→      let duration = '';
   188→      if (distanceKm) {
   189→        try {
   190→          const res = await fetch(
   191→            `/api/pricing?tripType=${tripType}&distanceKm=${distanceKm}`
   192→          );
   193→          if (res.ok) {
   194→            const data = await res.json();
   195→            price = roundTrip ? data.price * 2 : data.price;
   196→            duration = data.duration;
   197→          }
   198→        } catch {
   199→          // use fallback
   200→        }
   201→      }
   202→
   203→      // Call booking API
   204→      const bookingRes = await fetch('/api/booking', {
   205→        method: 'POST',
   206→        headers: { 'Content-Type': 'application/json' },
   207→        body: JSON.stringify({
   208→          originAddress: originDisplay,
   209→          destAddress: destDisplay,
   210→          distanceKm: distanceKm || 0,
   211→          tripType,
   212→          roundTrip,
   213→          passengerCount: booking.formData.passengerCount,
   214→          date: booking.formData.date,
   215→          time: booking.formData.time,
   216→          fullName: booking.formData.fullName,
   217→          phone: booking.formData.phone,
   218→          notes: booking.formData.notes,
   219→          paymentMethod: booking.formData.paymentMethod,
   220→          totalAmount: price,
   221→        }),
   222→      });
   223→
   224→      if (!bookingRes.ok) {
   225→        const errData = await bookingRes.json().catch(() => ({ error: 'خطا در ثبت رزرو' }));
   226→        throw new Error(errData.error || 'خطا در ثبت رزرو');
   227→      }
   228→
   229→      const result = await bookingRes.json();
   230→      const code = result.bookingCode || 'SV-' + Math.random().toString(36).substring(2, 8).toUpperCase();
   231→      setBookingCode(code);
   232→      setEstimatedPrice(price, duration || `~${booking.formData.durationMin || 60} دقیقه`);
   233→
   234→      // Close modal immediately after success
   235→      setOpen(false);
   236→      resetBooking();
   237→
   238→      // Show success toast for 3 seconds
   239→      toast.success(
   240→        <div className="flex flex-col gap-1.5">
   241→          <div className="flex items-center gap-2 text-green-600 font-bold text-sm">
   242→            <Check className="h-4 w-4" />
   243→            رزرو با موفقیت ثبت شد!
   244→          </div>
   245→          <div className="text-muted-foreground text-xs">
   246→            کد رهگیری: <span className="font-bold text-foreground" dir="ltr">{code}</span>
   247→          </div>
   248→          {distanceKm && (
   249→            <div className="text-muted-foreground text-xs">
   250→              هزینه: <span className="font-bold text-foreground">{formatPrice(price)}</span>
   251→            </div>
   252→          )}
   253→        </div>,
   254→        {
   255→          duration: 3000,
   256→          className: 'toast-success-custom',
   257→        }
   258→      );
   259→    } catch {
   260→      toast.error('خطا در ثبت رزرو. لطفاً دوباره تلاش کنید.', { duration: 3000 });
   261→    } finally {
   262→      setBookingSubmitting(false);
   263→    }
   264→  };
   265→
   266→  const isStepValid = (): boolean => {
   267→    switch (booking.currentStep) {
   268→      case 0:
   269→        return booking.formData.originCity.city.length > 0 && booking.formData.destCity.city.length > 0;
   270→      case 1:
   271→        return booking.formData.date.length > 0 && booking.formData.time.length > 0;
   272→      case 2:
   273→        return true;
   274→      case 3:
   275→        return booking.formData.fullName.length > 1 && booking.formData.phone.length >= 10;
   276→      default:
   277→        return false;
   278→    }
   279→  };
   280→
   281→  const canGoNext = (): boolean => {
   282→    if (!isStepValid()) return false;
   283→    if (booking.currentStep === 0 && distanceLoading) return false;
   284→    return true;
   285→  };
   286→
   287→  return (
   288→    <Dialog open={dialogOpen} onOpenChange={handleClose}>
   289→      <DialogContent className="bg-[#1a1a1a] border-[#333] max-w-2xl max-h-[90vh] overflow-y-auto p-0">
   290→        {/* Steps indicator */}
   291→        <div className="bg-[#0a0a0a] border-b border-[#333] p-4 sticky top-0 z-10">
   292→          <DialogHeader className="p-0 mb-3">
   293→            <DialogTitle className="text-right text-[#fafafa] text-lg flex items-center gap-2 justify-end">
   294→              <img src="/logo.png" alt="" className="h-6 w-6 rounded-full object-cover" />
   295→              رزرو تاکسی سیوان
   296→            </DialogTitle>
   297→            <DialogDescription className="text-right text-[#a1a1aa] text-xs mt-1">
   298→              مراحل رزرو سفر خود را تکمیل کنید
   299→            </DialogDescription>
   300→          </DialogHeader>
   301→          <div className="flex items-center gap-2">
   302→            {steps.map((step, idx) => (
   303→              <div key={step.id} className="flex-1 flex items-center gap-2">
   304→                <div
   305→                  className={`flex-1 flex items-center justify-center gap-1.5 py-2 rounded-lg text-xs font-medium transition-all ${
   306→                    idx === booking.currentStep
   307→                      ? 'bg-[#D4AF37] text-[#0a0a0a]'
   308→                      : idx < booking.currentStep
   309→                        ? 'bg-[#D4AF37]/20 text-[#D4AF37]'
   310→                        : 'bg-[#2d2d2d] text-[#a1a1aa]'
   311→                  }`}
   312→                >
   313→                  {idx < booking.currentStep ? (
   314→                    <Check className="h-3.5 w-3.5" />
   315→                  ) : (
   316→                    <step.icon className="h-3.5 w-3.5" />
   317→                  )}
   318→                  <span className="hidden sm:inline">{step.title}</span>
   319→                </div>
   320→                {idx < steps.length - 1 && (
   321→                  <div
   322→                    className={`w-6 h-0.5 rounded-full ${
   323→                      idx < booking.currentStep ? 'bg-[#D4AF37]' : 'bg-[#333]'
   324→                    }`}
   325→                  />
   326→                )}
   327→              </div>
   328→            ))}
   329→          </div>
   330→        </div>
   331→
   332→        <div className="p-6">
   333→          <AnimatePresence mode="wait">
   334→            {/* Step 0: Route */}
   335→            {booking.currentStep === 0 && (
   336→              <motion.div
   337→                key="step-0"
   338→                initial={{ opacity: 0, x: -20 }}
   339→                animate={{ opacity: 1, x: 0 }}
   340→                exit={{ opacity: 0, x: 20 }}
   341→                transition={{ duration: 0.3 }}
   342→                className="space-y-5"
   343→              >
   344→                <CitySelector
   345→                  label="شهر مبدأ"
   346→                  iconColor="#D4AF37"
   347→                  value={booking.formData.originCity}
   348→                  onChange={(val) => {
   349→                    updateBookingForm({
   350→                      originCity: val,
   351→                      origin: val.neighborhood || val.district || val.city,
   352→                    });
   353→                  }}
   354→                  placeholder="استان و شهر مبدأ را انتخاب کنید"
   355→                />
   356→
   357→                <CitySelector
   358→                  label="شهر مقصد"
   359→                  iconColor="#E5C76B"
   360→                  value={booking.formData.destCity}
   361→                  onChange={(val) => {
   362→                    updateBookingForm({
   363→                      destCity: val,
   364→                      destination: val.neighborhood || val.district || val.city,
   365→                    });
   366→                  }}
   367→                  placeholder="استان و شهر مقصد را انتخاب کنید"
   368→                />
   369→
   370→                {/* Distance info */}
   371→                {(distanceLoading || booking.formData.distanceKm || distanceError) && (
   372→                  <div className="p-4 bg-[#0a0a0a] rounded-xl border border-[#333]">
   373→                    {distanceLoading && (
   374→                      <div className="flex items-center gap-3 text-[#D4AF37]">
   375→                        <Loader2 className="h-5 w-5 animate-spin" />
   376→                        <span className="text-sm">در حال محاسبه فاصله از نقشه...</span>
   377→                      </div>
   378→                    )}
   379→                    {booking.formData.distanceKm && !distanceLoading && (
   380→                      <div className="space-y-2">
   381→                        <div className="flex items-center gap-2 text-[#fafafa]">
   382→                          <Route className="h-5 w-5 text-[#D4AF37]" />
   383→                          <span className="font-bold">
   384→                            {new Intl.NumberFormat('fa-IR').format(booking.formData.distanceKm)} کیلومتر
   385→                          </span>
   386→                        </div>
   387→                        {booking.formData.durationMin && (
   388→                          <div className="flex items-center gap-2 text-[#a1a1aa]">
   389→                            <Navigation className="h-4 w-4" />
   390→                            <span className="text-sm">
   391→                              زمان تقریبی:{' '}
   392→                              {booking.formData.durationMin >= 60
   393→                                ? `${Math.floor(booking.formData.durationMin / 60)} ساعت و ${booking.formData.durationMin % 60} دقیقه`
   394→                                : `${booking.formData.durationMin} دقیقه`}
   395→                            </span>
   396→                          </div>
   397→                        )}
   398→                      </div>
   399→                    )}
   400→                    {distanceError && !distanceLoading && (
   401→                      <div className="flex items-start gap-2 text-red-400">
   402→                        <AlertCircle className="h-5 w-5 flex-shrink-0 mt-0.5" />
   403→                        <span className="text-sm">{distanceError}</span>
   404→                      </div>
   405→                    )}
   406→                  </div>
   407→                )}
   408→
   409→                <div className="flex items-center gap-3 p-4 bg-[#0a0a0a] rounded-xl border border-[#333]">
   410→                  <label className="flex items-center gap-2 cursor-pointer">
   411→                    <input
   412→                      type="checkbox"
   413→                      checked={booking.formData.roundTrip}
   414→                      onChange={(e) => updateBookingForm({ roundTrip: e.target.checked })}
   415→                      className="rounded border-[#D4AF37] text-[#D4AF37] focus:ring-[#D4AF37]"
   416→                    />
   417→                    <span className="text-sm text-[#a1a1aa]">سفر رفت و برگشت</span>
   418→                  </label>
   419→                </div>
   420→              </motion.div>
   421→            )}
   422→
   423→            {/* Step 1: Time */}
   424→            {booking.currentStep === 1 && (
   425→              <motion.div
   426→                key="step-1"
   427→                initial={{ opacity: 0, x: -20 }}
   428→                animate={{ opacity: 1, x: 0 }}
   429→                exit={{ opacity: 0, x: 20 }}
   430→                transition={{ duration: 0.3 }}
   431→                className="space-y-5"
   432→              >
   433→                <div className="space-y-2">
   434→                  <JalaliDatePicker
   435→                    value={booking.formData.date}
   436→                    onChange={(val) => updateBookingForm({ date: val })}
   437→                    placeholder="انتخاب تاریخ سفر"
   438→                    label="تاریخ سفر"
   439→                  />
   440→                </div>
   441→                <div className="space-y-3">
   442→                  <Label className="text-[#fafafa]">

   444→                    ساعت حرکت
