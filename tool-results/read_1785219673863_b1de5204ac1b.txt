   101→   101→// ─── Invalidate Map Size Handler ─────────────────────────────────────────────
   102→   102→
   103→   103→function MapInvalidator() {
   104→   104→  const map = useMap();
   105→   105→
   106→   106→  useEffect(() => {
   107→   107→    // Force Leaflet to recalculate its size after the container is fully rendered
   108→   108→    const timer = setTimeout(() => {
   109→   109→      map.invalidateSize();
   110→   110→    }, 200);
   111→   111→    return () => clearTimeout(timer);
   112→   112→  }, [map]);
   113→   113→
   114→   114→  // Also listen for scroll/resize events that might change the container
   115→   115→  useEffect(() => {
   116→   116→    let frameId: number;
   117→   117→    const handleUpdate = () => {
   118→   118→      if (frameId) cancelAnimationFrame(frameId);
   119→   119→      frameId = requestAnimationFrame(() => {
   120→   120→        map.invalidateSize({ animate: false });
   121→   121→      });
   122→   122→    };
   123→   123→
   124→   124→    window.addEventListener('resize', handleUpdate);
   125→   125→    const observer = new IntersectionObserver((entries) => {
   126→   126→      for (const entry of entries) {
   127→   127→        if (entry.isIntersecting) {
   128→   128→          setTimeout(() => map.invalidateSize({ animate: false }), 100);
   129→   129→        }
   130→   130→      }
   131→   131→    }, { threshold: 0.1 });
   132→   132→
   133→   133→    const container = map.getContainer();
   134→   134→    if (container) observer.observe(container);
   135→   135→
   136→   136→    return () => {
   137→   137→      window.removeEventListener('resize', handleUpdate);
   138→   138→      observer.disconnect();
   139→   139→      if (frameId) cancelAnimationFrame(frameId);
   140→   140→    };
   141→   141→  }, [map]);
   142→   142→
   143→   143→  return null;
   144→   144→}

   148→   148→function FitBoundsHandler({
   149→   149→  points,
   150→   150→  routeData,
   151→   151→}: {
   152→   152→  points: { origin: MapPoint | null; destination: MapPoint | null };
   153→   153→  routeData: RouteResponse | null;
   154→   154→}) {
   155→   155→  const map = useMap();
   156→   156→
   157→   157→  useEffect(() => {
   158→   158→    if (routeData && routeData.routes.length > 0 && routeData.routes[0].path.length > 0) {
   159→   159→      const polyline = L.polyline(routeData.routes[0].path);
   160→   160→      const bounds = polyline.getBounds().pad(0.15);
   161→   161→      map.flyToBounds(bounds, { duration: 1.2 });
   162→   162→    } else if (points.origin && points.destination) {
   163→   163→      const bounds = L.latLngBounds(
   164→   164→        [points.origin.lat, points.origin.lng],
   165→   165→        [points.destination.lat, points.destination.lng]
   166→   166→      ).pad(0.3);
   167→   167→      map.flyToBounds(bounds, { duration: 1 });
   168→   168→    } else if (points.origin) {
   169→   169→      map.flyTo([points.origin.lat, points.origin.lng], 12, { duration: 1 });
   170→   170→    }
   171→   171→  }, [points, routeData, map]);
   172→   172→
   173→   173→  return null;
   174→   174→}

   297→   297→export default function InteractiveMapInner() {
   298→   298→  const [origin, setOrigin] = useState<MapPoint | null>(null);
   299→   299→  const [destination, setDestination] = useState<MapPoint | null>(null);
   300→   300→  const [selectionStep, setSelectionStep] = useState<SelectionStep>('origin');
   301→   301→  const [routeData, setRouteData] = useState<RouteResponse | null>(null);
   302→   302→  const [routeLoading, setRouteLoading] = useState(false);
   303→   303→  const [routeError, setRouteError] = useState('');
   304→   304→  const [activeRoute, setActiveRoute] = useState(0);
   305→   305→
   306→   306→  // Search state
   307→   307→  const [originSearch, setOriginSearch] = useState('');
   308→   308→  const [destSearch, setDestSearch] = useState('');
   309→   309→  const [originResults, setOriginResults] = useState<
   310→   310→    { display_name: string; lat: string; lon: string }[]
   311→   311→  >([]);
   312→   312→  const [destResults, setDestResults] = useState<
   313→   313→    { display_name: string; lat: string; lon: string }[]
   314→   314→  >([]);
   315→   315→  const [searchLoading, setSearchLoading] = useState<'origin' | 'dest' | null>(null);
   316→   316→  const searchTimer = useRef<ReturnType<typeof setTimeout>>();
   317→   317→
   318→   318→  // Handle map click
   319→   319→  const handleSelectPoint = useCallback(
   320→   320→    (lat: number, lng: number) => {
   321→   321→      if (selectionStep === 'origin') {
   322→   322→        setOrigin({ lat, lng });
   323→   323→        setRouteData(null);
   324→   324→        setActiveRoute(0);
   325→   325→        setSelectionStep('destination');
   326→   326→      } else if (selectionStep === 'destination') {
   327→   327→        if (origin && Math.abs(origin.lat - lat) < 0.001 && Math.abs(origin.lng - lng) < 0.001) {
   328→   328→          return;
   329→   329→        }
   330→   330→        setDestination({ lat, lng });
   331→   331→        setSelectionStep('ready');
   332→   332→      } else {
   333→   333→        clearSelection();
   334→   334→      }
   335→   335→    },
   336→   336→    [selectionStep, origin]
   337→   337→  );
   338→   338→
   339→   339→  // Search function
   340→   340→  const handleSearch = useCallback(
   341→   341→    (query: string, type: 'origin' | 'dest') => {
   342→   342→      if (searchTimer.current) clearTimeout(searchTimer.current);
   343→   343→
   344→   344→      if (!query || query.length < 2) {
   345→   345→        if (type === 'origin') setOriginResults([]);
   346→   346→        else setDestResults([]);

   348→   348→      }
   349→   349→
   350→   350→      setSearchLoading(type);
   351→   351→
   352→   352→      searchTimer.current = setTimeout(async () => {
   353→   353→        try {
   354→   354→          const encoded = encodeURIComponent(query + ', ایران');
   355→   355→          const url = `https://nominatim.openstreetmap.org/search?q=${encoded}&format=json&limit=5&accept-language=fa&addressdetails=1`;
   356→   356→
   357→   357→          const res = await fetch(url, {
   358→   358→            headers: {
   359→   359→              'User-Agent': 'SivanVIPTaxi/1.0',
   360→   360→            },
   361→   361→          });
   362→   362→
   363→   363→          if (!res.ok) return;
   364→   364→
   365→   365→          const data = await res.json();
   366→   366→          if (type === 'origin') {
   367→   367→            setOriginResults(data || []);
   368→   368→          } else {
   369→   369→            setDestResults(data || []);
   370→   370→          }
   371→   371→        } catch {
   372→   372→          // ignore search errors
   373→   373→        } finally {
   374→   374→          setSearchLoading(null);
   375→   375→        }
   376→   376→      }, 800);
   377→   377→    },
   378→   378→    []

   389→   389→          setSelectionStep('destination');

   391→   391→          setSelectionStep('ready');

   398→   398→          setSelectionStep('ready');
   399→   399→        } else {
   400→   400→          setSelectionStep('origin');
   401→   401→        }
   402→   402→      }
   403→   403→    },
   404→   404→    [origin, destination]

   406→   406→
   407→   407→  // Fetch route when both points selected

   410→   410→      setRouteLoading(true);
   411→   411→      setRouteError('');
   412→   412→
   413→   413→      const timer = setTimeout(async () => {
   414→   414→        try {

   416→   416→            `/api/map/route?originLat=${origin.lat}&originLng=${origin.lng}&destLat=${destination.lat}&destLng=${destination.lng}`
   417→   417→          );
   418→   418→
   419→   419→          if (!res.ok) {

   421→   421→            throw new Error(err.error || 'خطا');
   422→   422→          }
   423→   423→
   424→   424→          const data = await res.json();
   425→   425→          setRouteData(data);
   426→   426→
   427→   427→          if (data.origin?.name && !origin.name) {
   428→   428→            setOrigin((prev) => prev ? { ...prev, name: data.origin.name } : prev);
   429→   429→            setOriginSearch(data.origin.name);
   430→   430→          }
   431→   431→          if (data.destination?.name && !destination.name) {
   432→   432→            setDestination((prev) => prev ? { ...prev, name: data.destination.name } : prev);
   433→   433→            setDestSearch(data.destination.name);
   434→   434→          }
   435→   435→        } catch (err) {
   436→   436→          setRouteError(err instanceof Error ? err.message : 'خطا در محاسبه مسیر');
   437→   437→          setRouteData(null);
   438→   438→        } finally {
   439→   439→          setRouteLoading(false);
   440→   440→        }
   441→   441→      }, 500);
   442→   442→
   443→   443→      return () => clearTimeout(timer);
   444→   444→    }
   445→   445→  }, [origin?.lat, origin?.lng, destination?.lat, destination?.lng]);
   446→   446→
   447→   447→  const clearSelection = () => {
   448→   448→    setOrigin(null);
   449→   449→    setDestination(null);
   450→   450→    setRouteData(null);
   451→   451→    setRouteError('');

   453→   453→    setOriginSearch('');
   454→   454→    setDestSearch('');
   455→   455→    setOriginResults([]);
   456→   456→    setDestResults([]);
   457→   457→    setSelectionStep('origin');
   458→   458→  };
   459→   459→
   460→   460→  const swapPoints = () => {
   461→   461→    if (!origin || !destination) return;
   462→   462→    const tmpOrigin = origin;
   463→   463→    const tmpOriginSearch = originSearch;
   464→   464→    setOrigin(destination);
   465→   465→    setDestination(tmpOrigin);
   466→   466→    setOriginSearch(destSearch);
   467→   467→    setDestSearch(tmpOriginSearch);
   468→   468→    setRouteData(null);
   469→   469→    setActiveRoute(0);
   470→   470→  };
   471→   471→
   472→   472→  // Build polylines
   473→   473→  const routePolylines = routeData?.routes.map((route) => route.path) || [];
   474→   474→
   475→   475→  return (
   476→   476→    <section id="map" className="py-20 sm:py-24 bg-[#0a0a0a] relative">
   477→   477→      {/* Decorative */}
   478→   478→      <div className="absolute top-0 left-1/2 -translate-x-1/2 w-24 h-1 bg-gradient-to-l from-transparent via-[#D4AF37] to-transparent rounded-full" />
   479→   479→
